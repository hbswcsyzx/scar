positional arguments:
  command

options:
  -h, --help            show this help message and exit
  --out OUT
  --lines               record user python_line and bytecode loop back-edge
                        events
  --resources           sample traced-process CPU and nvidia-smi GPU resources
  --resource-interval RESOURCE_INTERVAL
                        resource sampling interval in seconds (with
                        --resources)
  --calls-only          record bounded Python call sites and reference
                        identities without Torch hooks or tensor values
  --call-operands       record bounded instruction/load/call evidence; implies
                        calls-only (CPython 3.14)
  --python-call-targets
                        record explicit v2 Python callee and receiver roles;
                        implies call-operands (CPython 3.14.7)
  --capture-plan CAPTURE_PLAN
                        use a source-derived lean interface capture plan;
                        implies Python call targets
  --capture-plan-sha256 CAPTURE_PLAN_SHA256
                        expected capture plan SHA256
  --capture-profile {program-ports,process-ports}
                        CALL-only monitoring and source bytes: project
                        descendants (V1), or installed target-process Python
                        threads (V2)
  --max-source-files MAX_SOURCE_FILES
  --max-source-file-bytes MAX_SOURCE_FILE_BYTES
  --max-source-bytes MAX_SOURCE_BYTES
                        raw source bytes read once per observed original file
                        (ports profiles only)
  --execution-evidence  collect native ETO/Kineto evidence alongside the call
                        recorder when the target imports Torch; requires
                        calls-only or call-operands
  --tensor-physical     also export owned profiler marker Tensor metadata and
                        allocation epochs after close; requires execution-
                        evidence
  --operand-capture {windows,instructions}
                        capture bounded call windows (default) or every
                        project instruction
  --project-root PROJECT_ROOT
                        source boundary for lightweight call-site recording;
                        defaults to target cwd
  --max-call-events MAX_CALL_EVENTS
  --max-call-bytes MAX_CALL_BYTES
  --max-call-codes MAX_CALL_CODES
  --max-call-references MAX_CALL_REFERENCES
                        bounded active weak identities shared by frame/operand
                        observers; no application contents
  --capture-transport {jsonl,snapshot}
                        JSONL in callbacks, or bounded raw snapshots with
                        JSONL conversion after collection stops
  --snapshot-buffer-bytes SNAPSHOT_BUFFER_BYTES
                        DRAM byte ceiling per snapshot stream; writer threads
                        perform IO only
  --max-operand-events MAX_OPERAND_EVENTS
  --max-operand-bytes MAX_OPERAND_BYTES
  --os-capture-root OS_CAPTURE_ROOT
                        opt in to owned syscall observation, lossless
                        retention and final coordinates; requires fresh ROOT
                        and --out ROOT/trace
  --os-profile {program-interfaces,all}
                        explicit OS scope; program-interfaces requests seccomp
                        filtering and excludes unlisted internals
  --archive-workers ARCHIVE_WORKERS
                        1..32 bounded compression/IO workers after capture
                        closes
  --strace-executable STRACE_EXECUTABLE
                        strace executable for the owned opt-in OS capture
  --max-os-bytes MAX_OS_BYTES
  --stream-interfaces   observe builtin stream calls and C completion without
                        reading contents or wrapping objects
  --max-os-lines MAX_OS_LINES
  --max-capture-bytes MAX_CAPTURE_BYTES
  --max-archive-bytes MAX_ARCHIVE_BYTES
  --max-archive-files MAX_ARCHIVE_FILES
                        lossless archive transport member count, including
                        per-TID OS files, native chunks and collector members;
                        not event coverage
  --os-timeout OS_TIMEOUT
                        recording seconds with --os-postprocess-timeout;
                        otherwise original total provider wall limit
  --os-postprocess-timeout OS_POSTPROCESS_TIMEOUT
                        explicit finite shutdown/conversion seconds after
                        qualified main atexit; total <= os-timeout + this,
                        plus 5s termination grace; absent keeps old total
                        budget
