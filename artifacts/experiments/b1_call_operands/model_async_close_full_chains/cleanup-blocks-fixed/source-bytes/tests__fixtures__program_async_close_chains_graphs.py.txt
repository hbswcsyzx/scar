_ROUTES = ('graceful-return', 'graceful-ge', 'graceful-ordinary', 'illegal')


def _op(graph, identity):
    return next(row for row in graph.operations if row.id == identity)


def _own(operation, kind, slot=0):
    number = next(i for i, edge in enumerate(operation.terminator.successors) if edge.kind is kind)
    return InvokeLiteralResult(ControlEdgeReference(operation.id, number), CompletionResult(slot))


def _body(route):
    """Prime is in the protected scope; only its packet ingress enters finally."""
    b = _Body('close.body')
    root = ControlScopeID('close.body.function')
    protected = ControlScopeID('close.body.protected')
    cleanup = ControlScopeID('close.body.finally')
    return_port, raise_port = ControlPortID('close.body.return'), ControlPortID('close.body.raise')
    pending, packets, enters, resumes, replacements = [], [], [], [], []
    protected_blocks, cleanup_blocks, root_blocks = [], [], []

    def args(name, kinds):
        return tuple(b.argument(f'{name}.v{i}', name, i, kind) for i, kind in enumerate(kinds))

    def body_role(value):
        item = next(row for row in b.values if row.id == value)
        return UseRole.STATE_CONTENT if type(item) is StateVersionValue else UseRole.PROTOCOL if type(item) is ProtocolValue else UseRole.REFERENCE

    def jump(name, target, actuals):
        uses = tuple(_body_use(f'{name}.u{i}', value, body_role(value)) for i, value in enumerate(actuals))
        return b.add(name, (Behavior.CONTROL,), uses, effects_complete=True,
            terminator=Terminator(TerminatorKind.JUMP, (Successor(BlockID(target), tuple(use.id for use in uses)),)))

    def pack(name, payload, port):
        actual = _body_use(name + '.payload', payload, body_role(payload))
        result = b.result(name + '.token', name, 0, P.PENDING_COMPLETION)
        operation = b.add(name, (Behavior.CONTROL,), (actual,), (result,), effects_complete=True)
        definition = PendingCompletionDefinition(result, port, actual.id)
        pending.append(definition)
        return result, operation, definition, OwnedUse(operation.id, actual.id)

    def route_into(name, token, states, caller, live, error, port):
        uses = (_body_use(name + '.selector', token),
            *(_body_use(name + f'.arg.{i}', value, body_role(value))
              for i, value in enumerate((token, *states, caller, live, error))))
        return b.add(name, (Behavior.CONTROL,), uses, effects_complete=True,
            terminator=CleanupRouteTerminator(uses[0].id, (
                CleanupRouteSuccessor(port, BlockID('close.body.cleanup'), tuple(use.id for use in uses[1:])),)))

    none = ReferenceValue(ValueID('close.body.none'), b.region, LiteralOrigin(None), targets=())
    b.values.append(none)
    entry = args('close.body.entry', ('reference', 'reference', b.world, b.context))
    function_entry = args('close.body.start', ('reference', 'reference', b.world, b.context))
    entry_jump = jump('close.body.start.enter', 'close.body.entry', function_entry)
    b.block('close.body.start', function_entry, (entry_jump.id.value,))
    root_blocks.append(BlockID('close.body.start'))
    frame = b.initial('close.body.initial')
    caller, context = b.switch('close.body.initial.attach', frame, entry[3], attach=True)
    saved_frame, detached = b.switch('close.body.prime.detach', caller, context, attach=False)
    normal = args('close.body.prime.normal', (b.world, P.FRAME_CONTEXT, 'reference', b.context, 'reference', 'reference'))
    raised = args('close.body.prime.packet', (b.context, P.FRAME_CONTEXT, P.EXCEPTION_PACKET, b.world, 'reference', 'reference'))
    saved = (_body_use('close.body.prime.frame', saved_frame),
             _body_use('close.body.prime.live', entry[0], UseRole.REFERENCE),
             _body_use('close.body.prime.error', entry[1], UseRole.REFERENCE))
    export = _body_use('close.body.prime.export', entry[0], UseRole.REFERENCE)
    world = _body_use('close.body.prime.world', entry[2], UseRole.STATE_CONTENT)
    ambient = _body_use('close.body.prime.context', detached, UseRole.STATE_CONTENT)
    prime = b.add('close.body.prime', (Behavior.CONTROL,), (export, *saved, world, ambient),
        effects_complete=True, terminator=SuspendTerminator(SuspensionFlavor.ASYNC_YIELD, export.id,
            (SavedSlot(S.FRAME_CONTEXT, saved[0].id), SavedSlot(S.VALUE, saved[1].id), SavedSlot(S.VALUE, saved[2].id)),
            (world.id, ambient.id), (
                ResumeSuccessor((ResumeSlot(I.STATE, b.context), ResumeSlot(I.SENT_VALUE), ResumeSlot(I.STATE, b.world)),
                    BlockID('close.body.prime.normal'), (ResumeInput(2), SavedInput(0), ResumeInput(1), ResumeInput(0), SavedInput(1), SavedInput(2))),
                ResumeSuccessor((ResumeSlot(I.THROWN_PACKET), ResumeSlot(I.STATE, b.world), ResumeSlot(I.STATE, b.context)),
                    BlockID('close.body.prime.packet'), (ResumeInput(2), SavedInput(0), ResumeInput(0), ResumeInput(1), SavedInput(1), SavedInput(2))))))
    b.block('close.body.entry', entry, ('close.body.initial', 'close.body.initial.attach', 'close.body.prime.detach', prime.id.value))
    protected_blocks.append(BlockID('close.body.entry'))
    prime_payloads = []
    for name, formals, frame_index, world_index, context_index, payload, port, live_index, error_index in (
        ('normal', normal, 1, 0, 3, none.id, return_port, 4, 5),
        ('packet', raised, 1, 3, 0, raised[2], raise_port, 4, 5),
    ):
        prefix = 'close.body.prime.' + name
        attached, state = b.switch(prefix + '.attach', formals[frame_index], formals[context_index], attach=True)
        token, producer, definition, payload_use = pack(prefix + '.pending', payload, port)
        transfer = route_into(prefix + '.route', token, (formals[world_index], state), attached,
                              formals[live_index], formals[error_index], port)
        b.block(prefix, formals, (prefix + '.attach', producer.id.value, transfer.id.value))
        protected_blocks.append(BlockID(prefix))
        prime_payloads.append({'kind': name, 'source': payload_use, 'definition': definition, 'operation': producer})

    start = args('close.body.cleanup', (P.PENDING_COMPLETION, b.world, b.context, P.DRIVE_CALLER_CONTEXT, 'reference', 'reference'))
    enter_id = OperationID('close.body.cleanup.enter')
    enter_pending = _body_use('close.body.enter.pending', start[0])
    before = (_body_use('close.body.enter.world', start[1], UseRole.STATE_CONTENT),
              _body_use('close.body.enter.context', start[2], UseRole.STATE_CONTENT))
    saved_context = b.result('close.body.cleanup.saved_context', enter_id.value, 0, P.SAVED_HANDLED_CONTEXT)
    entered_world = b.result('close.body.cleanup.world', enter_id.value, 1, b.world)
    entered_context = b.result('close.body.cleanup.context', enter_id.value, 2, b.context)
    enter = b.add(enter_id.value, (Behavior.CONTROL, Behavior.READ, Behavior.WRITE), (enter_pending, *before),
        (saved_context, entered_world, entered_context), effects_complete=False, accesses=(
            StateAccess(b.world, before[0].id, AccessMode.READ_WRITE, entered_world, Footprint(FootprintKind.WHOLE)),
            StateAccess(b.context, before[1].id, AccessMode.READ_WRITE, entered_context, Footprint(FootprintKind.WHOLE))))
    enters.append(CleanupEnter(enter.id, cleanup, enter_pending.id, saved_context))
    suspended_frame, suspended_context = b.switch('close.body.cleanup.detach', start[3], entered_context, attach=False)
    saved_uses = tuple(_body_use(f'close.body.cleanup.saved.{i}', value, body_role(value))
        for i, value in enumerate((suspended_frame, start[0], saved_context, start[4], start[5])))
    normal_kinds = (P.PENDING_COMPLETION, b.world, P.FRAME_CONTEXT, P.SAVED_HANDLED_CONTEXT,
                    b.context, 'reference', 'reference', 'reference')
    packet_kinds = (*normal_kinds[:5], P.EXCEPTION_PACKET, *normal_kinds[6:])
    landings = (args('close.body.cleanup.normal', normal_kinds), args('close.body.cleanup.packet', packet_kinds))
    export = _body_use('close.body.cleanup.export', start[4], UseRole.REFERENCE)
    states = (_body_use('close.body.cleanup.handoff.world', entered_world, UseRole.STATE_CONTENT),
              _body_use('close.body.cleanup.handoff.context', suspended_context, UseRole.STATE_CONTENT))
    cut = b.add('close.body.cleanup.suspend', (Behavior.CONTROL,), (export, *saved_uses, *states), effects_complete=True,
        terminator=SuspendTerminator(SuspensionFlavor.ASYNC_YIELD if route == 'illegal' else SuspensionFlavor.AWAIT,
            export.id, tuple(SavedSlot(role, use.id) for role, use in zip(
                (S.FRAME_CONTEXT, S.PENDING_COMPLETION, S.CLEANUP_CONTEXT, S.VALUE, S.VALUE), saved_uses)),
            tuple(use.id for use in states), tuple(ResumeSuccessor(
                (ResumeSlot(I.STATE, b.context), ResumeSlot(I.SENT_VALUE if i == 0 else I.THROWN_PACKET), ResumeSlot(I.STATE, b.world)),
                BlockID('close.body.cleanup.' + name),
                (SavedInput(1), ResumeInput(2), SavedInput(0), SavedInput(2), ResumeInput(0), ResumeInput(1), SavedInput(3), SavedInput(4)))
                for i, name in enumerate(('normal', 'packet')))))
    b.block('close.body.cleanup', start, (enter.id.value, 'close.body.cleanup.detach', cut.id.value))
    cleanup_blocks.append(BlockID('close.body.cleanup'))
    action_details = []
    for name, formals in zip(('normal', 'packet'), landings):
        prefix = 'close.body.cleanup.' + name
        caller, context = b.switch(prefix + '.attach', formals[2], formals[4], attach=True)
        active = [prefix + '.attach']
        replacement = None
        if name == 'packet':
            replacement, producer, definition, payload_use = pack(prefix + '.replacement', formals[5], raise_port)
            active.append(producer.id.value)
        elif route == 'graceful-return':
            replacement, producer, definition, payload_use = pack(prefix + '.replacement', none.id, return_port)
            active.append(producer.id.value)
        elif route == 'graceful-ordinary':
            exception = _body_use(prefix + '.exception', formals[7], UseRole.REFERENCE)
            traceback = _body_use(prefix + '.traceback', none.id, UseRole.REFERENCE)
            packet = b.result(prefix + '.fresh_packet', prefix + '.pack', 0, P.EXCEPTION_PACKET)
            packet_op = b.add(prefix + '.pack', (Behavior.CONTROL,), (exception, traceback), (packet,), effects_complete=True)
            packets.append(ExceptionPacketDefinition(packet, exception.id, traceback.id))
            replacement, producer, definition, payload_use = pack(prefix + '.replacement', packet, raise_port)
            active.extend((packet_op.id.value, producer.id.value))
        action_block = prefix + '.replace'
        action_formals = args(action_block, (P.PENDING_COMPLETION, P.SAVED_HANDLED_CONTEXT,
            b.world, b.context, P.DRIVE_CALLER_CONTEXT, *((P.PENDING_COMPLETION,) if replacement is not None else ())))
        prepare = jump(prefix + '.prepare', action_block,
            (formals[0], formals[3], formals[1], context, caller,
             *((replacement,) if replacement is not None else ())))
        b.block(prefix, formals, (*active, prepare.id.value))
        cleanup_blocks.append(BlockID(prefix))
        action_id = OperationID(prefix + '.action')
        uses = [_body_use(prefix + '.old', action_formals[0]), _body_use(prefix + '.saved_context', action_formals[1]),
                _body_use(prefix + '.world', action_formals[2], UseRole.STATE_CONTENT),
                _body_use(prefix + '.context', action_formals[3], UseRole.STATE_CONTENT)]
        if replacement is not None:
            uses.append(_body_use(prefix + '.new', action_formals[5]))
        world_after = b.result(prefix + '.world_after', action_id.value, 0, b.world)
        context_after = b.result(prefix + '.context_after', action_id.value, 1, b.context)
        action = b.add(action_id.value, (Behavior.CONTROL, Behavior.READ, Behavior.WRITE), uses,
            (world_after, context_after), effects_complete=False, accesses=(
                StateAccess(b.world, uses[2].id, AccessMode.READ_WRITE, world_after, Footprint(FootprintKind.WHOLE)),
                StateAccess(b.context, uses[3].id, AccessMode.READ_WRITE, context_after, Footprint(FootprintKind.WHOLE))))
        token = action_formals[0] if replacement is None else action_formals[5]
        transfer = jump(prefix + '.continue', 'close.body.route', (token, world_after, context_after, action_formals[4]))
        if replacement is None:
            record = CleanupResume(action.id, cleanup, uses[0].id, uses[1].id, ControlEdgeReference(transfer.id, 0))
            resumes.append(record)
        else:
            record = CleanupReplace(action.id, cleanup, uses[0].id, uses[4].id, uses[1].id, ControlEdgeReference(transfer.id, 0))
            replacements.append(record)
        b.block(action_block, action_formals, (action.id.value, transfer.id.value))
        cleanup_blocks.append(BlockID(action_block))
        action_details.append({'operation': action, 'record': record, 'continuation': transfer,
            'replacement': replacement, 'payload_use': None if replacement is None else payload_use})

    route_args = args('close.body.route', (P.PENDING_COMPLETION, b.world, b.context, P.DRIVE_CALLER_CONTEXT))
    dispatch_args = args('close.body.dispatch', (b.world, P.DRIVE_CALLER_CONTEXT, P.PENDING_COMPLETION, b.context))
    selector = _body_use('close.body.route.selector', route_args[0])
    uses, edges = [selector], []
    output_cases = ((raise_port, 'raise'),) if route == 'graceful-ordinary' else ((return_port, 'return'), (raise_port, 'raise'))
    for number, (port, _) in enumerate(output_cases):
        forwards = tuple(_body_use(f'close.body.route.{number}.{i}', value, body_role(value))
            for i, value in enumerate((route_args[1], route_args[3], route_args[0], route_args[2])))
        uses.extend(forwards)
        edges.append(CleanupRouteSuccessor(port, BlockID('close.body.dispatch'), tuple(use.id for use in forwards)))
    route_op = b.add('close.body.route.op', (Behavior.CONTROL,), uses, effects_complete=True,
        terminator=CleanupRouteTerminator(selector.id, tuple(edges)))
    final_args = {name: args('close.body.' + name,
        ('reference' if name == 'return' else P.EXCEPTION_PACKET, b.world, b.context, P.DRIVE_CALLER_CONTEXT))
        for _, name in output_cases}
    selector = _body_use('close.body.dispatch.selector', dispatch_args[2])
    uses, edges = [selector], []
    for number, (port, name) in enumerate(output_cases):
        forwards = tuple(_body_use(f'close.body.dispatch.{number}.{i}', value, body_role(value))
            for i, value in enumerate((dispatch_args[0], dispatch_args[3], dispatch_args[1])))
        uses.extend(forwards)
        edges.append(PendingSuccessor(port, BlockID('close.body.' + name), (PendingPayload(), *(use.id for use in forwards))))
    dispatch = b.add('close.body.dispatch.op', (Behavior.CONTROL,), uses, effects_complete=True,
        terminator=PendingDispatchTerminator(selector.id, tuple(edges)))
    b.block('close.body.route', route_args, (route_op.id.value,))
    b.block('close.body.dispatch', dispatch_args, (dispatch.id.value,))
    terminals = {}
    for name, formals in final_args.items():
        _, context = b.switch('close.body.' + name + '.detach', formals[3], formals[2], attach=False)
        uses = tuple(_body_use(f'close.body.{name}.out.{i}', value, body_role(value))
            for i, value in enumerate((formals[0], formals[1], context)))
        terminal = b.add('close.body.' + name + '.terminal', (Behavior.CONTROL,), uses, effects_complete=True,
            terminator=Terminator(TerminatorKind.RETURN if name == 'return' else TerminatorKind.RAISE,
                exit='normal' if name == 'return' else 'failure', values=tuple(use.id for use in uses)))
        b.block('close.body.' + name, formals, ('close.body.' + name + '.detach', terminal.id.value))
        terminals[name] = terminal
    root_blocks.extend(BlockID('close.body.' + name) for name in ('route', 'dispatch', *terminals))
    state_types = tuple(ValueType(ValueKind.STATE, location=location) for location in (b.world, b.context))
    graph = b.finish('close.body.start', (ExitPort('normal', ExitKind.NORMAL, (REFERENCE, *state_types)),
        ExitPort('failure', ExitKind.RAISE, (PACKET, *state_types))))
    scopes = (ControlScope(root, b.region, None, ControlScopeKind.FUNCTION, _source('close body'), tuple(root_blocks)),
        ControlScope(protected, b.region, root, ControlScopeKind.BLOCK, _source('close protected'), tuple(protected_blocks)),
        ControlScope(cleanup, b.region, root, ControlScopeKind.CLEANUP, _source('close finally'), tuple(cleanup_blocks)))
    environment_values = tuple(ReferenceValue(ValueID('close.body.environment.' + role.value), b.region,
        BlockArgument(BlockID('close.body.start'), len(function_entry) + i)) for i, role in enumerate(tuple(EnvironmentRole)[:5]))
    environment = EnvironmentInterface(EnvironmentInterfaceID('close.body.environment'), b.region,
        OperationDefinitionID('close.body.definition'), tuple(EnvironmentMember(role, value.id)
            for role, value in zip(tuple(EnvironmentRole)[:5], environment_values)))
    region = replace(graph.regions[0], blocks=tuple(replace(block,
        arguments=(*block.arguments, *(value.id for value in environment_values))) if block.id == graph.regions[0].entry else block
        for block in graph.regions[0].blocks))
    graph = replace(graph, regions=(region,), values=(*graph.values, *environment_values),
        control_scopes=scopes, control_ports=(ControlPort(return_port, root, LanguageCompletionKind.RETURN, exit='normal'),
            ControlPort(raise_port, root, LanguageCompletionKind.RAISE, exit='failure')),
        pending_completions=tuple(pending), exception_packets=tuple(packets),
        resumable_bodies=tuple(replace(row, kind=ResumableKind.ASYNC_GENERATOR) for row in graph.resumable_bodies),
        handled_context_domains=(HandledContextDomain(b.region, b.context),),
        cleanup_interfaces=(CleanupInterface(cleanup, protected, BlockID('close.body.cleanup'), start[0]),),
        cleanup_enters=tuple(enters), cleanup_resumes=tuple(resumes), cleanup_replacements=tuple(replacements),
        environments=(environment,), function_entries=(FunctionEntryLayout(environment.id,
            (FormalPort(ValueSlotID('close.body.parameter.live'), function_entry[0]), FormalPort(ValueSlotID('close.body.parameter.error'), function_entry[1]))),))
    return graph, {'region': region, 'prime': prime, 'cleanup_suspend': cut, 'terminals': terminals,
        'cleanup': {'scope': cleanup, 'interface': graph.cleanup_interfaces[0], 'enter': enter,
            'pending_payload_uses': tuple(row['source'] for row in prime_payloads),
            'pending_sources': tuple(prime_payloads), 'actions': tuple(action_details),
            'action': action_details[0]['operation'], 'dispatch': dispatch, 'route': route_op,
            'saved_slots': cut.terminator.saved, 'attaches': graph.frame_context_attaches,
            'detaches': graph.frame_context_detaches}, 'none': none}


class _CloseDriver:
    """One synchronous request invocation; body suspension remains in B."""

    def __init__(self, body, number, mode, route):
        self.body, self.number, self.mode, self.route = body, number, mode, route
        self.domains = tuple(location.id for location in body.locations)
        self.state_types = tuple(ValueType(ValueKind.STATE, location=location) for location in self.domains)
        self.b = _Builder('close.driver.' + str(number), self.domains)
        self.scope = ProtocolScope(ProtocolScopeID(f'close.driver.{number}.scope'),
            ProtocolProfile.ASYNC_GENERATOR_BODY_V1, DriverSubject(self.b.region), _source_gap())
        self.rows, self.packets, self.updates, self.bindings = [], [], [], []
        self.commits, self.releases, self.matchers, self.refinements = [], [], [], []
        self.replies, self.choices, self.closed_reads, self.closed_writes = [], {}, [], []
        self.none = ReferenceValue(ValueID(f'close.driver.{number}.none'), self.b.region, LiteralOrigin(None), targets=())
        self.b.values.append(self.none)
        self.types = (REFERENCE, RAW, REFERENCE, REFERENCE, RAW, *self.state_types)

    def terminal(self, name, payload, states, *, fail):
        operation = self.b.terminal(name, (payload, *states), fail=fail)
        operation = replace(operation, effects_complete=True)
        self.b.operations[-1] = operation
        return operation

    def guard(self, name, arguments, types, owner_index, subject, yes, no, selected):
        b = self.b
        owner = _use(name + '.owner', arguments[owner_index])
        before = tuple(_use(name + f'.state.{i}', value, state=True)
                       for i, value in enumerate(arguments[-len(self.domains):]))
        result = DataValue(ValueID(name + '.bool'), b.region, OperationResult(OperationID(name), 0), 'builtins.bool')
        b.values.append(result)
        read = Operation(OperationID(name), 'original current predicate read',
            (Behavior.CONTROL, Behavior.READ), (owner, *before), (result.id,),
            tuple(StateAccess(location, use.id, AccessMode.READ) for location, use in zip(self.domains, before)),
            effects_complete=True)
        b.operations.append(read)
        condition = Use(UseID(name + '.condition'), result.id, UseRole.DATA)
        forwards, edges, outputs = [], [], []
        values = {value.id: value for value in b.values}
        for i, target in enumerate((yes, no)):
            formals = b.formals(target, types)
            uses = tuple(Use(UseID(name + f'.edge.{i}.{j}'), value,
                UseRole.STATE_CONTENT if type(values[value]) is StateVersionValue else
                UseRole.PROTOCOL if type(values[value]) is ProtocolValue else UseRole.REFERENCE)
                for j, value in enumerate(arguments))
            forwards.extend(uses)
            edges.append(Successor(BlockID(target), tuple(use.id for use in uses)))
            outputs.append(formals)
        branch = b.add(name + '.branch', (condition, *forwards), effects_complete=True,
            terminator=Terminator(TerminatorKind.BRANCH, tuple(edges), condition=condition.id))
        self.choices[branch.id] = selected
        sites = tuple(ExistingAccess(read.id, i) for i in range(len(before)))
        if subject == 'initial' or subject == 'complete':
            condition_row = RequestStateCondition(owner.id, sites,
                RequestStateRule.IS_INITIAL_STEP if subject == 'initial' else RequestStateRule.IS_COMPLETE)
        elif subject == 'finished':
            condition_row = BodyStateCondition(owner.id, sites, BodyStatePredicateRule.IS_CLOSED)
        elif subject == 'busy':
            condition_row = SessionOccupancyCondition(owner.id, sites)
        else:
            condition_row = None
        if condition_row is not None:
            self.rows.append(ProtocolOperation(self.scope.id, ProtocolClause.ASYNC_REQUEST_SOURCE,
                read.id, 0, AdmissionPredicate(condition_row, result.origin, branch.id), _source_gap()))
        else:
            # This is an addressed native field READ, not lifecycle F. A partial
            # refinement supplies no isolation or field-content constancy proof.
            from scar.ir.program.state_addresses import (AddressCoverage, AddressEffect,
                ObjectField, ReferenceOwner, StateAddressRefinement)
            self.refinements.append(StateAddressRefinement(ExistingAccess(read.id, 0),
                (AddressEffect(ReferenceOwner(owner.id), (ObjectField('ag_closed'),), AccessMode.READ),),
                AddressCoverage.PARTIAL))
            self.closed_reads.append(read)
        return read, branch, tuple(outputs)

    def update(self, name, request, body, states, post):
        b = self.b
        request_use, body_use = _use(name + '.request', request), _use(name + '.body', body)
        before = tuple(_use(name + f'.state.{i}', value, state=True) for i, value in enumerate(states))
        after = StateVersionValue(ValueID(name + '.world_after'), b.region, OperationResult(OperationID(name), 0), self.domains[0])
        b.values.append(after)
        accesses = tuple(StateAccess(location, use.id, AccessMode.READ_WRITE if i == 0 else AccessMode.READ,
            after.id if i == 0 else None, Footprint(FootprintKind.WHOLE))
            for i, (location, use) in enumerate(zip(self.domains, before)))
        operation = Operation(OperationID(name), 'original session metadata before reply construction',
            (Behavior.CONTROL, Behavior.READ, Behavior.WRITE), (request_use, body_use, *before),
            (after.id,), accesses, effects_complete=True)
        b.operations.append(operation)
        declaration = SessionUpdate(operation.id, request_use.id, body_use.id,
                                    LifecyclePostBinding.AFTER_OPERATION, post)
        self.updates.append(declaration)
        return operation, declaration, (after.id, *states[1:])

    def constructor(self, name, input_value, states, *, source=None, rule=None, to_match=False):
        b = self.b
        normal = b.formals(name + '.normal', (REFERENCE, *self.state_types))
        raised = b.formals(name + '.raise', (PACKET, *self.state_types))
        actual = _use(name + '.input', input_value, protocol=type(next(value for value in b.values if value.id == input_value)) is ProtocolValue)
        before = tuple(_use(name + f'.state.{i}', value, state=True) for i, value in enumerate(states))
        operation = b.add(name, (actual, *before), effects_complete=False,
            invoke_arguments=(actual.id, *(use.id for use in before)),
            completions=(CompletionSignature(ExitKind.NORMAL, (REFERENCE, *self.state_types)),
                         CompletionSignature(ExitKind.RAISE, (PACKET, *self.state_types))),
            terminator=Terminator(TerminatorKind.INVOKE, (
                CompletionSuccessor(ExitKind.NORMAL, BlockID(name + '.normal'), tuple(CompletionResult(i) for i in range(1 + len(states)))),
                CompletionSuccessor(ExitKind.RAISE, BlockID(name + '.raise'), tuple(CompletionResult(i) for i in range(1 + len(states)))))))
        exception, traceback = _use(name + '.exception', normal[0]), _use(name + '.traceback', self.none.id)
        packet = b.value(name + '.packet', OperationResult(OperationID(name + '.pack'), 0), PACKET)
        pack = b.add(name + '.pack', (exception, traceback), results=(packet,), effects_complete=True)
        definition = ExceptionPacketDefinition(packet, exception.id, traceback.id)
        self.packets.append(definition)
        if to_match:
            normal_jump = self.jump(name + '.normal.continue', 'close.reply.match.stop_async', (packet, *normal[1:]))
            raised_jump = self.jump(name + '.raise.continue', 'close.reply.match.stop_async', raised)
            b.block(name + '.normal', normal, (pack, normal_jump))
            b.block(name + '.raise', raised, (raised_jump,))
            hosts = ()
            self.choices[normal_jump.id] = self.choices[raised_jump.id] = 0
        else:
            normal_host = self.terminal(name + '.normal.host', packet, normal[1:], fail=True)
            raised_host = self.terminal(name + '.raise.host', raised[0], raised[1:], fail=True)
            b.block(name + '.normal', normal, (pack, normal_host))
            b.block(name + '.raise', raised, (raised_host,))
            hosts = (normal_host, raised_host)
        if source is not None:
            for i in range(2):
                self.rows.append(ProtocolOperation(self.scope.id,
                    ProtocolClause.CLOSE_ROUTE if rule is OutcomeAdapterRule.ASYNCGEN_CLOSE_YIELD_ERROR else ProtocolClause.ASYNC_ITEM_ADAPTER,
                    operation.id, i, OutcomeAdapter(source, actual.id,
                        InvokeLiteralResult(ControlEdgeReference(operation.id, i), CompletionResult(0)), rule), _source_gap()))
        details = {'stage': self.number, 'source': source, 'matcher': None, 'branch': None,
            'constructor': operation, 'input_use': OwnedUse(operation.id, actual.id),
            'own_results_by_kind': {kind: tuple(_own(operation, kind, i) for i in range(1 + len(states)))
                                    for kind in (ExitKind.NORMAL, ExitKind.RAISE)},
            'packet_definition': definition, 'pack': pack, 'hosts': hosts}
        self.replies.append(details)
        self.choices[operation.id] = 0
        return operation, details

    def jump(self, name, target, actuals):
        values = {value.id: value for value in self.b.values}
        uses = tuple(Use(UseID(name + f'.{i}'), value,
            UseRole.STATE_CONTENT if type(values[value]) is StateVersionValue else
            UseRole.PROTOCOL if type(values[value]) is ProtocolValue else UseRole.REFERENCE)
            for i, value in enumerate(actuals))
        return self.b.add(name, uses, effects_complete=True,
            terminator=Terminator(TerminatorKind.JUMP, (Successor(BlockID(target), tuple(use.id for use in uses)),)))

    def close_reply(self):
        """The error matcher consumes the intermediate packet, not BodyReturn."""
        b = self.b
        kinds = (PACKET, *self.state_types)
        entries = {name: b.formals(name, kinds) for name in
            ('close.reply.match.stop_async', 'close.reply.match.ge', 'close.reply.matched', 'close.reply.propagate')}
        choices = []
        for name, category, yes, no in (
            ('close.reply.match.stop_async', BuiltinExceptionCategory.STOP_ASYNC_ITERATION, 'close.reply.matched', 'close.reply.match.ge'),
            ('close.reply.match.ge', BuiltinExceptionCategory.GENERATOR_EXIT, 'close.reply.matched', 'close.reply.propagate')):
            formals = entries[name]
            actual = _use(name + '.packet', formals[0], protocol=True)
            before = tuple(_use(name + f'.state.{i}', value, state=True) for i, value in enumerate(formals[1:]))
            result = DataValue(ValueID(name + '.bool'), b.region, OperationResult(OperationID(name), 0), 'builtins.bool')
            b.values.append(result)
            operation = Operation(OperationID(name), 'fixed close check_error matcher',
                (Behavior.READ, Behavior.COMPUTE), (actual, *before), (result.id,),
                tuple(StateAccess(location, use.id, AccessMode.READ) for location, use in zip(self.domains, before)), effects_complete=True)
            b.operations.append(operation)
            condition = Use(UseID(name + '.condition'), result.id, UseRole.DATA)
            uses, edges = [condition], []
            for i, target in enumerate((yes, no)):
                forwarded = tuple(_use(name + f'.forward.{i}.{j}', value, state=j > 0, protocol=j == 0)
                                  for j, value in enumerate(formals))
                uses.extend(forwarded)
                edges.append(Successor(BlockID(target), tuple(use.id for use in forwarded)))
            branch = b.add(name + '.branch', uses, effects_complete=True,
                terminator=Terminator(TerminatorKind.BRANCH, tuple(edges), condition=condition.id))
            matcher = ExceptionMatcher(operation.id, actual.id, BlockArgument(BlockID('close.reply.match.stop_async'), 0),
                FixedBuiltinHandler(category, _source_gap()), result.origin, None, branch.id,
                tuple(ExistingAccess(operation.id, i) for i in range(len(before))))
            self.matchers.append(matcher)
            b.block(name, formals, (operation, branch))
            self.choices[branch.id] = (0 if self.route == 'graceful-return' else 1) if category is BuiltinExceptionCategory.STOP_ASYNC_ITERATION else (0 if self.route == 'graceful-ge' else 1)
            choices.append((matcher, branch))
        current = entries['close.reply.matched']
        constructor, reply = self.constructor('close.reply.stop_iteration', self.none.id, current[1:])
        b.block('close.reply.matched', current, (constructor,))
        reply.update(matcher=choices[0][0], branch=choices[0][1], matchers=tuple(item[0] for item in choices),
                     branches=tuple(item[1] for item in choices), control_source=BlockArgument(BlockID('close.reply.match.stop_async'), 0))
        ordinary = entries['close.reply.propagate']
        terminal = self.terminal('close.reply.propagate.host', ordinary[0], ordinary[1:], fail=True)
        b.block('close.reply.propagate', ordinary, (terminal,))
        return reply, terminal

    def finish(self):
        from scar.ir.program.driver_flow import body_drive_target, body_outcome_uses
        from scar.ir.program.resumption_flow import first_body_outcome_operations
        b, mode = self.b, self.mode
        entry_types = (REFERENCE, RAW, REFERENCE, *self.state_types)
        entry = b.formals('close.driver.entry', entry_types)
        read, branch, (reuse, associate) = self.guard('close.request.complete', entry, entry_types, 0,
            'complete', 'close.driver.reuse', 'close.driver.associate', 0 if mode == 'retry' else 1)
        b.block('close.driver.entry', entry, (read, branch))
        error, _ = self.constructor('close.reuse.error', self.none.id, reuse[-len(self.domains):])
        b.block('close.driver.reuse', reuse, (error,))

        associated = b.formals('close.driver.associated', self.types)
        failed = b.formals('close.driver.association.failed', (PACKET, *self.state_types))
        actuals = (_use('close.association.request', associate[0]),
            *(_use(f'close.association.state.{i}', value, state=True) for i, value in enumerate(associate[-len(self.domains):])))
        forwarded = (_use('close.association.forward.request', associate[0]),
            _use('close.association.forward.scheduler', associate[1], protocol=True),
            _use('close.association.forward.sent', associate[2]))
        association = b.add('close.association', (*actuals, *forwarded), effects_complete=False,
            invoke_arguments=tuple(use.id for use in actuals),
            completions=(CompletionSignature(ExitKind.NORMAL, (REFERENCE, RAW, *self.state_types)),
                         CompletionSignature(ExitKind.RAISE, (PACKET, *self.state_types))),
            terminator=Terminator(TerminatorKind.INVOKE, (
                CompletionSuccessor(ExitKind.NORMAL, BlockID('close.driver.associated'),
                    (*(use.id for use in forwarded), *(CompletionResult(i) for i in range(2 + len(self.domains))))),
                CompletionSuccessor(ExitKind.RAISE, BlockID('close.driver.association.failed'),
                    tuple(CompletionResult(i) for i in range(1 + len(self.domains)))))))
        b.block('close.driver.associate', associate, (association,))
        failed_host = self.terminal('close.association.failed.host', failed[0], failed[1:], fail=True)
        b.block('close.driver.association.failed', failed, (failed_host,))
        self.choices[association.id] = 0
        association_row = ProtocolOperation(self.scope.id, ProtocolClause.ASYNC_REQUEST_ASSOCIATE, association.id, 0,
            RequestAssociation(actuals[0].id, _own(association, ExitKind.NORMAL, 0), _own(association, ExitKind.NORMAL, 1),
                tuple(InvokeStateOperand(association.id, use.id) for use in actuals[1:])), _source_gap())
        self.rows.append(association_row)

        read, branch, (finished, not_finished) = self.guard('close.body.finished', associated, self.types, 3,
            'finished', 'close.driver.finished', 'close.driver.not_finished', 1)
        b.block('close.driver.associated', associated, (read, branch))
        post, _, current = self.update('close.finished.complete', finished[0], finished[3], finished[-len(self.domains):], SessionPost.COMPLETE)
        reply, _ = self.constructor('close.finished.reply', self.none.id, current)
        b.block('close.driver.finished', finished, (post, reply))
        read, branch, (initial, subsequent) = self.guard('close.request.initial', not_finished, self.types, 0,
            'initial', 'close.driver.initial', 'close.driver.subsequent', 1 if mode in ('resume', 'retry') else 0)
        b.block('close.driver.not_finished', not_finished, (read, branch))
        read, branch, (busy, free) = self.guard('close.body.busy', initial, self.types, 3,
            'busy', 'close.driver.busy', 'close.driver.free', 1)
        b.block('close.driver.initial', initial, (read, branch))
        post, _, current = self.update('close.busy.complete', busy[0], busy[3], busy[-len(self.domains):], SessionPost.COMPLETE)
        error, _ = self.constructor('close.busy.error', self.none.id, current)
        b.block('close.driver.busy', busy, (post, error))
        if mode in ('prime', 'next'):
            # ASend/NEXT can resume a suspended frame even after ag_closed was
            # set by an earlier failing aclose. It has no athrow closed-bit gate.
            opened = b.formals('close.driver.open', self.types)
            transfer = self.jump('close.free.to_open', 'close.driver.open', free)
            b.block('close.driver.free', free, (transfer,))
            self.choices[transfer.id] = 0
        else:
            read, branch, (closed, opened) = self.guard('close.body.closed_bit', free, self.types, 3,
                None, 'close.driver.closed', 'close.driver.open', 0 if mode in ('new-close', 'new-throw') else 1)
            b.block('close.driver.free', free, (read, branch))
            post, _, current = self.update('close.closed.complete', closed[0], closed[3], closed[-len(self.domains):], SessionPost.COMPLETE)
            reply, _ = self.constructor('close.closed.reply', self.none.id, current)
            b.block('close.driver.closed', closed, (post, reply))

        # The open path and ITER path carry independent exact tuples. The
        # request's latent argument is used only for an initial close/throw.
        delivery_kind = 'packet' if mode in ('close', 'new-close', 'new-throw') else 'value'
        delivery_type = PACKET if delivery_kind == 'packet' else REFERENCE
        delivery_types = (REFERENCE, RAW, REFERENCE, REFERENCE, RAW, delivery_type, *self.state_types)
        delivery = b.formals('close.driver.delivery', delivery_types)
        if delivery_kind == 'packet':
            normal = b.formals('close.normalize.normal', delivery_types)
            failed = b.formals('close.normalize.raise', (PACKET, *self.state_types))
            raw = _use('close.normalize.latent', opened[4], protocol=True)
            before = tuple(_use(f'close.normalize.state.{i}', value, state=True) for i, value in enumerate(opened[-len(self.domains):]))
            forwarded = tuple(_use(f'close.normalize.forward.{i}', value, protocol=i in (1, 4)) for i, value in enumerate(opened[:5]))
            normalize = b.add('close.normalize', (raw, *before, *forwarded), effects_complete=False,
                invoke_arguments=(raw.id, *(use.id for use in before)),
                completions=(CompletionSignature(ExitKind.NORMAL, (PACKET, *self.state_types)),
                             CompletionSignature(ExitKind.RAISE, (PACKET, *self.state_types))),
                terminator=Terminator(TerminatorKind.INVOKE, (
                    CompletionSuccessor(ExitKind.NORMAL, BlockID('close.normalize.normal'),
                        (*(use.id for use in forwarded), CompletionResult(0), *(CompletionResult(i + 1) for i in range(len(self.domains))))),
                    CompletionSuccessor(ExitKind.RAISE, BlockID('close.normalize.raise'),
                        tuple(CompletionResult(i) for i in range(1 + len(self.domains)))))))
            # Setting ag_closed precedes GE injection; it is an ordinary field
            # effect, separate from F and SessionPost.
            current = opened[-len(self.domains):]
            closed_write = None
            if mode in ('close', 'new-close'):
                closed_write, current = self.closed_write(opened[3], current)
            normalize = replace(normalize, uses=tuple(replace(use, value=current[next(i for i, candidate in enumerate(before) if candidate.id == use.id)])
                if use.id in {candidate.id for candidate in before} else use for use in normalize.uses))
            b.operations[b.operations.index(next(op for op in b.operations if op.id == normalize.id))] = normalize
            b.block('close.driver.open', opened, ((closed_write, normalize) if mode in ('close', 'new-close') else (normalize,)))
            jump = self.jump('close.normalize.continue', 'close.driver.delivery', normal)
            b.block('close.normalize.normal', normal, (jump,))
            host = self.terminal('close.normalize.failed.host', failed[0], failed[1:], fail=True)
            b.block('close.normalize.raise', failed, (host,))
            self.choices[normalize.id] = self.choices[jump.id] = 0
            rule = RequestProcessingRule.NORMALIZE_ASYNC_THROW if mode == 'new-throw' else RequestProcessingRule.NORMALIZE_GENERATOR_EXIT
            for i in range(2):
                self.rows.append(ProtocolOperation(self.scope.id, ProtocolClause.ASYNC_REQUEST_DELIVERY,
                    normalize.id, i, RequestProcessing(raw.id, (), InvokeLiteralResult(ControlEdgeReference(normalize.id, i), CompletionResult(0)),
                        tuple(InvokeStateOperand(normalize.id, use.id) for use in before), rule), _source_gap()))
            # The counterfactual ITER branch still sends current S. Its body
            # target is a packet ingress for this fixed close/throw site.
            fallback, _ = self.constructor('close.iter.unexpanded', subsequent[2], subsequent[-len(self.domains):])
            b.block('close.driver.subsequent', subsequent, (fallback,))
        else:
            for name, formals in (('open', opened), ('subsequent', subsequent)):
                jump = self.jump('close.' + name + '.delivery', 'close.driver.delivery',
                    (*formals[:5], formals[2], *formals[-len(self.domains):]))
                b.block('close.driver.' + name, formals, (jump,))
                self.choices[jump.id] = 0

        target = BodyStart(self.body.root) if mode == 'prime' else BodyResume(
            OperationID('close.body.prime') if mode == 'close' else OperationID('close.body.cleanup.suspend'),
            1 if delivery_kind == 'packet' else 0)
        # FinalBodyGate owns its exact post-state even when a rejection is
        # counterfactual in this particular supplied capture.
        ingress = b.formals('close.driver.ingress', delivery_types)
        rejected = b.formals('close.driver.gate.raise', (PACKET, *self.state_types))
        values = {value.id: value for value in b.values}
        roles = tuple(_use(f'close.gate.role.{i}', value, protocol=type(values[value]) is ProtocolValue) for i, value in enumerate(delivery[:6]))
        before = tuple(_use(f'close.gate.state.{i}', value, state=True) for i, value in enumerate(delivery[6:]))
        forwards = tuple(_use(f'close.gate.forward.{i}', value, protocol=type(values[value]) is ProtocolValue) for i, value in enumerate(delivery[:6]))
        gate = b.add('close.gate', (*roles, *before, *forwards), effects_complete=True,
            invoke_arguments=tuple(use.id for use in (*roles, *before)),
            completions=(CompletionSignature(ExitKind.NORMAL, self.state_types), CompletionSignature(ExitKind.RAISE, (PACKET, *self.state_types))),
            terminator=Terminator(TerminatorKind.INVOKE, (
                CompletionSuccessor(ExitKind.NORMAL, BlockID('close.driver.ingress'),
                    (*(use.id for use in forwards), *(CompletionResult(i) for i in range(len(self.domains))))),
                CompletionSuccessor(ExitKind.RAISE, BlockID('close.driver.gate.raise'), tuple(CompletionResult(i) for i in range(1 + len(self.domains)))))))
        b.block('close.driver.delivery', delivery, (gate,))
        host = self.terminal('close.gate.failed.host', rejected[0], rejected[1:], fail=True)
        b.block('close.driver.gate.raise', rejected, (host,))
        self.choices[gate.id] = 0
        drive_uses = tuple(_use(f'close.drive.role.{i}', value, protocol=type(values[value]) is ProtocolValue) for i, value in enumerate(ingress[:6]))
        state_uses = tuple(_use(f'close.drive.state.{i}', value, state=True) for i, value in enumerate(ingress[6:]))
        drive = BodyDrive(OperationID('close.drive'), drive_uses[3].id, drive_uses[1].id, drive_uses[0].id,
            target, tuple(BodyStateInput(location, use.id) for location, use in zip(self.domains, state_uses)),
            () if mode == 'prime' else (BodyResumeInput(0 if mode == 'close' else 1, drive_uses[5].id),))
        _, body_entry, _ = body_drive_target(self.body, drive)
        sources = first_body_outcome_operations(self.body.regions[0], {op.id: op for op in self.body.operations}, body_entry)
        edges, forwards, post_blocks = [], [], []
        body_values = {value.id: value for value in self.body.values}
        for i, source_id in enumerate(sources):
            source_op = _op(self.body, source_id)
            payload_uses = body_outcome_uses(source_op)
            own_types = tuple(value_type(body_values[next(use.value for use in source_op.uses if use.id == identity)]) for identity in payload_uses)
            formals = b.formals(f'close.drive.outcome.{i}', (*own_types, REFERENCE, REFERENCE))
            copied = (_use(f'close.drive.{i}.request', ingress[0]), _use(f'close.drive.{i}.body', ingress[3]))
            forwards.extend(copied)
            edges.append(BodyDriveSuccessor(source_id, BlockID(f'close.drive.outcome.{i}'),
                (*(BodyDriveResult(j) for j in range(len(own_types))), *(use.id for use in copied))))
            post_blocks.append((source_op, formals))
        actual_roles = tuple(drive_uses[i] for i in (0, 1, 3, *((5,) if mode != 'prime' else ())))
        drive_op = b.add('close.drive', (*actual_roles, *state_uses, *forwards), effects_complete=True,
            terminator=BodyDriveTerminator(tuple(edges)))
        b.block('close.driver.ingress', ingress, (drive_op,))
        selected_source = (OperationID('close.body.prime') if mode == 'prime' else
            OperationID('close.body.cleanup.suspend') if mode == 'close' else
            OperationID('close.body.return.terminal') if self.route == 'graceful-return'
            else OperationID('close.body.raise.terminal'))
        selected_successor = next(i for i, edge in enumerate(edges) if edge.source == selected_source)
        self.choices[drive_op.id] = selected_successor
        selected_update = selected_commit = None
        for i, (source_op, formals) in enumerate(post_blocks):
            body_source = DriveOutcomeRef(drive.operation, source_op.id)
            states, request, owner = formals[1:1 + len(self.domains)], formals[-2], formals[-1]
            marker, marker_states = self.publication(f'close.publish.{i}', owner, states, body_source)
            kind = source_op.terminator.kind
            flavor = source_op.terminator.flavor if type(source_op.terminator) is SuspendTerminator else None
            post = SessionPost.HOLD if flavor is SuspensionFlavor.AWAIT else SessionPost.COMPLETE
            update_op, update, current = self.update(f'close.session.{i}', request, owner, marker_states, post)
            self.bindings.append(AsyncSessionOutcome(self.scope.id, update.operation, body_source))
            active = [marker, update_op]
            if kind in (TerminatorKind.RETURN, TerminatorKind.RAISE):
                release, current = self.release(f'close.release.{i}', owner, current)
                active.append(release)
            source = DriveResult(body_source, 0)
            if flavor is SuspensionFlavor.AWAIT:
                host = self.terminal(f'close.handoff.{i}.host', formals[0], current, fail=False)
                active.append(host)
            elif flavor is SuspensionFlavor.ASYNC_YIELD:
                constructor, _ = self.constructor(f'close.item.{i}.constructor', formals[0], current, source=source,
                    rule=OutcomeAdapterRule.ASYNCGEN_CLOSE_YIELD_ERROR if mode == 'close' and self.route == 'illegal'
                         else OutcomeAdapterRule.ASYNCGEN_YIELD_TO_REQUEST_ITEM)
                active.append(constructor)
            elif kind is TerminatorKind.RETURN:
                constructor, _ = self.constructor(f'close.completion.{i}.constructor', formals[0], current, source=source,
                    rule=OutcomeAdapterRule.ASYNCGEN_RETURN_TO_STOP_ASYNC_ITERATION, to_match=mode == 'resume')
                active.append(constructor)
            elif mode == 'resume':
                jump = self.jump(f'close.exception.{i}.continue', 'close.reply.match.stop_async', (formals[0], *current))
                active.append(jump)
                self.choices[jump.id] = 0
            else:
                host = self.terminal(f'close.exception.{i}.host', formals[0], current, fail=True)
                active.append(host)
            b.block(f'close.drive.outcome.{i}', formals, active)
            if source_op.id == selected_source:
                selected_update, selected_commit = update, self.commits[-1]
        if mode == 'resume':
            self.close_reply()
        actual_kinds = {ExitKind.RAISE if op.terminator.kind is TerminatorKind.RAISE else ExitKind.NORMAL
            for op in b.operations if op.terminator is not None and op.terminator.kind in (TerminatorKind.RETURN, TerminatorKind.RAISE)}
        region = b.finish('close.driver.entry')
        region = replace(region, exits=tuple(row for row in region.exits if row.kind in actual_kinds))
        interface = DriverInterface(region.id, self.body.root, DriverReceiverKind.ASYNC_REQUEST,
            entry[0], entry[1], (self.domains[0],), driver_rule())
        gate_row = FinalBodyGate(gate.id, drive.operation, 3, 1, 0, () if mode == 'prime' else (5,))
        graph = replace(self.body, root=region.id, regions=(region, *self.body.regions),
            operations=(*self.body.operations, *b.operations), values=(*self.body.values, *b.values),
            control_scopes=(*self.body.control_scopes, _scope(region.id, region.blocks)),
            driver_interfaces=(interface,), final_body_gates=(gate_row,), body_drives=(drive,),
            driver_lifecycles=(DriverLifecycle(region.id),), body_outcome_commits=tuple(self.commits),
            terminal_frame_releases=tuple(self.releases), session_updates=tuple(self.updates),
            protocol_scopes=(self.scope,), protocol_operations=tuple(self.rows), protocol_bindings=tuple(self.bindings),
            request_association_reads=(RequestAssociationRead(association.id, actuals[0].id,
                NormalResultPort(0), NormalResultPort(1), driver_rule()),),
            exception_packets=(*self.body.exception_packets, *self.packets), exception_matches=tuple(self.matchers),
            state_address_refinements=tuple(self.refinements))
        path = _selected_path(graph, region, self.choices)
        selected_host = _op(graph, next(block.operations[-1] for block in region.blocks if block.id == path[-1][0]))
        actual_drive = None if mode in ('retry', 'new-close', 'new-throw') else drive
        return graph, {'driver_region': region, 'interface': interface, 'scope': self.scope,
            'path': path, 'path_choices': dict(self.choices), 'host_terminal': selected_host,
            'drive': actual_drive, 'available_drive': drive, 'gate': gate_row,
            'outcome': None if actual_drive is None else selected_commit.outcome,
            'outcome_source': None if actual_drive is None else selected_commit.outcome.source,
            'selected_successor': selected_successor,
            'publication': None if actual_drive is None else selected_commit,
            'commit': None if actual_drive is None else selected_commit,
            'update': (next((row for row in self.updates if row.operation == OperationID('close.closed.complete')), None)
                       if mode in ('new-close', 'new-throw') else None if actual_drive is None else selected_update),
            'association': association_row, 'association_operation': association,
            'association_body_result': _own(association, ExitKind.NORMAL, 0),
            'entry_formals': entry, 'scheduler': entry[1], 'delivery': entry[2],
            'closed_reads': tuple(self.closed_reads), 'closed_writes': tuple(self.closed_writes),
            'reply_stages': tuple(self.replies), 'releases': tuple(self.releases), 'mode': mode}

    def closed_write(self, owner, states):
        from scar.ir.program.state_addresses import (AddressCoverage, AddressEffect,
            ObjectField, ReferenceOwner, StateAddressRefinement)
        b = self.b
        identity = OperationID('close.native.mark_closed')
        own = _use('close.native.mark_closed.body', owner)
        before = _use('close.native.mark_closed.world', states[0], state=True)
        after = StateVersionValue(ValueID('close.native.mark_closed.world_after'), b.region,
            OperationResult(identity, 0), self.domains[0])
        b.values.append(after)
        operation = Operation(identity, 'native ag_closed field write; not frame F',
            (Behavior.WRITE,), (own, before), (after.id,),
            (StateAccess(self.domains[0], before.id, AccessMode.WRITE, after.id, Footprint(FootprintKind.WHOLE)),), effects_complete=False)
        b.operations.append(operation)
        self.refinements.append(StateAddressRefinement(ExistingAccess(identity, 0),
            (AddressEffect(ReferenceOwner(own.id), (ObjectField('ag_closed'),), AccessMode.WRITE),), AddressCoverage.PARTIAL))
        self.closed_writes.append(operation)
        return operation, (after.id, *states[1:])

    def publication(self, name, owner, states, outcome):
        b = self.b
        instance = _use(name + '.body', owner)
        before = tuple(_use(name + f'.state.{i}', value, state=True) for i, value in enumerate(states))
        after = StateVersionValue(ValueID(name + '.after'), b.region, OperationResult(OperationID(name), 0), self.domains[0])
        b.values.append(after)
        operation = Operation(OperationID(name), 'first original body publication',
            (Behavior.CONTROL, Behavior.READ, Behavior.WRITE), (instance, *before), (after.id,),
            tuple(StateAccess(location, use.id, AccessMode.READ_WRITE if i == 0 else AccessMode.READ,
                after.id if i == 0 else None, Footprint(FootprintKind.WHOLE))
                for i, (location, use) in enumerate(zip(self.domains, before))), effects_complete=True)
        b.operations.append(operation)
        self.commits.append(BodyOutcomeCommit(operation.id, outcome, instance.id))
        return operation, (after.id, *states[1:])

    def release(self, name, owner, states):
        b = self.b
        instance = _use(name + '.owner', owner)
        before = tuple(_use(name + f'.state.{i}', value, state=True) for i, value in enumerate(states))
        afters = tuple(StateVersionValue(ValueID(name + f'.after.{i}'), b.region,
            OperationResult(OperationID(name), i), location) for i, location in enumerate(self.domains))
        b.values.extend(afters)
        operation = Operation(OperationID(name), 'conditional terminal frame reference release',
            (Behavior.READ, Behavior.WRITE, Behavior.MEMORY), (instance, *before), tuple(value.id for value in afters),
            tuple(StateAccess(location, use.id, AccessMode.READ_WRITE, value.id, Footprint(FootprintKind.WHOLE))
                for location, use, value in zip(self.domains, before, afters)), effects_complete=False)
        b.operations.append(operation)
        self.releases.append(TerminalFrameRelease(operation.id, instance.id))
        return operation, tuple(value.id for value in afters)


def _part(graph, details, number):
    """Namespace this driver's original IDs, without copying its body region."""
    from tests.fixtures.program_async_pipeline_graphs import _driver_component
    part = _driver_component(graph, f'close.stage.{number}')
    rename = part['rename']
    driver_operations = {identity for block in details['driver_region'].blocks for identity in block.operations}
    part['exception_matches'] = rename(tuple(row for row in graph.exception_matches if row.operation in driver_operations))
    part['state_address_refinements'] = rename(graph.state_address_refinements)
    part['request_association_reads'] = rename(graph.request_association_reads)
    return part, rename(details)


def _body_paths(body, metadata, route, stages):
    """Exact paths stop at the named original first outcome of each Drive."""
    from tests.fixtures.execution_async_pipeline_graphs import _body_path
    operations = {op.id: op for op in body.operations}
    region = body.regions[0]
    result = {}
    prior = None
    for i, stage in enumerate(stages):
        drive = stage['drive']
        if drive is None:
            continue
        if prior is None:
            start = region.entry
        else:
            start = operations[prior].terminator.successors[drive.target.successor].block
        source = stage['outcome_source']
        if source in (metadata['prime'].id, metadata['cleanup_suspend'].id):
            path = _body_path(body, region, start, source)
        else:
            kind = LanguageCompletionKind.RETURN if route == 'graceful-return' else LanguageCompletionKind.RAISE
            ports = {port.id: port for port in body.control_ports}
            selected_port = next(i for i, edge in enumerate(metadata['cleanup']['route'].terminator.successors)
                                 if ports[edge.port].kind is kind)
            target = BlockID('close.body.return' if route == 'graceful-return' else 'close.body.raise')
            path = ((start, 0), (BlockID(start.value + '.replace'), 0), (BlockID('close.body.route'), selected_port),
                    (BlockID('close.body.dispatch'), selected_port), (target, None))
        result[i] = path
        prior = source
    return result


def async_close_chains_program(*, route='graceful-return'):
    """Return one B and its complete selected graceful/illegal-close pipeline.

    Every constructor's N/R stays in the graph. The default supplied path takes
    constructor NORMAL; callers may select the exact alternative original edges
    when rebuilding an ExecutionModel. No source is chosen by an observed bool.
    """
    if route not in _ROUTES:
        raise ValueError('unknown async close chain route')
    body, body_details = _body(route)
    modes = ('prime', 'close', 'retry', 'new-close', 'new-throw', 'next') if route == 'illegal' else ('prime', 'close', 'resume')
    components, stages = [], []
    for i, mode in enumerate(modes):
        graph, details = _CloseDriver(body, i, mode, route).finish()
        part, details = _part(graph, details, i)
        components.append(part)
        stages.append(details)
    body_region = body_details['region']
    domains = tuple(location.id for location in body.locations)
    state_types = tuple(ValueType(ValueKind.STATE, location=location) for location in domains)
    body_values = {value.id: value for value in body.values}
    body_entry = next(block for block in body_region.blocks if block.id == body_region.entry)
    capture_formals = tuple(value for value in body_entry.arguments if type(body_values[value]) is ReferenceValue)
    b = _Builder('close.caller', domains)
    entry = b.formals('close.caller.entry', (*(REFERENCE for _ in capture_formals), REFERENCE, *state_types))
    initial_states = entry[-len(domains):]
    sent_argument = entry[len(capture_formals)]
    none = ReferenceValue(ValueID('close.caller.none'), b.region, LiteralOrigin(None), targets=())
    b.values.append(none)
    body_created = b.formals('close.caller.body.created', (REFERENCE, *state_types))
    body_failed = b.formals('close.caller.body.failed', (PACKET, *state_types))
    actuals = tuple(_use(f'close.caller.create.capture.{i}', value) for i, value in enumerate(entry[:len(capture_formals)]))
    state_uses = tuple(_use(f'close.caller.create.state.{i}', value, state=True) for i, value in enumerate(initial_states))
    create = b.add('close.caller.body.create', (*actuals, *state_uses), effects_complete=True,
        invoke_arguments=tuple(use.id for use in (*actuals, *state_uses)),
        completions=(CompletionSignature(ExitKind.NORMAL, (REFERENCE, *state_types)),
                     CompletionSignature(ExitKind.RAISE, (PACKET, *state_types))),
        terminator=Terminator(TerminatorKind.INVOKE, (
            CompletionSuccessor(ExitKind.NORMAL, BlockID('close.caller.body.created'),
                tuple(CompletionResult(i) for i in range(1 + len(domains)))),
            CompletionSuccessor(ExitKind.RAISE, BlockID('close.caller.body.failed'),
                tuple(CompletionResult(i) for i in range(1 + len(domains)))))))
    capture_rows = tuple(CreationCapture(formal, use.id) for formal, use in zip(capture_formals, actuals))
    function = next(member.reference for member in body.environments[0].members if member.role is EnvironmentRole.FUNCTION_OBJECT)
    creation = ResumableCreate(create.id, body_region.id,
        next(row.source for row in capture_rows if row.formal == function), capture_rows, NormalResultPort(0), driver_rule())
    b.block('close.caller.entry', entry, (create,))
    b.block('close.caller.body.failed', body_failed, (b.terminal('close.caller.body.failed.host', body_failed, fail=True),))
    current_block, current_formals = 'close.caller.body.created', body_created
    current_body, current_states = body_created[0], tuple(body_created[1:])
    current_request, prior_packet = None, None
    requests, raws, calls, catches, caller_choices = [], [], [], [], {create.id: 0}
    active = []

    def catch(name, packet, states):
        position = next(i for i, domain in enumerate(domains) if domain == body.resumable_bodies[0].context_location)
        payload = _use(name + '.packet', packet, protocol=True)
        before = _use(name + '.context', states[position], state=True)
        after = StateVersionValue(ValueID(name + '.after'), b.region, OperationResult(OperationID(name), 0), domains[position])
        b.values.append(after)
        operation = Operation(OperationID(name), 'original caught completion context install',
            (Behavior.CONTROL, Behavior.READ, Behavior.WRITE), (payload, before), (after.id,),
            (StateAccess(domains[position], before.id, AccessMode.READ_WRITE, after.id, Footprint(FootprintKind.WHOLE)),), effects_complete=True)
        b.operations.append(operation)
        catches.append({'operation': operation, 'packet': OwnedUse(operation.id, payload.id),
            'before': tuple(states), 'after': tuple(after.id if i == position else value for i, value in enumerate(states))})
        return operation, catches[-1]['after']

    for i, (stage, mode, component) in enumerate(zip(stages, modes, components)):
        make_request = mode not in ('resume', 'retry')
        if make_request:
            request_kind = RawRequestKind.NEXT if mode in ('prime', 'next') else RawRequestKind.THROW if mode == 'new-throw' else RawRequestKind.CLOSE
            latent_uses = (_use(f'close.caller.request.{i}.argument', entry[1]),) if request_kind is RawRequestKind.THROW else ()
            latent = b.value(f'close.caller.request.{i}.latent', OperationResult(OperationID(f'close.caller.request.{i}.pack'), 0), RAW)
            latent_pack = b.add(f'close.caller.request.{i}.pack', latent_uses, results=(latent,), effects_complete=True)
            raw = RawRequestDefinition(latent, request_kind, tuple(use.id for use in latent_uses))
            raws.append(raw)
            active.append(latent_pack)
            created_name, failed_name = f'close.caller.request.{i}.created', f'close.caller.request.{i}.failed'
            created = b.formals(created_name, (REFERENCE, REFERENCE, *state_types))
            failed = b.formals(failed_name, (PACKET, *state_types))
            own = (_use(f'close.caller.request.{i}.body', current_body),
                _use(f'close.caller.request.{i}.latent_use', latent, protocol=True),
                *(_use(f'close.caller.request.{i}.state.{j}', value, state=True) for j, value in enumerate(current_states)))
            forwarded = _use(f'close.caller.request.{i}.forward.body', current_body)
            request_create = b.add(f'close.caller.request.{i}.create', (*own, forwarded), effects_complete=True,
                invoke_arguments=tuple(use.id for use in own),
                completions=(CompletionSignature(ExitKind.NORMAL, (REFERENCE, *state_types)),
                             CompletionSignature(ExitKind.RAISE, (PACKET, *state_types))),
                terminator=Terminator(TerminatorKind.INVOKE, (
                    CompletionSuccessor(ExitKind.NORMAL, BlockID(created_name),
                        (forwarded.id, CompletionResult(0), *(CompletionResult(j + 1) for j in range(len(domains))))),
                    CompletionSuccessor(ExitKind.RAISE, BlockID(failed_name), tuple(CompletionResult(j) for j in range(1 + len(domains)))))))
            declaration = AsyncRequestCreate(request_create.id, own[0].id, own[1].id, NormalResultPort(0), driver_rule())
            requests.append({'stage': i, 'kind': request_kind, 'create': declaration, 'operation': request_create,
                'body_use': OwnedUse(request_create.id, own[0].id), 'latent': raw, 'result': _own(request_create, ExitKind.NORMAL),
                'current_states': tuple(current_states), 'normal_formals': created, 'raise_formals': failed})
            b.block(current_block, current_formals, (*active, request_create))
            b.block(failed_name, failed, (b.terminal(failed_name + '.host', failed, fail=True),))
            caller_choices[request_create.id] = 0
            current_block, current_formals = created_name, created
            current_body, current_request, current_states = created[0], created[1], tuple(created[2:])
            active = []
        sent = sent_argument if mode == 'resume' else none.id
        operand = _use(f'close.caller.scheduler.{i}.operand', sent)
        scheduler = b.value(f'close.caller.scheduler.{i}.raw', OperationResult(OperationID(f'close.caller.scheduler.{i}.pack'), 0), RAW)
        pack = b.add(f'close.caller.scheduler.{i}.pack', (operand,), results=(scheduler,), effects_complete=True)
        scheduler_row = RawRequestDefinition(scheduler, RawRequestKind.SEND, (operand.id,))
        raws.append(scheduler_row)
        actuals = (_use(f'close.caller.call.{i}.request', current_request),
            _use(f'close.caller.call.{i}.scheduler', scheduler, protocol=True),
            _use(f'close.caller.call.{i}.sent', sent),
            *(_use(f'close.caller.call.{i}.state.{j}', value, state=True) for j, value in enumerate(current_states)))
        region = stage['driver_region']
        selected_kind = ExitKind.RAISE if stage['host_terminal'].terminator.kind is TerminatorKind.RAISE else ExitKind.NORMAL
        edges, forwarding, landings = [], [], {}
        for number, exit_ in enumerate(region.exits):
            name = f'close.caller.call.{i}.{exit_.kind.value.lower()}'
            formals = b.formals(name, (REFERENCE, REFERENCE, *exit_.types))
            forwarded = (_use(f'close.caller.call.{i}.edge.{number}.body', current_body),
                         _use(f'close.caller.call.{i}.edge.{number}.request', current_request))
            forwarding.extend(forwarded)
            edges.append(CompletionSuccessor(exit_.kind, BlockID(name),
                (*(use.id for use in forwarded), *(CompletionResult(j) for j in range(len(exit_.types))))))
            landings[exit_.kind] = formals
            if exit_.kind is not selected_kind:
                b.block(name, formals, (b.terminal(name + '.unselected.host', formals[2:], fail=exit_.kind is ExitKind.RAISE),))
        call = b.add(f'close.caller.call.{i}', (*actuals, *forwarding), callee=region.id, effects_complete=True,
            invoke_arguments=tuple(use.id for use in actuals),
            completions=tuple(CompletionSignature(exit_.kind, exit_.types) for exit_ in region.exits),
            terminator=Terminator(TerminatorKind.INVOKE, tuple(edges)))
        b.block(current_block, current_formals, (*active, pack, call))
        calls.append(call)
        caller_choices[call.id] = next(j for j, edge in enumerate(edges) if edge.kind is selected_kind)
        current_block = f'close.caller.call.{i}.{selected_kind.value.lower()}'
        current_formals = landings[selected_kind]
        current_body, current_request = current_formals[:2]
        payload, current_states = current_formals[2], tuple(current_formals[3:])
        active = []
        if selected_kind is ExitKind.RAISE:
            caught, current_states = catch(f'close.caller.catch.{i}', payload, current_states)
            active.append(caught)
            prior_packet = payload
        stage.update(caller_call=call, scheduler_definition=scheduler_row, caller_sent=sent,
            caller_states=tuple(next(use.value for use in actuals if use.id == identity) for identity in call.invoke_arguments[3:]),
            selected_kind=selected_kind, own_results_by_kind={kind: tuple(_own(call, kind, j)
                for j in range(len(next(exit_ for exit_ in region.exits if exit_.kind is kind).types))) for kind in landings},
            caller_landings=landings, caller_payload=payload,
            current_catch=None if selected_kind is ExitKind.NORMAL else catches[-1])
    final = b.terminal('close.caller.final', (none.id, *current_states))
    b.block(current_block, current_formals, (*active, final))
    caller = b.finish('close.caller.entry')
    tables = ('driver_interfaces', 'final_body_gates', 'body_drives', 'driver_lifecycles',
        'body_outcome_commits', 'terminal_frame_releases', 'protocol_scopes', 'protocol_operations',
        'literal_evaluations', 'exception_packets', 'request_association_reads', 'session_updates',
        'protocol_bindings', 'protocol_call_inputs', 'protocol_result_consumers', 'exception_matches',
        'state_address_refinements')
    graph = replace(body, root=caller.id,
        regions=(caller, replace(body_region, parent=caller.id),
                 *(replace(stage['driver_region'], parent=caller.id) for stage in stages)),
        operations=(*body.operations, *(op for part in components for op in part['operations']), *b.operations),
        values=(*body.values, *(value for part in components for value in part['values']), *b.values),
        control_scopes=(*body.control_scopes, *(scope for part in components for scope in part['control_scopes']),
            _scope(caller.id, caller.blocks)),
        handled_context_domains=(*body.handled_context_domains, HandledContextDomain(caller.id, body.resumable_bodies[0].context_location)),
        resumable_creates=(creation,), async_request_creates=tuple(item['create'] for item in requests),
        raw_driver_requests=tuple(raws),
        **{name: (*getattr(body, name), *(row for part in components for row in part[name])) for name in tables})
    caller_path = _selected_path(graph, caller, caller_choices)
    body_paths = _body_paths(body, body_details, route, stages)
    for i, stage in enumerate(stages):
        stage['body_path'] = body_paths.get(i)
    body_details['create'] = creation
    replies = tuple(row for stage in stages for row in stage['reply_stages'])
    capture = {'caller': caller, 'caller_path': caller_path, 'body': body_region,
               'primary_creation': creation, 'drivers': tuple(stages)}
    return graph, {'route': route, 'caller': caller, 'caller_path': caller_path,
        'caller_choices': caller_choices, 'body': body_details, 'primary_creation': creation,
        'capture_details': capture, 'body_paths': body_paths, 'drivers': tuple(stages),
        'stages': tuple(stages), 'requests': tuple(requests), 'reply_stages': replies,
        'calls': tuple(calls), 'catches': tuple(catches), 'raws': tuple(raws),
        'capture_formals': capture_formals, 'initial_states': tuple(initial_states),
        'final': final, 'final_packet': prior_packet,
        'releases': tuple(row for stage in stages for row in stage['releases']),
        'conditions': ('supplied original native request/control applicability',
                       'no runtime identity, phase recovery or transformation authority')}
