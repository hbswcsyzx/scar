"""Root-only seal/check of the frozen focused gate; never runs target code."""
import hashlib
import json
from pathlib import Path
import subprocess
import sys
import xml.etree.ElementTree as ET

from scripts.run_gate import fingerprint

ROOT = Path.cwd()
HERE = Path(__file__).resolve().parent
PRIOR = HERE.parent / 'model_value_source_consumers'
CHECKPOINT = HERE / 'checkpoint.json'


def read(path):
    return json.loads(path.read_text())


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write(path, value):
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')


def cases(folder):
    rows = tuple(ET.parse(folder / 'junit.xml').iter('testcase'))
    assert rows
    identities = {(row.get('classname'), row.get('name')) for row in rows}
    assert len(rows) == len(identities)
    assert not any(row.find(tag) is not None for row in rows for tag in ('failure', 'error', 'skipped'))
    return identities


def check(report):
    current, sources = fingerprint()
    assert current == report['source_sha256']
    assert len(sources) == report['source_files']
    for path, expected in report['files'].items():
        assert digest(ROOT / path) == expected, path
    for name, original in report['protected'].items():
        path = original['path']
        head = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=path, text=True).strip()
        status = subprocess.check_output(['git', 'status', '--porcelain'], cwd=path, text=True)
        assert head == original['head'] and status == original['status'], name
    print(json.dumps({'status': report['status'], 'source_files': len(sources),
        'source_sha256': current, 'focused_cases': report['current_focused']['identities'],
        'known_case_catalog': report['current_case_catalog_size'],
        'checkpoint_sha256': digest(CHECKPOINT)}, ensure_ascii=False))


if sys.argv[1:] == ['--check']:
    check(read(CHECKPOINT))
    raise SystemExit(0)
assert not sys.argv[1:]
assert not CHECKPOINT.exists(), 'immutable checkpoint already exists; use --check'
design = read(HERE / 'design-baseline.json')
assert digest(ROOT / design['scope_plan']) == design['plan_sha256']
assert digest(PRIOR / 'checkpoint.json') == design['prior_checkpoint_sha256']
prior = read(PRIOR / 'checkpoint.json')
for path, expected in prior['files'].items():
    assert digest(ROOT / path) == expected, ('prior immutable evidence', path)
current, sources = fingerprint()
selection = read(HERE / 'focused-selection.json')
attempts, sets = {}, {}
for name, selectors in (('verified-new', selection['new_selectors']),
                        ('verified-regression', selection['selectors'])):
    folder = HERE / name
    result, command = read(folder / 'result.json'), read(folder / 'command.json')
    assert result['returncode'] == 0 and result['source_unchanged']
    assert result['source_sha256_before'] == result['source_sha256_after'] == current
    assert read(folder / 'sources.before.json') == read(folder / 'sources.after.json') == sources
    assert command['command'][4:-1] == selectors
    sets[name] = cases(folder)
    attempts[name] = {**result, 'identities': len(sets[name])}
assert len(sets['verified-new']) == 30
assert not sets['verified-new'] & sets['verified-regression']
prior_sources = read(PRIOR / 'verified-current/sources.after.json')
assert all(sources.get(path) == sha for path, sha in prior_sources.items() if path.startswith('tests/'))
prior_catalog = {tuple(row) for row in read(PRIOR / 'case-catalog.json')['identities']}
assert len(prior_catalog) == 2325
selected = sets['verified-new'] | sets['verified-regression']
catalog = prior_catalog | selected
write(HERE / 'case-catalog.json', {
    'scope': 'retained focused obligations, not the whole repository or all currently rerun',
    'prior_catalog': len(prior_catalog), 'new': len(sets['verified-new']),
    'additional_preexisting': len(sets['verified-regression'] - prior_catalog),
    'currently_retested': len(selected), 'total': len(catalog),
    'identities': sorted(catalog)})
write(HERE / 'source-changes.json', {
    'changed': sorted(path for path, sha in sources.items() if path in prior_sources and prior_sources[path] != sha),
    'added': sorted(set(sources) - set(prior_sources)),
    'removed': sorted(set(prior_sources) - set(sources)),
    'all_prior_tests_and_fixtures_unchanged': True})
attempt_history = {folder.name: read(folder / 'result.json') for folder in sorted(HERE.iterdir())
                   if folder.is_dir() and (folder / 'result.json').is_file()}
review = read(HERE / 'independent-review.json')
assert review['status'] == 'ACCEPT_CONDITIONAL_F19' and review['source_sha256'] == current
bound = [*HERE.rglob('*'), *(ROOT / path for path in (
    'docs/M1_ASYNC_CLOSE_CHAINS_PLAN.md', 'docs/M1_ASYNC_CLOSE_CHAINS_API.md',
    'docs/M1_ASYNC_CLOSE_CHAINS_PROGRESS.md', 'docs/M1_ASYNC_CLOSE_CHAINS_ACCEPTANCE.md',
    'docs/M1_AFTER_F19_ORIGINAL_SCOPE.md', 'docs/M1_YIELD_FROM_INITIAL_PLAN.md',
    'docs/MODEL.md', 'docs/ROADMAP.md',
    'docs/FIRST_STAGE_COMPLETION_MATRIX.md', 'docs/WHOLE_PROGRAM_MODEL_PLAN.md'))]
files = {str(path.relative_to(ROOT)): digest(path) for path in sorted(bound)
         if path.is_file() and path.name not in ('checkpoint.json', 'checkpoint.sha256')}
report = {
    'status': 'VERIFIED_CONDITIONAL_ASYNC_CLOSE_CHAINS',
    'base_revision': design['base_revision'], 'source_sha256': current, 'source_files': len(sources),
    'current_focused': {'identities': len(selected), 'failures': 0, 'errors': 0, 'skips': 0},
    'new_case_identities': len(sets['verified-new']),
    'prior_case_catalog_retained': len(prior_catalog),
    'prior_catalog_retested': len(selected & prior_catalog),
    'additional_preexisting_cases': len(sets['verified-regression'] - prior_catalog),
    'current_case_catalog_size': len(catalog), 'case_catalog_is_whole_repository': False,
    'prior_full_catalog_rerun': False, 'all_prior_tests_and_fixtures_unchanged': True,
    'ordinary_copy_semantics_unchanged': True, 'program_schema': 27, 'execution_schema': 8,
    'accepted_families': 10, 'open_families': 6, 'm2_denominator': [113, 196],
    'first_stage_complete': False, 'attempts': attempts, 'attempt_history': attempt_history,
    'protected': design['protected'], 'source_runtime_verified': False,
    'native_identity_verified': False, 'transformation_authorized': False,
    'new_lewm_speedup': None, 'files': files}
write(CHECKPOINT, report)
(HERE / 'checkpoint.sha256').write_text(digest(CHECKPOINT) + '\n')
check(report)
