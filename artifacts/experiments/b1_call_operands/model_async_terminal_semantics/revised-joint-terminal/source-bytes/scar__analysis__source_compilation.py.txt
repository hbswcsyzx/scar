"""Bounded diagnostics for caller-owned source; never execute the target.

Return syntax is inventoried independently of optional core-operation links.
Only CPython decides whether the complete module is legal. In particular this
module does not reconstruct generator scopes, or build a runtime graph for an
illegal program. Source fingerprints cover decoded text re-encoded as UTF-8.
"""
from __future__ import annotations

import ast
import codecs
import hashlib
import io
from pathlib import Path
import platform
import re
import sys
import tokenize

from scar.ir.execution.validity_budget import ValidityBudget
from scar.ir.frontend_v2 import source_reference
from scar.ir.program.ids import OperationID
from scar.ir.program.source import OperationSource, SourceRole
from scar.ir.semantics_v2 import SourceSnapshot
from scar.ir.v2.common import SourceReference


MAX_SOURCE_BYTES = 4 * 1024 * 1024


def _syntax_span(error, lines, *, phase):
    """Preserve compiler byte offsets; convert parser character offsets.

    The compile phase runs only after AST parsing succeeded. CPython semantic
    diagnostics then retain the AST's UTF-8 byte columns (plus one), whereas
    parser diagnostics expose one-based Unicode character offsets.
    """
    start, end = error.lineno, error.end_lineno
    left, right = error.offset, error.end_offset
    if (type(start) is not int or type(end) is not int
            or not 1 <= start <= end <= len(lines)
            or type(left) is not int or type(right) is not int
            or left < 1 or right < 1):
        return None
    first, last = lines[start - 1].rstrip("\r\n"), lines[end - 1].rstrip("\r\n")
    left, right = left - 1, right - 1
    if start == end and right < left:
        return None
    if phase == "parse":
        if left > len(first) or right > len(last):
            return None
        left = len(first[:left].encode("utf-8"))
        right = len(last[:right].encode("utf-8"))
    elif phase == "compile":
        first_bytes, last_bytes = first.encode("utf-8"), last.encode("utf-8")
        if left > len(first_bytes) or right > len(last_bytes):
            return None
        try:
            first_bytes[:left].decode("utf-8")
            last_bytes[:right].decode("utf-8")
        except UnicodeDecodeError:
            return None
    else:
        return None
    return {"start_line": start, "end_line": end,
            "start_column": left, "end_column": right}


def source_compilation(snapshot: SourceSnapshot, *, text: str | None = None,
                       operation_links: tuple[OperationSource, ...] = (),
                       source_root: str | Path | None = None,
                       budget: int = 4096) -> dict:
    """Check one pinned source using this CPython, without import or execution.

    ``source_root`` is the frontend's source namespace, not a filesystem read
    permission. Its default is the lexical parent of ``snapshot.path``. Supply
    the original project root when comparing frontend OperationSource records.
    A requested patch version is binding; a two-component version constrains
    only the major and minor components. The actual full version is reported.
    A completed Return inventory does not mean every site was diagnosed after
    the compiler stopped at its first error. No code object escapes this query.
    """
    if type(snapshot) is not SourceSnapshot:
        raise TypeError("source compilation requires an exact SourceSnapshot")
    snapshot.__post_init__()
    if text is not None and type(text) is not str:
        raise TypeError("caller-owned source must be exact text or None")
    if type(operation_links) is not tuple:
        raise TypeError("operation links must be an exact tuple")
    shared = ValidityBudget(budget)
    path = Path(snapshot.path)
    root = path.parent if source_root is None else Path(source_root)
    returns, gaps, diagnostics = [], [], []
    links = [{"link": link, "match": None, "return_source": None}
             for link in operation_links]
    result = {"snapshot": snapshot, "source_root": str(root),
              "implementation": sys.implementation.name,
              "python_version": platform.python_version(),
              "requested_python_version": snapshot.python_version,
              "fingerprint_match": None, "encoding_match": None,
              "version_match": None, "compiled": None,
              "compiler_attempted": False, "status": "PARTIAL",
              "returns": (), "operation_links": (), "diagnostics": (),
              "inventory": {"returns_total": None, "enumerated_returns": 0,
                            "known_inventory_complete": False,
                            "diagnostics_complete": False, "checked_returns": 0},
              "gaps": (), "cursor": None, "budget": {},
              "target_executed": False, "source_runtime_verified": False,
              "runtime_identity_verified": False,
              "native_applicability_verified": False,
              "transformation_authorized": False}

    def gap(code, message, **support):
        gaps.append({"code": code, "message": message, **support})

    def spend(kind, cost=1, **cursor):
        if shared.spend(kind, cost, "source_compilation." + kind):
            return True
        result["cursor"] = {"phase": kind, **cursor}
        gap("source_compilation_budget_exhausted", "source query work is incomplete", **cursor)
        return False

    def finish():
        result["returns"] = tuple({**row, "operation_links": tuple(row["operation_links"])}
                                  for row in returns)
        result["operation_links"] = tuple(links)
        result["diagnostics"] = tuple(diagnostics)
        result["gaps"] = tuple(gaps)
        result["inventory"]["enumerated_returns"] = len(returns)
        result["budget"] = {"limit": shared.limit, "used": shared.used,
                            "truncated": shared.truncated, "costs": dict(shared.costs)}
        if result["compiled"] is False:
            result["status"] = "REJECTED"
        elif result["compiled"] is True and not gaps:
            result["status"] = "ACCEPTED"
        return result

    def diagnostic(error, phase):
        # Python source lines end at CR/LF, not at every Unicode separator
        # recognized by str.splitlines (which may occur inside a literal).
        span = _syntax_span(error, re.split(r"\r\n|\r|\n", text), phase=phase)
        if span is None:
            gap("source_compilation_diagnostic_span_unresolved",
                "compiler diagnostic has no valid complete source span")
        matching = tuple(row["source"] for row in returns if span is not None
                         and all(getattr(row["source"], key) == value
                                 for key, value in span.items()))
        diagnostics.append({"kind": type(error).__name__, "message": error.msg,
                            "phase": phase, "filename": error.filename,
                            "line": error.lineno, "offset": error.offset,
                            "end_line": error.end_lineno, "end_offset": error.end_offset,
                            "span": span, "return_sources": matching})
        result["compiled"] = False
        gap("source_compilation_first_error_only",
            "compiler rejection does not diagnose all remaining source sites")

    if text is None:
        gap("source_compilation_text_missing", "caller did not supply source text")
        return finish()
    # Check the character bound before allocating an encoded copy. A UTF-8
    # code point uses at least one byte, so this cannot exclude a valid input.
    if len(text) > MAX_SOURCE_BYTES:
        gap("source_compilation_input_limit", "source exceeds the 4 MiB compile ceiling")
        return finish()
    if not spend("source_fingerprint", max(1, (len(text) + 1023) // 1024)):
        return finish()
    try:
        utf8 = text.encode("utf-8")
    except UnicodeError as error:
        gap("source_compilation_encoding_mismatch", str(error))
        return finish()
    if len(utf8) > MAX_SOURCE_BYTES:
        gap("source_compilation_input_limit", "UTF-8 source exceeds the 4 MiB compile ceiling")
        return finish()
    result["fingerprint_match"] = snapshot.fingerprint == "sha256:" + hashlib.sha256(utf8).hexdigest()
    if not result["fingerprint_match"]:
        gap("source_compilation_fingerprint_mismatch", "supplied text differs from the pinned snapshot")
        return finish()
    try:
        codec = codecs.lookup(snapshot.encoding)
        raw = text.encode(snapshot.encoding)
        detected, _ = tokenize.detect_encoding(io.BytesIO(raw).readline)
        result["encoding_match"] = (codecs.lookup(detected).name == codec.name
                                    and raw.decode(snapshot.encoding) == text)
    except (LookupError, UnicodeError, SyntaxError) as error:
        result["encoding_match"] = False
        gap("source_compilation_encoding_mismatch", str(error))
        return finish()
    if not result["encoding_match"]:
        gap("source_compilation_encoding_mismatch", "source encoding declaration differs from the snapshot")
        return finish()
    requested = snapshot.python_version
    if type(requested) is not str or re.fullmatch(r"\d+\.\d+(?:\.\d+)?", requested) is None:
        gap("source_compilation_version_unresolved", "snapshot needs an explicit Python major.minor version")
        return finish()
    requested_parts = tuple(map(int, requested.split(".")))
    result["version_match"] = requested_parts == sys.version_info[:len(requested_parts)]
    if not result["version_match"] or sys.implementation.name != "cpython":
        gap("source_compilation_version_mismatch", "this CPython does not match the requested snapshot version")
        return finish()
    try:
        path.relative_to(root)
    except ValueError:
        gap("source_compilation_namespace_mismatch", "source path is outside the declared frontend namespace")
        return finish()
    work = max(1, (len(utf8) + 1023) // 1024)
    if not spend("source_ast_parse", work):
        return finish()
    try:
        tree = ast.parse(text, filename=snapshot.path, type_comments=True)
    except SyntaxError as error:
        diagnostic(error, "parse")
        return finish()
    except (ValueError, RecursionError, MemoryError, OverflowError) as error:
        gap("source_compilation_parser_unresolved", str(error))
        return finish()
    for index, node in enumerate(ast.walk(tree)):
        if not spend("source_ast_nodes", index=index):
            return finish()
        if isinstance(node, ast.Return):
            returns.append({"source": source_reference(path, root, snapshot.fingerprint, index, node),
                            "expression_present": node.value is not None,
                            "expression_kind": None if node.value is None else type(node.value).__name__,
                            "compiler_checked": None, "operation_links": []})
    result["inventory"].update(returns_total=len(returns), known_inventory_complete=True)
    by_source = {row["source"]: row for row in returns}
    for index, item in enumerate(links):
        if not spend("source_operation_links", index=index):
            return finish()
        link = item["link"]
        if (type(link) is not OperationSource or type(link.operation) is not OperationID
                or type(link.source) is not SourceReference or type(link.role) is not SourceRole):
            item["match"] = False
            gap("source_compilation_operation_link_invalid", "operation link has incompatible canonical types", index=index)
            continue
        if link.role is not SourceRole.RETURN:
            continue
        row = by_source.get(link.source)
        item["match"] = row is not None
        if row is None:
            gap("source_compilation_operation_link_mismatch",
                "return link must match the complete original atom and span", index=index,
                operation=link.operation)
        else:
            item["return_source"] = row["source"]
            row["operation_links"].append(link)
    if not spend("source_compile", work):
        return finish()
    result["compiler_attempted"] = True
    try:
        # Do not expose or execute this code object; ignore inherited future
        # flags from this analysis module and preserve the original filename.
        compile(text, snapshot.path, "exec", dont_inherit=True, optimize=0)
    except SyntaxError as error:
        diagnostic(error, "compile")
    except (ValueError, RecursionError, MemoryError, OverflowError) as error:
        gap("source_compilation_compiler_unresolved", str(error))
    else:
        result["compiled"] = True
        result["inventory"].update(diagnostics_complete=True, checked_returns=len(returns))
        for row in returns:
            row["compiler_checked"] = True
    return finish()


__all__ = ["MAX_SOURCE_BYTES", "source_compilation"]
