_BUDGET = 1048576
_FLAGS = ('source_runtime_verified', 'runtime_identity_verified',
          'value_identity_verified', 'transformation_authorized')


def _index(program, model):
    require_valid(program)
    validation = execution_validate(model, program)
    assert validation.valid, validation.issues
    return ExecutionIndex(model, program)


def _use(model, operation, use):
    rows = tuple(row for row in model.uses if (row.operation, row.use) == (operation, use))
    assert len(rows) == 1
    return rows[0]


def _value(model, value):
    return next(row for row in model.values if row.id == value)


def _own(program, model, operation, port):
    definition = next(row for row in program.operations if row.id == port.edge.operation)
    transfer, = tuple(row for row in model.transfers if row.operation == operation)
    assert transfer.successor_index == port.edge.successor_index
    edge = definition.terminator.successors[transfer.successor_index]
    landing = next(block for region in program.regions for block in region.blocks if block.id == edge.block)
    formal, = tuple(formal for actual, formal in zip(edge.arguments, landing.arguments)
                    if type(actual) is CompletionResult and actual.index == port.result.index)
    value, = tuple(row for row in model.values if row.origin == transfer.target and row.value == formal)
    return value


def _current(index, program, model, occurrence):
    actual = next(row for row in model.operations if row.id == occurrence)
    operation = next(row for row in program.operations if row.id == actual.operation)
    definitions = {row.id: row for row in program.values}
    current = index.state_at(OccurrencePoint(occurrence, OccurrencePhase.BEFORE))
    assert current is not None
    uses = tuple(use for use in operation.uses if type(definitions[use.value]) is StateVersionValue)
    assert {definitions[use.value].location for use in uses} == {row.id for row in program.locations}
    for use in uses:
        assert current[definitions[use.value].location] == _use(model, occurrence, use.id).value
    return current


def _selected_controls(index, program, model, details, example):
    definitions = {row.id: row for row in program.operations}
    values = {row.id: row for row in program.values}
    for item in example['control_predicates']:
        row, occurrence = item['record'], item['occurrence']
        predicate = definitions[row.operation]
        result, = tuple(value for value in model.values
                        if value.origin == occurrence and type(values[value.value].origin) is OperationResult
                        and values[value.value].origin.index == row.contract.result.index)
        branch = definitions[row.contract.branch]
        assert _use(model, item['branch'], branch.terminator.condition).value == result.id
        transfer, = tuple(transfer for transfer in model.transfers if transfer.operation == item['branch'])
        assert transfer == item['transfer']
        assert transfer.successor_index == details['drivers'][item['driver_ordinal']]['path_choices'][branch.id]
        if predicate.accesses:
            current = index.state_at(OccurrencePoint(occurrence, OccurrencePhase.BEFORE))
            for access in predicate.accesses:
                assert _use(model, occurrence, access.before).value == current[access.location]
        subject = row.contract.subject
        if type(subject) is BodyStateCondition:
            body = _value(model, _use(model, occurrence, subject.instance_use).value)
            route = index.formal_route(body.id)
            assert route.transfer is not None and route.source_use is not None
        if type(subject) is ReferenceIsNoneCondition:
            driver = details['drivers'][item['driver_ordinal']]
            call = index.call_activation(example['call_activations'][item['driver_ordinal']].id)
            input_, = tuple(input_ for input_ in call['inputs']
                            if input_['formal_definition'].id == driver['initial_controls']['operand'].id)
            assert _use(model, occurrence, subject.input_use).value == input_['formal'].id
            assert input_['actual_value'] is not None
            assert input_['actual_definition'].value == driver['effective_operand'].id


def _return_adapter(index, program, model, driver, example, ordinal, drive_view):
    """The fourth source already has a real constructor and both own exits."""
    definitions = {row.id: row for row in program.operations}
    values = {row.id: row for row in program.values}
    scope = example['driver_scopes'][ordinal][2]
    rows = tuple(row for row in program.protocol_operations if type(row.contract) is OutcomeAdapter
        and type(row.contract.input) is DriveResult and row.contract.input.outcome == driver['outcome'])
    assert len(rows) == 2 and {row.slot for row in rows} == {0, 1}
    assert all(row.contract.rule is OutcomeAdapterRule.ASYNCGEN_RETURN_TO_STOP_ASYNC_ITERATION for row in rows)
    assert len({row.operation for row in rows}) == 1
    constructor = definitions[rows[0].operation]
    assert {signature.kind for signature in constructor.completions} == {ExitKind.NORMAL, ExitKind.RAISE}
    assert len(constructor.completions) == 2
    for row in rows:
        port = row.contract.output
        edge = constructor.terminator.successors[port.edge.successor_index]
        assert row.slot == port.edge.successor_index and port.result.index == 0
        assert edge.kind is (ExitKind.NORMAL if row.slot == 0 else ExitKind.RAISE)
        assert tuple(argument.index for argument in edge.arguments) == tuple(range(1 + len(program.locations)))
        landing = next(block for region in program.regions for block in region.blocks if block.id == edge.block)
        terminal = definitions[landing.operations[-1]]
        assert terminal.terminator.kind is TerminatorKind.RAISE
        assert {values[value].location for value in landing.arguments
                if type(values[value]) is StateVersionValue} == {location.id for location in program.locations}
    transfer, = tuple(row for row in model.transfers if row.operation == scope[constructor.id])
    selected, = tuple(row for row in rows if row.slot == transfer.successor_index)
    assert transfer.successor_index == 0
    actual_input = _use(model, scope[constructor.id], selected.contract.input_use)
    payload, = tuple(row for row in drive_view['outputs'] if type(row['formal_definition']) is not StateVersionValue)
    assert payload['formal'].id == actual_input.value
    outcome = drive_view['outcome']
    body = definitions[outcome.operation]
    assert body.terminator.kind is TerminatorKind.RETURN
    assert payload['source_value'].id == _use(model, outcome.id, body.terminator.values[0]).value
    produced = _own(program, model, scope[constructor.id], selected.contract.output)
    packet, = tuple(row for row in program.exception_packets
        if type(values[row.value].origin) is OperationResult and values[row.value].origin.operation in scope
        and next(use.value for use in definitions[values[row.value].origin.operation].uses
                 if use.id == row.exception) == produced.value)
    packet_operation = values[packet.value].origin.operation
    assert _use(model, scope[packet_operation], packet.exception).value == produced.id
    packed, = tuple(row for row in model.values if row.origin == scope[packet_operation] and row.value == packet.value)
    assert _use(model, scope[driver['host_terminal'].id], driver['host_terminal'].terminator.values[0]).value == packed.id
    _current(index, program, model, scope[constructor.id])
    for slot, identity in enumerate(driver['host_terminal'].terminator.values[1:], 1):
        own = _own(program, model, scope[constructor.id], replace(selected.contract.output,
            result=CompletionResult(slot)))
        assert _use(model, scope[driver['host_terminal'].id], identity).value == own.id
    return constructor, selected


def _complete(index, program, model, details, number):
    static, example = details['static'], details['instances'][number]
    expected_ordinals = (1, 2, 3) if static['scenario'] == 'created-reject' else tuple(range(len(static['drivers'])))
    assert example['active_driver_ordinals'] == expected_ordinals
    assert len(example['body_drives']) == len(example['body_segments']) == len(expected_ordinals)
    assert {drive.activation for drive in example['body_drives']} == {example['body_activation']}
    assert example['body_origin'].creation == example['creation']
    assert index.body_activation_origin(example['body_activation'], budget=_BUDGET)['association'] == example['body_origin']
    assert len(example['request_creations']) == len(static['requests'])
    assert len(set(example['request_creations'])) == len(static['requests'])
    _selected_controls(index, program, model, static, example)
    for ordinal, association in enumerate(example['call_activations']):
        driver = static['drivers'][ordinal]
        view = index.call_activation(association.id)
        assert view['association'] == association
        assert view['caller'].id == example['caller_operations'][driver['caller_call'].id]
        assert view['activation'].id == example['driver_scopes'][ordinal][0]
        assert view['completion'].id == example['driver_scopes'][ordinal][2][driver['host_terminal'].id]
        assert len(view['inputs']) == len(driver['entry_formals'])
        assert all(row['actual'] is not None and row['actual_value'] is not None and row['formal'] is not None
                   for row in view['inputs'])
        assert len(view['outputs']) == 1 + len(program.locations)
        assert {row['ordinal'] for row in view['outputs']} == set(range(1 + len(program.locations)))
        assert all(row['edge_argument'] == CompletionResult(row['ordinal']) and
                   row['callee_use'] is not None and row['caller_formal'] is not None for row in view['outputs'])
        _current(index, program, model, association.caller)
        _current(index, program, model, association.completion)
        assert all(view[key] is False for key in _FLAGS)
    for body_ordinal, driver_ordinal in enumerate(expected_ordinals):
        association = example['body_drives'][body_ordinal]
        driver = static['drivers'][driver_ordinal]
        view = index.drive_execution(association.drive, budget=_BUDGET)
        assert view['association'] == association
        assert association.entry == example['body_entries'][body_ordinal]
        assert association.outcome == example['body_occurrences'][body_ordinal][driver['outcome_source']]
        assert view['entry_bridge'] is True and view['path_complete'] is True
        assert all(row['source_value'] is not None and row['formal'] is not None
                   for row in (*view['inputs'], *view['outputs']))
        session = view['session']
        assert session['source'] == driver['outcome']
        expected_post = SessionPost.HOLD if body_ordinal < 2 else SessionPost.COMPLETE
        assert session['expected_post'] is expected_post
        target, = session['targets']
        assert target['update'] == driver['update'] and target['post_established'] is True
        assert target['state_bindings']
        assert all(row['use'] is not None and row['value'] is not None and
                   row['current'] == row['value'].id for row in target['state_bindings'])
        if static['scenario'] == 'suspended-new' and body_ordinal == 3:
            _return_adapter(index, program, model, driver, example, driver_ordinal, view)
        host, = session['hosts']
        assert host['operation'].id == example['call_activations'][driver_ordinal].completion
        assert host['definition'] == driver['host_terminal']
        segment = next(row for row in model.segments if row.id == example['body_segments'][body_ordinal])
        if body_ordinal == 0:
            assert type(segment.ingress) is InitialEntry and type(driver['drive'].target) is BodyStart
        else:
            assert type(segment.ingress) is ResumeIngress and type(driver['drive'].target) is BodyResume
            assert segment.ingress.suspension == example['body_drives'][body_ordinal - 1].outcome
            transfer, = tuple(row for row in model.transfers if row.operation == segment.ingress.suspension)
            assert transfer.target == association.entry
            assert transfer.successor_index == driver['drive'].target.successor
        assert all(view[key] is False for key in _FLAGS)
    if static['scenario'] == 'created-reject':
        rejected = static['drivers'][0]
        scope = example['driver_scopes'][0][2]
        assert rejected['drive'].operation not in scope
        assert not any(row.drive in set(scope.values()) for row in model.body_drive_executions)
        assert not any(row.operation in {entry.operation for entry in program.frame_context_attaches}
                       for row in model.operations if row.id in set(scope.values()))
        helper = static['initial_control']['reject_helper']
        selected = next(row for row in static['initial_control']['processing']
            if row.operation == helper and row.contract.result.edge.successor_index == int(static['reject_raised']))
        own = _own(program, model, scope[helper], selected.contract.result)
        terminal = rejected['host_terminal']
        assert _use(model, scope[terminal.id], terminal.terminator.values[0]).value == own.id
        helper_definition = next(row for row in program.operations if row.id == helper)
        assert helper_definition.terminator.successors[int(static['reject_raised'])].kind is (
            ExitKind.RAISE if static['reject_raised'] else ExitKind.NORMAL)
        assert terminal.terminator.kind is TerminatorKind.RAISE
        before = _current(index, program, model, scope[helper])
        update = rejected['reject_complete']
        assert update.post is SessionPost.COMPLETE
        assert not any(type(row) is AsyncSessionOutcome and row.update == update.operation
                       for row in program.protocol_bindings)
        complete = next(row for row in model.operations if row.id == scope[update.operation])
        actual_helper = next(row for row in model.operations if row.id == scope[helper])
        assert complete.visit == actual_helper.visit
        visit = next(row for row in model.visits if row.id == complete.visit)
        block = next(block for region in program.regions for block in region.blocks if block.id == visit.block)
        assert block.operations.index(update.operation) + 1 == block.operations.index(helper)
        after = index.state_at(OccurrencePoint(complete.id, OccurrencePhase.AFTER))
        assert before == after
        current = _current(index, program, model, scope[terminal.id])
        definitions = {row.id: row for row in program.values}
        for slot, identity in enumerate(terminal.terminator.values[1:], 1):
            own_state = _own(program, model, scope[helper], replace(selected.contract.result,
                result=CompletionResult(slot)))
            use = _use(model, scope[terminal.id], identity)
            assert use.value == own_state.id
            assert current[definitions[own_state.value].location] == own_state.id
        first = index.call_activation(example['call_activations'][0].id)
        assert first['outputs'][0]['callee_use'].value == own.id
        # A fresh request reads the same independent B after this actual
        # rejection. The failed R is not restarted or converted into B's F.
        fresh = static['requests'][1]
        fresh_occurrence = example['caller_operations'][fresh['operation'].id]
        fresh_current = _current(index, program, model, fresh_occurrence)
        assert fresh_current == {row['caller_formal_definition'].location: row['caller_formal'].id
            for row in first['outputs'] if type(row['caller_formal_definition']) is StateVersionValue}
        assert example['body_drives'][0].entry == example['body_entries'][0]
    if static['scenario'] == 'suspended-new':
        last = static['drivers'][-1]
        assert last['drive'].target.suspend == static['body_definition']['async_yield'].id
        assert last['request']['operand'].origin.value == 61
        assert len(example['body_segments']) == 4
        assert example['request_creations'][1] != example['request_creations'][0]
        assert static['drivers'][-2]['update'].post is SessionPost.COMPLETE
        target = example['body_drives'][-1].outcome
        definition = next(row for row in model.operations if row.id == target)
        terminal = next(row for row in program.operations if row.id == definition.operation)
        assert terminal.terminator.kind is TerminatorKind.RETURN
        payload = _value(model, _use(model, target, terminal.terminator.values[0]).value)
        assert next(row for row in program.values if row.id == payload.value).origin.value is None
    return example


@pytest.mark.parametrize('scenario,reject_raised', (
    ('created-none', False), ('created-reject', False),
    ('created-reject', True), ('suspended-new', False)),
    ids=('created-none-start', 'created-reject-own-normal',
         'created-reject-own-raise', 'yielded-new-nonNone-resume'))
def test_actual_initial_control_has_exact_request_body_current_states_and_continuation(scenario, reject_raised):
    program, model, details = async_initial_control_execution(scenario=scenario,
        reject_raised=reject_raised, instances=2)
    index = _index(program, model)
    assert len(model.body_activation_origins) == 2
    assert len({example['creation'] for example in details['instances']}) == 2
    assert len({example['body_activation'] for example in details['instances']}) == 2
    for number in range(2):
        _complete(index, program, model, details, number)


@pytest.mark.parametrize('missing', ('call', 'operand', 'initial-transfer'))
def test_missing_current_initial_input_or_transfer_cannot_borrow_complete_sibling(missing):
    keywords = {'call': 'missing_call', 'operand': 'missing_operand',
                'initial-transfer': 'missing_initial_transfer'}
    program, model, details = async_initial_control_execution(**{keywords[missing]: 0})
    index = _index(program, model)
    static, example = details['static'], details['instances'][0]
    stage = static['drivers'][0]
    association = example['call_activations'][0]
    if missing == 'call':
        view = index.call_entries(association.caller)
        assert view['status'] == 'MISSING' and view['entries'] == ()
        assert any(gap.code == 'call_activation_missing' for gap in view['gaps'])
    elif missing == 'operand':
        view = index.call_activation(association.id)
        operand, = tuple(row for row in view['inputs']
            if row['formal_definition'].id == stage['initial_controls']['operand'].id)
        assert operand['actual'] is None and operand['actual_value'] is None
        assert operand['formal'] is not None
        assert any(gap.code == 'call_actual_unobserved' for gap in view['gaps'])
    else:
        phase = stage['initial_controls']['phase']
        row = next(row for row in static['initial_control']['predicates'] if row.operation == phase['predicate'])
        occurrence = example['driver_scopes'][0][2][row.operation]
        owner = _use(model, occurrence, row.contract.subject.instance_use)
        route = index.formal_route(owner.value)
        assert route.transfer is None and route.source_use is None and route.argument is None
        branch = example['driver_scopes'][0][2][stage['predicate']['branch']]
        assert not any(transfer.operation == branch for transfer in model.transfers)
    _complete(index, program, model, details, 1)


def test_budget_one_retains_each_actual_drive_port_and_session_target_without_true_prefix():
    program, model, details = async_initial_control_execution(scenario='created-reject')
    index = _index(program, model)
    for example in details['instances']:
        for association in example['body_drives']:
            full = index.drive_execution(association.drive, budget=_BUDGET)
            limited = index.drive_execution(association.drive, budget=1)
            assert limited['association'] == association
            assert limited['input_count'] == full['input_count']
            assert limited['output_count'] == full['output_count']
            assert limited['session']['target_count'] == full['session']['target_count']
            assert limited['session']['source'] == full['session']['source']
            assert limited['budget']['truncated'] and limited['budget']['used'] == 1
            assert limited['cursor'] is not None and limited['status'] == 'PARTIAL'
            assert any(row['check_status'] == 'NOT_CHECKED' and row['compatibility'] is None
                       for row in (*limited['inputs'], *limited['outputs']))
            assert all(limited[key] is False for key in _FLAGS)


def test_missing_fourth_return_adapter_input_cannot_borrow_earlier_item_or_sibling():
    program, model, details = async_initial_control_execution(scenario='suspended-new', instances=2)
    original = _index(program, model)
    static, example = details['static'], details['instances'][0]
    driver = static['drivers'][-1]
    association = example['body_drives'][-1]
    view = original.drive_execution(association.drive, budget=_BUDGET)
    constructor, selected = _return_adapter(original, program, model, driver, example, 3, view)
    occurrence = example['driver_scopes'][3][2][constructor.id]
    changed = replace(model, uses=tuple(row for row in model.uses
        if (row.operation, row.use) != (occurrence, selected.contract.input_use)))
    index = _index(program, changed)
    actual = index.drive_execution(association.drive, budget=_BUDGET)
    assert actual['association'] == association
    assert actual['session']['source'] == driver['outcome']
    assert actual['session']['target_count'] == 1
    assert actual['session']['status'] == 'PARTIAL'
    host, = actual['session']['hosts']
    assert host['operation'].id == example['call_activations'][3].completion
    assert host['adapter']['declaration'] == selected
    assert host['adapter']['occurrence'].id == occurrence
    assert host['adapter']['input'] is None
    assert host['bindings'][0]['source_value'] == view['outputs'][0]['source_value']
    assert host['bindings'][0]['target_value'] is None
    assert host['bindings'][0]['compatibility'] is None
    assert host['bindings'][0]['copy_status'] == 'UNPROVED'
    assert any(gap.code == 'body_session_copy_unproved' for gap in actual['session']['gaps'])
    assert host['adapter']['output'] is not None and host['target_value'] is not None
    assert all(actual[key] is False for key in _FLAGS)
    # Both earlier item constructor and the other caller remain fully captured;
    # neither fills this operation's missing input Use.
    previous = index.drive_execution(example['body_drives'][2].drive, budget=_BUDGET)
    previous_host, = previous['session']['hosts']
    assert previous_host['adapter']['input'] is not None
    assert previous_host['adapter']['occurrence'].id != occurrence
    _complete(index, program, changed, details, 1)
