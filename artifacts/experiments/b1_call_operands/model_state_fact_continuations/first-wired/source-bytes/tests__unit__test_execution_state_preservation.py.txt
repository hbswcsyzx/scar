QUERY_BUDGET = 65536
_FLAGS = ('source_runtime_verified', 'runtime_identity_verified',
          'value_identity_verified', 'transformation_authorized')


def _formals(builder, name, types):
    identities = []
    for ordinal, type_ in enumerate(types):
        origin = BlockArgument(BlockID(name), ordinal)
        if getattr(type_, 'kind', None) is ValueKind.DATA:
            value = DataValue(ValueID(f'{name}.arg.{ordinal}'), builder.region, origin,
                              type_.data_type)
            builder.values.append(value)
            identities.append(value.id)
        else:
            identities.append(builder.value(f'{name}.arg.{ordinal}', origin, type_))
    return tuple(identities)


def _uses(builder, name, identities):
    values = {value.id: value for value in builder.values}
    return tuple(Use(UseID(f'{name}.{ordinal}'), identity,
        UseRole.STATE_CONTENT if type(values[identity]) is StateVersionValue else
        UseRole.PROTOCOL if type(values[identity]) is ProtocolValue else
        UseRole.DATA if type(values[identity]) is DataValue else UseRole.REFERENCE)
        for ordinal, identity in enumerate(identities))


def _terminal(builder, name, kind, identities):
    uses = _uses(builder, name + '.use', identities)
    return builder.add(name, uses, effects_complete=True,
        terminator=Terminator(TerminatorKind.RETURN if kind is ExitKind.NORMAL
            else TerminatorKind.RAISE, exit='normal' if kind is ExitKind.NORMAL else 'raise',
            values=tuple(use.id for use in uses)))


def _preservation_program(*, outer_complete=True, child_complete=True,
                          write_location=None, write_mode=AccessMode.WRITE):
    """Two original domains and a complete getter with independently typed N/R.

    The source writes before the queried AFTER point. The outer INVOKE has
    explicit complete READ address declarations for every state actual; its
    callee still has real READ, branch, packet, and completion operations.
    A different LocationID is never offered as an isolation proof.
    """
    world, context = LocationID('preserve.world'), LocationID('preserve.context')
    domains = (world, context)
    states = tuple(ValueType(ValueKind.STATE, location=location) for location in domains)
    boolean = ValueType(ValueKind.DATA, data_type='builtins.bool')
    parent, child = _Builder('preserve.parent', domains), _Builder('preserve.getter', domains)
    entry_types = (REFERENCE, boolean, *states)
    parent_entry = _formals(parent, 'preserve.parent.entry', entry_types)
    child_entry = _formals(child, 'preserve.getter.entry', entry_types)

    source_id = OperationID('preserve.source.post')
    source_uses = _uses(parent, source_id.value + '.use', parent_entry[2:])
    post_states = tuple(parent.value(source_id.value + f'.state.{slot}',
        OperationResult(source_id, slot), type_) for slot, type_ in enumerate(states))
    source = Operation(source_id, 'establish the source post before the interval',
        (Behavior.READ, Behavior.WRITE), source_uses, post_states,
        tuple(StateAccess(location, use.id, AccessMode.READ_WRITE, after,
                          Footprint(FootprintKind.WHOLE))
              for location, use, after in zip(domains, source_uses, post_states)),
        effects_complete=True)
    parent.operations.append(source)

    getter_uses = _uses(child, 'preserve.getter.read.use', child_entry[2:])
    getter = Operation(OperationID('preserve.getter.read'), 'complete original getter READ',
        (Behavior.READ,), getter_uses,
        accesses=tuple(StateAccess(location, use.id, AccessMode.READ)
                       for location, use in zip(domains, getter_uses)),
        effects_complete=child_complete)
    child.operations.append(getter)
    active_states, child_prefix = list(child_entry[2:]), [getter]
    writer = None
    if write_location is not None:
        slot = {'same': 0, 'different': 1}[write_location]
        identity = OperationID('preserve.getter.write')
        use, = _uses(child, identity.value + '.before', (active_states[slot],))
        after = child.value(identity.value + '.after', OperationResult(identity, 0), states[slot])
        writer = Operation(identity, 'unexcluded original write',
            (Behavior.WRITE,) if write_mode is AccessMode.WRITE else (Behavior.READ, Behavior.WRITE),
            (use,), (after,), (StateAccess(domains[slot], use.id, write_mode, after,
                Footprint(FootprintKind.WHOLE)),), effects_complete=True)
        child.operations.append(writer)
        child_prefix.append(writer)
        active_states[slot] = after

    n = _formals(child, 'preserve.getter.normal', (REFERENCE, *states))
    r = _formals(child, 'preserve.getter.raise', (REFERENCE, *states))
    condition, = _uses(child, 'preserve.getter.condition', (child_entry[1],))
    normal_forwards = _uses(child, 'preserve.getter.normal.forward', (child_entry[0], *active_states))
    raise_forwards = _uses(child, 'preserve.getter.raise.forward', (child_entry[0], *active_states))
    branch = child.add('preserve.getter.branch', (condition, *normal_forwards, *raise_forwards),
        effects_complete=True, terminator=Terminator(TerminatorKind.BRANCH, (
            Successor(BlockID('preserve.getter.normal'), tuple(use.id for use in normal_forwards)),
            Successor(BlockID('preserve.getter.raise'), tuple(use.id for use in raise_forwards))),
            condition=condition.id))
    child.block('preserve.getter.entry', child_entry, (*child_prefix, branch))
    normal_terminal = _terminal(child, 'preserve.getter.return', ExitKind.NORMAL, n)
    child.block('preserve.getter.normal', n, (normal_terminal,))
    traceback_literal = ReferenceValue(ValueID('preserve.getter.traceback'), child.region,
                                       LiteralOrigin(None), targets=())
    child.values.append(traceback_literal)
    traceback = traceback_literal.id
    packet_id = OperationID('preserve.getter.packet')
    packet_uses = _uses(child, packet_id.value + '.use', (r[0], traceback))
    packet = child.value(packet_id.value + '.value', OperationResult(packet_id, 0), PACKET)
    pack = child.add(packet_id.value, packet_uses, results=(packet,), effects_complete=True)
    # The RAISE signature deliberately has context, packet, world ordinals.
    raised_terminal = _terminal(child, 'preserve.getter.raise.exit', ExitKind.RAISE,
                                 (r[2], packet, r[1]))
    child.block('preserve.getter.raise', r, (pack, raised_terminal))
    normal_types, raise_types = (REFERENCE, *states), (states[1], PACKET, states[0])
    exits = (ExitPort('normal', ExitKind.NORMAL, normal_types),
             ExitPort('raise', ExitKind.RAISE, raise_types))
    child_region = Region(child.region, 'complete supplied getter',
        BlockID('preserve.getter.entry'), tuple(child.blocks), exits, parent=parent.region)

    normal = _formals(parent, 'preserve.parent.normal', (REFERENCE, states[1], REFERENCE, states[0]))
    raised = _formals(parent, 'preserve.parent.raise', (REFERENCE, states[0], PACKET, states[1]))
    actuals = _uses(parent, 'preserve.call.argument', (*parent_entry[:2], *post_states))
    forward_normal, = _uses(parent, 'preserve.call.normal.forward', (parent_entry[0],))
    forward_raise, = _uses(parent, 'preserve.call.raise.forward', (parent_entry[0],))
    call = parent.add('preserve.call', (*actuals, forward_normal, forward_raise),
        effects_complete=outer_complete, callee=child.region,
        invoke_arguments=tuple(use.id for use in actuals),
        completions=(CompletionSignature(ExitKind.NORMAL, normal_types),
                     CompletionSignature(ExitKind.RAISE, raise_types)),
        terminator=Terminator(TerminatorKind.INVOKE, (
            CompletionSuccessor(ExitKind.RAISE, BlockID('preserve.parent.raise'),
                (forward_raise.id, CompletionResult(2), CompletionResult(1), CompletionResult(0))),
            CompletionSuccessor(ExitKind.NORMAL, BlockID('preserve.parent.normal'),
                (forward_normal.id, CompletionResult(2), CompletionResult(0), CompletionResult(1))))))
    parent.block('preserve.parent.entry', parent_entry, (source, call))
    targets, terminals = {}, {}
    for kind, name, state_values, outputs in (
        (ExitKind.NORMAL, 'preserve.parent.normal', (normal[3], normal[1]), (normal[2], normal[3], normal[1])),
        (ExitKind.RAISE, 'preserve.parent.raise', (raised[1], raised[3]), (raised[3], raised[2], raised[1])),
    ):
        uses = _uses(parent, name + '.read.use', state_values)
        read = Operation(OperationID(name + '.read'), 'this actual next state READ',
            (Behavior.READ,), uses, accesses=tuple(StateAccess(location, use.id, AccessMode.READ)
                for location, use in zip(domains, uses)), effects_complete=True)
        parent.operations.append(read)
        terminal = _terminal(parent, name + '.exit', kind, outputs)
        parent.block(name, normal if kind is ExitKind.NORMAL else raised, (read, terminal))
        targets[kind], terminals[kind] = read, terminal
    parent_region = Region(parent.region, 'supplied caller with original N/R',
        BlockID('preserve.parent.entry'), tuple(parent.blocks), exits)
    written_slot = None if write_location is None else {'same': 0, 'different': 1}[write_location]
    refinements = tuple(StateAddressRefinement(InvokeStateOperand(call.id, use.id),
        (AddressEffect(DomainOwner(), (WholeOwner(),),
            AccessMode.READ_WRITE if slot == written_slot else AccessMode.READ),),
        AddressCoverage.DECLARED_COMPLETE) for slot, use in enumerate(actuals[2:]))
    program = ProgramGraph(parent.region, (parent_region, child_region),
        (*parent.operations, *child.operations), (*parent.values, *child.values),
        tuple(StateLocation(location, 'original shared coherence carrier') for location in domains),
        control_scopes=(_scope(parent.region, parent.blocks), _scope(child.region, child.blocks)),
        exception_packets=(ExceptionPacketDefinition(packet, packet_uses[0].id, packet_uses[1].id),),
        state_address_refinements=refinements)
    return program, {'world': world, 'context': context, 'locations': domains,
        'source': source, 'call': call, 'getter': getter, 'writer': writer,
        'branch': branch, 'pack': pack, 'targets': targets, 'terminals': terminals,
        'child_terminals': {ExitKind.NORMAL: normal_terminal, ExitKind.RAISE: raised_terminal}}


def _preservation_execution(*, raised=False, **options):
    program, original = _preservation_program(**options)
    require_valid(program)
    kind = ExitKind.RAISE if raised else ExitKind.NORMAL
    label = 'raise' if raised else 'normal'
    caller_path = ((BlockID('preserve.parent.entry'), 0 if raised else 1),
                   (BlockID('preserve.parent.' + label), None))
    child_path = ((BlockID('preserve.getter.entry'), int(raised)),
                  (BlockID('preserve.getter.' + label), None))
    paths = tuple(path for instance in range(2) for path in (
        (program.root, caller_path, f'preservation.{instance}.caller'),
        (original['call'].callee, child_path, f'preservation.{instance}.getter')))
    model, scopes, evidence = _model(program, paths)
    examples, links = [], []
    for instance in range(2):
        parent, child = scopes[2 * instance:2 * instance + 2]
        link = CallActivation(CallActivationID(f'preservation.call.{instance}'),
            parent[2][original['call'].id], child[0], child[1][0],
            child[2][original['child_terminals'][kind].id], evidence)
        links.append(link)
        examples.append({'parent': parent, 'child': child, 'link': link,
            'start': OccurrencePoint(parent[2][original['source'].id], OccurrencePhase.AFTER),
            'end': OccurrencePoint(parent[2][original['targets'][kind].id], OccurrencePhase.BEFORE)})
    model = replace(model, call_activations=tuple(links))
    return program, model, {**original, 'kind': kind, 'examples': tuple(examples)}


def _index(program, model):
    require_valid(program)
    report = validate_execution(model, program)
    assert report.valid, report.issues
    return ExecutionIndex(model, program)


def _query(index, details, instance=0, *, budget=QUERY_BUDGET):
    example = details['examples'][instance]
    return index.state_preservation(example['start'], example['end'],
        locations=(details['world'],), budget=budget)


def _assert_flags(view):
    assert all(view[flag] is False for flag in _FLAGS)


def _assert_complete(index, model, details, instance):
    example = details['examples'][instance]
    view = _query(index, details, instance)
    assert view['preserved'] is True and view['reached'] is True
    assert view['blockers'] == () and view['gaps'] == () and view['cursor'] is None
    assert not view['budget']['truncated']
    allowed = set(example['parent'][2].values()) | set(example['child'][2].values())
    observed = {row['occurrence'].id for row in view['operations'] if row['occurrence'] is not None}
    expected = {example['link'].caller, example['child'][2][details['getter'].id],
        example['child'][2][details['branch'].id], example['link'].completion}
    if details['kind'] is ExitKind.RAISE:
        expected.add(example['child'][2][details['pack'].id])
    assert expected <= observed <= allowed
    for row in view['operations']:
        if row['occurrence'] is not None:
            assert row['definition'].id == row['occurrence'].operation
            assert row['occurrence'].id in allowed
    call, = tuple(row for row in view['calls'] if row['association'] == example['link'])
    assert call['caller'].id == example['link'].caller
    assert call['completion'].id == example['link'].completion
    resolved = index.call_activation(example['link'].id)
    assert call['inputs'] == resolved['inputs'] and call['outputs'] == resolved['outputs']
    assert len(call['outputs']) == 3
    assert {route['edge_argument'].index for route in call['outputs']} == {0, 1, 2}
    values = {value.id: value for value in model.values}
    expected_pairs = {(route['actual_value'].id, route['formal'].id)
        for route in resolved['inputs'] if type(route['formal_definition']) is StateVersionValue}
    expected_pairs.update((values[route['callee_use'].value].id, route['caller_formal'].id)
        for route in resolved['outputs'] if type(route['caller_formal_definition']) is StateVersionValue)
    pairs = {(row['source'].id, row['target'].id) for row in view['state_bindings']
             if row['source'] is not None and row['target'] is not None}
    assert expected_pairs <= pairs
    # Both original state actuals/fullslots remain even though this query asks
    # about world only. Context is not a declaration of disjoint storage.
    assert set(details['locations']) <= {row['location'] for row in view['state_bindings']}
    actual_transfers = {row.operation: row for row in model.transfers}
    assert any(row['transfer'] == actual_transfers[example['link'].caller] for row in view['transfers'])
    _assert_flags(view)
    return view


@pytest.mark.parametrize('raised', (False, True), ids=('normal-fullslots', 'raise-fullslots'))
def test_complete_boundary_and_actual_read_only_getter_preserve_state_on_each_completion(raised):
    program, model, details = _preservation_execution(raised=raised)
    index = _index(program, model)
    for instance in range(2):
        _assert_complete(index, model, details, instance)


@pytest.mark.parametrize('layer', ('outer', 'child'))
def test_completeness_is_required_at_the_call_boundary_and_inside_its_actual_child(layer):
    program, model, details = _preservation_execution(
        outer_complete=layer != 'outer', child_complete=layer != 'child')
    view = _query(_index(program, model), details)
    assert view['preserved'] is None and view['reached'] is True
    example = details['examples'][0]
    culprit = example['link'].caller if layer == 'outer' else example['child'][2][details['getter'].id]
    assert any(row['code'] == 'state_preservation_effects_incomplete'
               and row['operation'] == culprit for row in view['blockers'])
    _assert_flags(view)


@pytest.mark.parametrize('location,mode', (
    ('same', AccessMode.WRITE), ('different', AccessMode.READ_WRITE)),
    ids=('same-location-write', 'different-location-rw-without-isolation'))
def test_unexcluded_writes_cannot_preserve_content_by_location_name(location, mode):
    program, model, details = _preservation_execution(write_location=location, write_mode=mode)
    view = _query(_index(program, model), details)
    assert view['preserved'] is None and view['reached'] is True
    writer = details['examples'][0]['child'][2][details['writer'].id]
    assert any(row['code'] == 'state_preservation_potential_clobber'
               and row['operation'] == writer for row in view['blockers'])
    _assert_flags(view)


@pytest.mark.parametrize('missing', ('call', 'entry', 'completion', 'use', 'transfer', 'operation'))
def test_missing_this_instances_route_or_effect_capture_does_not_borrow_its_complete_sibling(missing):
    program, model, details = _preservation_execution()
    example, sibling = details['examples']
    link = example['link']
    if missing == 'call':
        model = replace(model, call_activations=(sibling['link'],))
    elif missing in ('entry', 'completion'):
        point = next(row.entry for row in model.activations if row.id == link.activation)
        changed = replace(link, **{missing: Unresolved('this call endpoint was not captured', point)})
        model = replace(model, call_activations=(changed, sibling['link']))
    elif missing == 'use':
        state_use = next(use.id for use in details['call'].uses
            if use.id in details['call'].invoke_arguments
            and any(value.id == use.value and type(value) is StateVersionValue for value in program.values))
        model = replace(model, uses=tuple(use for use in model.uses
            if not (use.operation == link.caller and use.use == state_use)))
    elif missing == 'transfer':
        model = replace(model, transfers=tuple(row for row in model.transfers if row.operation != link.caller))
    else:
        omitted = example['child'][2][details['getter'].id]
        model = replace(model, operations=tuple(row for row in model.operations if row.id != omitted),
            uses=tuple(row for row in model.uses if row.operation != omitted))
    index = _index(program, model)
    view = _query(index, details)
    assert view['preserved'] is None
    expected = {
        'call': {'state_preservation_call_missing'},
        'entry': {'state_preservation_call_entry_missing'},
        'completion': {'state_preservation_call_completion_missing'},
        'use': {'state_preservation_call_state_missing', 'state_preservation_state_missing'},
        'transfer': {'state_preservation_transfer_missing'},
        'operation': {'state_preservation_operation_missing'},
    }
    assert expected[missing] & {row['code'] for row in view['blockers']}
    assert all(row['operation'] not in sibling['child'][2].values() for row in view['blockers'])
    _assert_flags(view)
    _assert_complete(index, model, details, 1)


def test_budget_one_retains_a_cursor_without_granting_state_preservation():
    program, model, details = _preservation_execution()
    view = _query(_index(program, model), details, budget=1)
    assert view['preserved'] is None and view['cursor'] is not None
    assert view['budget']['truncated'] is True and view['budget']['used'] <= 1
    assert any(row['code'] == 'state_preservation_budget_exhausted' for row in view['blockers'])
    _assert_flags(view)


def test_a_truncated_checked_prefix_does_not_stand_in_for_the_complete_call_path():
    program, model, details = _preservation_execution(raised=True)
    index = _index(program, model)
    full = _assert_complete(index, model, details, 0)
    prefix = _query(index, details, budget=max(2, full['budget']['used'] // 2))
    assert prefix['operations']
    assert prefix['preserved'] is None and prefix['cursor'] is not None
    assert prefix['budget']['truncated'] is True
    assert any(row['code'] == 'state_preservation_budget_exhausted' for row in prefix['blockers'])
    assert {row['definition'].id for row in prefix['operations']} <= {
        row['definition'].id for row in full['operations']}
    _assert_flags(prefix)
