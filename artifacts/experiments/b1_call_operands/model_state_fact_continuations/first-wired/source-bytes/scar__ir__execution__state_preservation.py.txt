"""Bounded content preservation along original, selected occurrence paths.

This module inventories effects and exact state ports. It creates no state,
producer, alias, phase, or identity relation. The public query owns detaching
the result; the internal query shares its caller's one ValidityBudget.
"""
from __future__ import annotations

from scar.ir.program.ids import LocationID, UseID
from scar.ir.program.model import (
    AccessMode, Behavior, CompletionResult, StateVersionValue, TerminatorKind,
)
from scar.ir.program.resumption import SuspendTerminator
from scar.ir.program.state_addresses import AddressCoverage, InvokeStateOperand

from .common import Unresolved
from .coordinates import AccessRef, OccurrencePhase, OccurrencePoint, UseRef
from .effects import effect_context, resolve_canonical_effect
from .validity_budget import ValidityBudget
from .work import ActualCompletionState, InvokeBoundaryStateEffect


def _validate(ctx, start, end, locations, shared):
    if type(shared) is not ValidityBudget:
        raise TypeError('state_preservation requires an exact shared ValidityBudget')
    for point in (start, end):
        if type(point) is not OccurrencePoint:
            raise TypeError('state preservation endpoints require exact OccurrencePoint')
        point.__post_init__()
        if point.operation not in ctx.occurrences:
            raise KeyError(point.operation)
    if start.phase not in (OccurrencePhase.AFTER, OccurrencePhase.EDGE):
        raise ValueError('state preservation starts AFTER or on an actual EDGE')
    if end.phase is not OccurrencePhase.BEFORE:
        raise ValueError('state preservation ends BEFORE the exact consumer')
    definition = ctx.core_operations[ctx.occurrences[start.operation].operation]
    if start.phase is OccurrencePhase.EDGE:
        if definition.terminator is None or start.successor_index >= len(definition.terminator.successors):
            raise ValueError('start EDGE does not select an original successor')
        transfer = ctx.outgoing.get(start.operation)
        if transfer is not None and transfer.successor_index != start.successor_index:
            raise ValueError('start EDGE differs from the actual selected transfer')
    elif definition.terminator is not None and definition.terminator.kind in (
            TerminatorKind.INVOKE, TerminatorKind.BODY_DRIVE):
        raise ValueError('INVOKE and BodyDrive have no common AFTER state; use the actual EDGE')
    if type(locations) is not tuple or not locations:
        raise TypeError('locations must be a nonempty tuple of original LocationID')
    for location in locations:
        if type(location) is not LocationID:
            raise TypeError('locations must contain exact original LocationID')
        location.__post_init__()
    if len(set(locations)) != len(locations):
        raise ValueError('locations must be unique')
    known = {row.id for row in ctx.program.locations}
    for location in locations:
        if location not in known:
            raise KeyError(location)


def _prepare(ctx):
    """Original indexes only; no path/effect proof is cached across queries."""
    cached = getattr(ctx, '_state_preservation_indexes', None)
    if cached is not None:
        return cached
    occurrences = {(row.visit, row.operation): row for row in ctx.model.operations}
    positions = {identity: ordinal for block in ctx.blocks.values()
                 for ordinal, identity in enumerate(block.operations)}
    parents = {}
    for call in ctx.resolved_call_activations.values():
        parents.setdefault(call['activation'].id, []).append(call)
    refinements = {row.site: row for row in ctx.program.state_address_refinements}
    preparation = {
        'operation_occurrences': len(occurrences), 'original_operation_positions': len(positions),
        'call_associations': len(ctx.resolved_call_activations),
        'state_address_refinements': len(refinements),
        'scope': 'original port and association indexing only; all traversal is budgeted',
    }
    cached = occurrences, positions, {key: tuple(rows) for key, rows in parents.items()}, refinements, preparation
    ctx._state_preservation_indexes = cached
    return cached


class _Walk:
    def __init__(self, ctx, start, end, locations, shared):
        self.ctx, self.start, self.end, self.locations, self.shared = ctx, start, end, locations, shared
        self.index, self.positions, self.parents, self.refinements, self.preparation = _prepare(ctx)
        self.operations, self.calls, self.transfers, self.state_bindings = [], [], [], []
        self.blockers, self.gaps, self.cursor = [], [], None
        self.incomplete = False
        self.stack, self.seen = [], set()
        self.recorded_calls = {}
        self.position = None
        self.reached = None

    def block(self, suffix, operation, reason, **details):
        code = 'state_preservation_' + suffix
        self.blockers.append({'code': code, 'operation': operation, 'reason': reason, **details})
        if code not in self.gaps:
            self.gaps.append(code)

    def spend(self, kind, operation=None):
        if self.cursor is not None:
            return False
        if not self.shared.truncated and self.shared.spend(
                'state_preservation_' + kind, path=str(operation or kind)):
            return True
        self.incomplete = True
        self.cursor = {'phase': kind, 'operation': operation, 'position': self.position}
        self.block('budget_exhausted', operation, 'shared traversal budget exhausted; unvisited work remains')
        return False

    def value(self, identity):
        return None if identity is None or type(identity) is Unresolved else self.ctx.values.get(identity)

    def use(self, occurrence, identity):
        actual = self.ctx.uses.get((occurrence.id, identity))
        value = None if actual is None else self.value(actual.value)
        if value is None:
            self.block('state_missing' if isinstance(self.ctx.core_values[
                self.ctx.core_uses[identity][1].value], StateVersionValue) else 'operand_missing',
                occurrence.id, 'original owned Use has no captured, resolved value', use=identity)
        return actual, value

    def snapshot(self, point, kind):
        current = self.ctx.current(point) or {}
        for location in self.locations:
            if not self.spend('endpoint_states', point.operation):
                return
            value = self.value(current.get(location))
            self.state_bindings.append({'kind': kind, 'point': point, 'location': location,
                                        'source': value, 'target': value})
            if value is None:
                self.block('state_missing', point.operation,
                    'endpoint lacks this original current-state domain', location=location, point=point)

    def effect(self, reference):
        issues, gaps = [], []
        view = resolve_canonical_effect(effect_context(self.ctx,
            issue=lambda code, path, reason: issues.append((code, reason)),
            gap=lambda code, path, reason: gaps.append((code, reason))), reference, 'state_preservation.effect')
        operation = reference.operation if type(reference) is AccessRef else reference.invocation
        for code, reason in (*issues, *gaps):
            self.block('state_missing', operation, reason, original_code=code)
        return view

    def scan(self, occurrence, definition):
        if not self.spend('operations', None if occurrence is None else occurrence.id):
            return
        row = {'occurrence': occurrence, 'definition': definition, 'effects_complete': definition.effects_complete,
               'accesses': [], 'uses': [], 'results': [], 'scan_complete': False}
        self.operations.append(row)
        if occurrence is None:
            self.incomplete = True
            self.block('operation_missing', None, 'this original block position was not captured',
                       definition=definition, visit=self.position[0])
        if not definition.effects_complete:
            self.block('effects_incomplete', None if occurrence is None else occurrence.id,
                       'original operation does not declare complete effects', definition=definition)
        if Behavior.WRITE in definition.behaviors and not any(
                access.mode is not AccessMode.READ for access in definition.accesses):
            self.block('potential_clobber', None if occurrence is None else occurrence.id,
                       'declared write has no independent isolation proof', definition=definition)
        for use in definition.uses:
            if not self.spend('uses', None if occurrence is None else occurrence.id):
                return
            actual, value = (None, None) if occurrence is None else self.use(occurrence, use.id)
            row['uses'].append({'definition': use, 'actual': actual, 'value': value})
            original = self.ctx.core_values[use.value]
            if isinstance(original, StateVersionValue) and occurrence is not None:
                current = self.ctx.before.get(occurrence.id, {}).get(original.location)
                self.state_bindings.append({'kind': 'operation_input', 'operation': occurrence,
                    'definition': use, 'use': actual, 'location': original.location,
                    'source': value, 'target': self.value(current)})
                if value is not None and current != value.id:
                    self.block('state_missing', occurrence.id,
                        'state input is not established at this exact program point', location=original.location)
        for identity in definition.results:
            if not self.spend('results', None if occurrence is None else occurrence.id):
                return
            value = None if occurrence is None else self.ctx.results.get((occurrence.id, identity))
            row['results'].append({'definition': self.ctx.core_values[identity], 'value': value})
            if occurrence is not None and value is None:
                self.block('result_missing', occurrence.id, 'original result occurrence is missing', result=identity)
        for ordinal, access in enumerate(definition.accesses):
            if not self.spend('accesses', None if occurrence is None else occurrence.id):
                return
            if access.mode is not AccessMode.READ:
                self.block('potential_clobber', None if occurrence is None else occurrence.id,
                    'WRITE/RW may change the fact; different domains do not establish isolation', access=access)
            effect = None if occurrence is None else self.effect(AccessRef(occurrence.id, ordinal))
            row['accesses'].append({'index': ordinal, 'definition': access, 'effect': effect})
        term = definition.terminator
        if term is not None and term.kind is TerminatorKind.INVOKE:
            if definition.callee is None:
                self.block('primitive_invoke', None if occurrence is None else occurrence.id,
                    'primitive INVOKE has no whole-boundary preservation implementation')
            self.invoke_states(occurrence, definition)
        elif term is not None and term.kind is TerminatorKind.BODY_DRIVE:
            self.block('body_drive', None if occurrence is None else occurrence.id,
                       'BodyDrive is not a state-preserving primitive')
        elif type(term) is SuspendTerminator:
            self.block('suspend', None if occurrence is None else occurrence.id,
                       'a temporal suspension has no state-preservation guarantee')
        row['uses'], row['results'], row['accesses'] = tuple(row['uses']), tuple(row['results']), tuple(row['accesses'])
        row['scan_complete'] = self.cursor is None

    def invoke_states(self, occurrence, definition):
        transfer = None if occurrence is None else self.ctx.outgoing.get(occurrence.id)
        edge = None if transfer is None else definition.terminator.successors[transfer.successor_index]
        block = None if edge is None else self.ctx.blocks[edge.block]
        for identity in definition.invoke_arguments:
            if not self.spend('invoke_arguments', None if occurrence is None else occurrence.id):
                return
            use = self.ctx.core_uses[identity][1]
            state = self.ctx.core_values[use.value]
            if not isinstance(state, StateVersionValue):
                continue
            site = InvokeStateOperand(definition.id, identity)
            refinement = self.refinements.get(site)
            if refinement is None or refinement.coverage is not AddressCoverage.DECLARED_COMPLETE:
                self.block('invoke_effect_unproved', None if occurrence is None else occurrence.id,
                    'whole original INVOKE state-boundary effects are not completely declared', site=site)
            if refinement is not None:
                for effect in refinement.effects:
                    if not self.spend('invoke_effects', None if occurrence is None else occurrence.id):
                        return
                    if effect.mode is not AccessMode.READ:
                        self.block('potential_clobber', None if occurrence is None else occurrence.id,
                            'INVOKE WRITE/RW has no independent isolation proof', effect=effect, site=site)
            post = None
            if block is not None:
                for position, argument in enumerate(edge.arguments):
                    if not self.spend('invoke_slots', occurrence.id):
                        return
                    formal = self.ctx.core_values[block.arguments[position]]
                    if isinstance(formal, StateVersionValue) and formal.location == state.location:
                        value = self.ctx.results.get((transfer.target, formal.id))
                        if type(argument) is CompletionResult and value is not None:
                            post = ActualCompletionState(transfer.successor_index, argument.index, value.id)
                            break
            if occurrence is None or post is None:
                self.block('state_missing', None if occurrence is None else occurrence.id,
                    'actual INVOKE state completion port is missing', site=site)
                continue
            effect = self.effect(InvokeBoundaryStateEffect(occurrence.id, UseRef(occurrence.id, identity), post))
            before = None if effect is None else effect['input']
            self.state_bindings.append({'kind': 'invoke_state', 'operation': occurrence,
                'location': state.location, 'definition': site, 'effect': effect,
                'argument': CompletionResult(post.result_index), 'successor_index': post.successor_index,
                'source': None if before is None else self.value(before.value),
                'target': self.value(post.formal)})

    def call(self, occurrence):
        if occurrence.id in self.recorded_calls:
            return self.recorded_calls[occurrence.id]
        if not self.spend('calls', occurrence.id):
            return None
        links = self.ctx.call_activations_by_caller.get(occurrence.id, ())
        resolved = self.ctx.resolved_call_activations.get(links[0]) if len(links) == 1 else None
        row = {'association': None if resolved is None else resolved['association'], 'caller': occurrence,
            'entry': None if resolved is None else resolved['entry'],
            'completion': None if resolved is None else resolved['completion'],
            'inputs': () if resolved is None else resolved['inputs'],
            'outputs': () if resolved is None else resolved['outputs'], 'resolved': resolved}
        self.calls.append(row)
        self.recorded_calls[occurrence.id] = row
        if resolved is None:
            self.incomplete = True
            self.block('call_missing', occurrence.id,
                'known callee has no unique exact CallActivation; no activation is guessed')
        return row

    def call_inputs(self, call):
        caller = call['caller']
        if call['entry'] is None:
            self.incomplete = True
            self.block('call_entry_missing', caller.id, 'the linked callee entry visit is not captured')
        for route in call['inputs']:
            if not self.spend('call_inputs', caller.id):
                return
            source, target = route['actual_value'], route['formal']
            if source is None or type(source) is Unresolved or target is None:
                self.block('call_state_missing', caller.id, 'original call actual/formal port is missing',
                           ordinal=route['ordinal'])
            definition = route['formal_definition']
            if isinstance(definition, StateVersionValue):
                self.state_bindings.append({'kind': 'call_input', 'call': call['association'],
                    'ordinal': route['ordinal'], 'location': definition.location,
                    'definition': definition, 'actual': route['actual'],
                    'source': None if type(source) is Unresolved else source, 'target': target})

    def call_outputs(self, call):
        caller, completion = call['caller'], call['completion']
        if completion is None:
            self.incomplete = True
            self.block('call_completion_missing', caller.id, 'the exact child completion is not captured')
        for route in call['outputs']:
            if not self.spend('call_outputs', caller.id):
                return
            use = route['callee_use']
            source = None if use is None else self.value(use.value)
            target = route['caller_formal'] or route['caller_result']
            if source is None or target is None:
                self.block('call_state_missing', caller.id,
                    'original child exit/caller result port is missing', ordinal=route['ordinal'])
            original = route['callee_use_definition']
            definition = None if original is None else self.ctx.core_values[original.value]
            if isinstance(definition, StateVersionValue):
                self.state_bindings.append({'kind': 'call_output', 'call': call['association'],
                    'ordinal': route['ordinal'], 'location': definition.location,
                    'definition': definition, 'use': use, 'argument': route['edge_argument'],
                    'source': source, 'target': target})

    def follow(self, occurrence, call=None, *, anchor=False):
        if not self.spend('transfers', occurrence.id):
            return None
        transfer = self.ctx.outgoing.get(occurrence.id)
        if transfer is None:
            self.incomplete = True
            self.block('transfer_missing', occurrence.id, 'the exact selected transfer is not captured')
            return None
        definition = self.ctx.core_operations[occurrence.operation]
        edge = definition.terminator.successors[transfer.successor_index]
        target = self.ctx.visits[transfer.target]
        self.transfers.append({'transfer': transfer, 'operation': occurrence, 'successor': edge, 'target': target})
        block = self.ctx.blocks[target.block]
        outputs = {} if call is None else {route['ordinal']: route for route in call['outputs']}
        for position, argument in enumerate(edge.arguments):
            if not self.spend('transfer_slots', occurrence.id):
                return None
            original = self.ctx.core_values[block.arguments[position]]
            if not isinstance(original, StateVersionValue):
                continue
            formal = self.ctx.results.get((target.id, original.id))
            actual, source = None, None
            if isinstance(argument, UseID):
                actual, source = self.use(occurrence, argument)
            elif anchor and type(argument) is CompletionResult:
                # The query begins on this established post. This is an
                # output anchor, not an input-to-output preservation claim.
                source = formal
            elif type(argument) is CompletionResult and argument.index in outputs:
                actual = outputs[argument.index]['callee_use']
                source = None if actual is None else self.value(actual.value)
            else:
                self.block('state_missing', occurrence.id,
                    'state projection has no proved preservation source', argument=argument,
                    location=original.location)
            self.state_bindings.append({'kind': 'start_state' if anchor else 'transfer_state', 'operation': occurrence,
                'transfer': transfer, 'position': position, 'argument': argument,
                'location': original.location, 'definition': original, 'use': actual,
                'source': source, 'target': formal})
            if formal is None or source is None:
                self.block('state_missing', occurrence.id, 'original edge state binding is not captured',
                           location=original.location)
        return target.id, 0

    def resume(self, call):
        self.call_outputs(call)
        if self.cursor is not None:
            return None
        caller = call['caller']
        definition = self.ctx.core_operations[caller.operation]
        if definition.terminator is not None and definition.terminator.kind is TerminatorKind.INVOKE:
            return self.follow(caller, call)
        return caller.visit, self.positions[caller.operation] + 1

    def escape(self, occurrence):
        """Resume only an explicit parent association, never a sibling frame."""
        activation = self.ctx.segments[self.ctx.visits[self.position[0]].segment].activation
        if self.stack:
            call = self.stack.pop()
        else:
            candidates = self.parents.get(activation, ())
            if len(candidates) != 1:
                return None
            call = self.call(candidates[0]['caller'])
            if call is None:
                return None
            # The call began before start, but its boundary still must be
            # complete when this post is lifted to the caller.
            self.scan(call['caller'], self.ctx.core_operations[call['caller'].operation])
            self.call_inputs(call)
        completion = call['completion']
        if occurrence is None or completion is None or occurrence.id != completion.id:
            self.incomplete = True
            self.block('call_completion_missing', call['caller'].id,
                       'captured child path has not reached its exact associated completion')
        return self.resume(call)

    def run(self):
        self.snapshot(self.start, 'start')
        self.snapshot(self.end, 'end')
        if self.cursor is not None:
            return self.result()
        source = self.ctx.occurrences[self.start.operation]
        if self.start.phase is OccurrencePhase.EDGE:
            self.position = (source.visit, self.positions[source.operation])
            self.position = self.follow(source, anchor=True)
        else:
            definition = self.ctx.core_operations[source.operation]
            term = definition.terminator
            if term is not None and term.kind not in (TerminatorKind.RETURN, TerminatorKind.RAISE):
                self.position = source.visit, self.positions[source.operation]
                if type(term) is SuspendTerminator:
                    self.block('suspend', source.id, 'temporal resumption has no state-preservation guarantee')
                self.position = self.follow(source)
            else:
                self.position = source.visit, self.positions[source.operation] + 1
        while self.position is not None and self.cursor is None:
            visit_id, position = self.position
            if not self.spend('positions'):
                break
            if self.position in self.seen:
                self.incomplete = True
                self.block('route_missing', None, 'selected occurrence path repeats a previously examined position')
                break
            self.seen.add(self.position)
            visit = self.ctx.visits[visit_id]
            block = self.ctx.blocks[visit.block]
            if position >= len(block.operations):
                last = self.index.get((visit_id, block.operations[-1]))
                next_position = self.escape(last)
                if next_position is None and self.cursor is None and not self.incomplete:
                    self.reached = False if self.ctx.activation_of(self.end.operation) == \
                        self.ctx.segments[visit.segment].activation else None
                self.position = next_position
                continue
            identity = block.operations[position]
            occurrence = self.index.get((visit_id, identity))
            if occurrence is not None and occurrence.id == self.end.operation:
                self.reached = None if self.incomplete else True
                break
            definition = self.ctx.core_operations[identity]
            self.scan(occurrence, definition)
            if self.cursor is not None:
                break
            if definition.callee is not None and occurrence is not None:
                call = self.call(occurrence)
                if call is None:
                    break
                self.call_inputs(call)
                if call['entry'] is not None:
                    self.stack.append(call)
                    self.position = call['entry'].id, 0
                else:
                    self.position = self.resume(call)
                continue
            term = definition.terminator
            if term is None:
                self.position = visit_id, position + 1
            elif term.kind in (TerminatorKind.RETURN, TerminatorKind.RAISE):
                next_position = self.escape(occurrence)
                if next_position is None and self.cursor is None and not self.incomplete:
                    self.reached = False if self.ctx.activation_of(self.end.operation) == \
                        self.ctx.segments[visit.segment].activation else None
                self.position = next_position
            elif occurrence is None:
                self.position = self.escape(None)
            else:
                self.position = self.follow(occurrence)
                if self.position is None and self.cursor is None:
                    self.position = self.escape(None)
        if self.reached is not True and self.cursor is None:
            self.block('route_missing', None, 'no complete original selected route reaches the endpoint')
        return self.result()

    def result(self):
        for row in self.operations:
            for name in ('uses', 'results', 'accesses'):
                row[name] = tuple(row[name])
        return {'start': self.start, 'end': self.end, 'locations': self.locations,
            'preserved': True if self.reached is True and not self.blockers and not self.shared.truncated else None,
            'reached': None if self.shared.truncated else self.reached,
            'operations': tuple(self.operations), 'calls': tuple(self.calls),
            'transfers': tuple(self.transfers), 'state_bindings': tuple(self.state_bindings),
            'blockers': tuple(self.blockers), 'gaps': tuple(self.gaps), 'cursor': self.cursor,
            'budget': {'limit': self.shared.limit, 'used': self.shared.used,
                'costs': dict(self.shared.costs), 'truncated': self.shared.truncated},
            'preparation': dict(self.preparation), 'source_runtime_verified': False,
            'runtime_identity_verified': False, 'value_identity_verified': False,
            'transformation_authorized': False}


def state_preservation(ctx, start, end, locations, *, shared):
    """Inspect one exact interval without changing validation diagnostics."""
    _validate(ctx, start, end, locations, shared)
    return _Walk(ctx, start, end, locations, shared).run()


__all__ = ['state_preservation']
