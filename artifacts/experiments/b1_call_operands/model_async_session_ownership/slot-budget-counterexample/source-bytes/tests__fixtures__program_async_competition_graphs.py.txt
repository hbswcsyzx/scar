def _operation(graph, identity):
    return next(row for row in graph.operations if row.id == identity)


def _roles(values, name, identities):
    return tuple(Use(UseID(f'{name}.{slot}'), identity,
        UseRole.STATE_CONTENT if type(values[identity]) is StateVersionValue else
        UseRole.PROTOCOL if type(values[identity]) is ProtocolValue else
        UseRole.DATA if type(values[identity]) is DataValue else UseRole.REFERENCE)
        for slot, identity in enumerate(identities))


def _terminal(builder, name, identities, *, raised=False):
    uses = _roles({row.id: row for row in builder.values}, name + '.use', identities)
    return builder.add(name, uses, effects_complete=True,
        terminator=Terminator(TerminatorKind.RAISE if raised else TerminatorKind.RETURN,
            exit='failure' if raised else 'normal', values=tuple(use.id for use in uses)))


def _getter(domains):
    """An explicit getter with a genuine raising native field boundary.

    Association meanings remain the independent RequestAssociation contract.
    The native NORMAL port loads RAW; it does not construct a raw request or
    select its kind. Its opaque primitive remains a preservation blocker.
    """
    states = tuple(ValueType(ValueKind.STATE, location=location) for location in domains)
    b = _Builder('competition.association.getter', domains)
    entry = b.formals('competition.getter.entry', (REFERENCE, *states))
    identity = OperationID('competition.getter.read')
    loaded = BlockID('competition.getter.loaded')
    body = b.value('competition.getter.body', BlockArgument(loaded, 0), REFERENCE)
    latent = b.value('competition.getter.latent', BlockArgument(loaded, 1), RAW)
    boolean = DataValue(ValueID('competition.getter.accepted'), b.region,
                        BlockArgument(loaded, 2), 'builtins.bool')
    exception = b.value('competition.getter.exception', BlockArgument(loaded, 3), REFERENCE)
    traceback = b.value('competition.getter.traceback', BlockArgument(loaded, 4), REFERENCE)
    loaded_states = tuple(b.value(f'competition.getter.loaded.state.{slot}',
        BlockArgument(loaded, 5 + slot), type_) for slot, type_ in enumerate(states))
    b.values.append(boolean)
    actuals = _roles({row.id: row for row in b.values}, identity.value + '.use', entry)
    native_failure = b.formals('competition.getter.native.failure', (PACKET, *states))
    loaded_types = (REFERENCE, RAW, ValueType(ValueKind.DATA, data_type='builtins.bool'),
                    REFERENCE, REFERENCE, *states)
    read = Operation(identity, 'raising native association field access',
        (Behavior.CONTROL,), actuals, effects_complete=False,
        invoke_arguments=tuple(use.id for use in actuals),
        completions=(CompletionSignature(ExitKind.NORMAL, loaded_types),
                     CompletionSignature(ExitKind.RAISE, (PACKET, *states))),
        terminator=Terminator(TerminatorKind.INVOKE, (
            CompletionSuccessor(ExitKind.NORMAL, loaded,
                tuple(CompletionResult(slot) for slot in range(len(loaded_types)))),
            CompletionSuccessor(ExitKind.RAISE, BlockID('competition.getter.native.failure'),
                tuple(CompletionResult(slot) for slot in range(len(native_failure)))))))
    b.operations.append(read)
    normal = b.formals('competition.getter.normal', (REFERENCE, RAW, *states))
    failure = b.formals('competition.getter.failure', (REFERENCE, REFERENCE, *states))
    values = {row.id: row for row in b.values}
    condition, = _roles(values, 'competition.getter.condition', (boolean.id,))
    normal_uses = _roles(values, 'competition.getter.normal.forward', (body, latent, *loaded_states))
    failure_uses = _roles(values, 'competition.getter.failure.forward', (exception, traceback, *loaded_states))
    branch = b.add('competition.getter.branch', (condition, *normal_uses, *failure_uses), effects_complete=True,
        terminator=Terminator(TerminatorKind.BRANCH, (
            Successor(BlockID('competition.getter.normal'), tuple(use.id for use in normal_uses)),
            Successor(BlockID('competition.getter.failure'), tuple(use.id for use in failure_uses))),
            condition=condition.id))
    b.block('competition.getter.entry', entry, (read,))
    b.block(loaded.value, (body, latent, boolean.id, exception, traceback, *loaded_states), (branch,))
    returned = _terminal(b, 'competition.getter.return', normal)
    b.block('competition.getter.normal', normal, (returned,))
    packet = b.value('competition.getter.packet', OperationResult(OperationID('competition.getter.pack'), 0), PACKET)
    components = (_use('competition.getter.pack.exception', failure[0]),
                  _use('competition.getter.pack.traceback', failure[1]))
    pack = b.add('competition.getter.pack', components, results=(packet,), effects_complete=True)
    raised = _terminal(b, 'competition.getter.raise', (packet, *failure[2:]), raised=True)
    b.block('competition.getter.failure', failure, (pack, raised))
    native_raise = _terminal(b, 'competition.getter.native.raise', native_failure, raised=True)
    b.block('competition.getter.native.failure', native_failure, (native_raise,))
    region = replace(b.finish('competition.getter.entry'), exits=(
        ExitPort('normal', ExitKind.NORMAL, (REFERENCE, RAW, *states)),
        ExitPort('failure', ExitKind.RAISE, (PACKET, *states))))
    return {'region': region, 'operations': tuple(b.operations), 'values': tuple(b.values),
        'control_scope': _scope(region.id, region.blocks),
        'packet': ExceptionPacketDefinition(packet, components[0].id, components[1].id),
        'read': read, 'branch': branch, 'terminal': returned,
        'native_raise': native_raise,
        'path': ((region.entry, 0), (loaded, 0), (BlockID('competition.getter.normal'), None))}


def _ordinary_handoff(graph, *, local_probe):
    """A new explicit CONTROL return; the old opaque return is not relabeled."""
    region = next(row for row in graph.regions if row.id == graph.root)
    update, = graph.session_updates
    block = next(row for row in region.blocks if update.operation in row.operations)
    old = _operation(graph, block.operations[-1])
    uses = tuple(replace(use, id=UseID('competition.handoff.' + str(slot)))
                 for slot, use in enumerate(old.uses))
    terminal = Operation(OperationID('competition.handoff.return'), 'explicit ordinary await handoff return',
        (Behavior.CONTROL,), uses, effects_complete=True,
        terminator=Terminator(TerminatorKind.RETURN, exit='normal', values=tuple(use.id for use in uses)))
    operations = tuple(terminal if row.id == old.id else row for row in graph.operations)
    blocks = tuple(replace(row, operations=(*row.operations[:-1], terminal.id)) if row.id == block.id else row
                   for row in region.blocks)
    graph = replace(graph, operations=operations,
        regions=tuple(replace(row, blocks=blocks) if row.id == region.id else row for row in graph.regions))
    probe = None
    if local_probe:
        # The entire proved interval is inside D0, before its host envelope.
        # Both branch edges copy the same original full tuple to a real join.
        values = {row.id: row for row in graph.values}
        source = _operation(graph, update.operation)
        owner = next(use.value for use in source.uses if use.id == update.body)
        state_values = tuple(use.value for use in terminal.uses
                             if type(values[use.value]) is StateVersionValue)
        identity = OperationID('competition.local.occupancy.read')
        boolean = DataValue(ValueID('competition.local.occupancy.bool'), region.id,
                            OperationResult(identity, 0), 'builtins.bool')
        actuals = (_use('competition.local.body', owner),
                   *(_use(f'competition.local.state.{i}', value, state=True) for i, value in enumerate(state_values)))
        read = Operation(identity, 'read active session after this actual HOLD',
            (Behavior.CONTROL, Behavior.READ), actuals, (boolean.id,),
            tuple(StateAccess(values[use.value].location, use.id, AccessMode.READ) for use in actuals[1:]),
            effects_complete=True)
        all_values = (*[use.value for use in terminal.uses], owner)
        join = BlockID('competition.local.occupancy.join')
        formals = tuple(replace(values[value], id=ValueID(f'competition.local.join.{slot}'),
                                origin=BlockArgument(join, slot)) for slot, value in enumerate(all_values))
        condition = Use(UseID('competition.local.condition'), boolean.id, UseRole.DATA)
        forwarded = tuple(_roles(values, f'competition.local.edge.{edge}', all_values) for edge in range(2))
        branch = Operation(OperationID('competition.local.branch'), 'observe occupancy without rejecting R1',
            (Behavior.CONTROL,), (condition, *forwarded[0], *forwarded[1]), effects_complete=True,
            terminator=Terminator(TerminatorKind.BRANCH, tuple(Successor(join, tuple(use.id for use in actual))
                for actual in forwarded), condition=condition.id))
        replacement = dict(zip(all_values, (row.id for row in formals)))
        terminal = replace(terminal, uses=tuple(replace(use, value=replacement[use.value]) for use in terminal.uses))
        blocks = tuple(replace(row, operations=(*row.operations[:-1], read.id, branch.id))
                       if row.id == block.id else row for row in blocks)
        blocks = (*blocks, Block(join, tuple(row.id for row in formals), (terminal.id,)))
        scope, = graph.protocol_scopes
        row = ProtocolOperation(scope.id, ProtocolClause.ASYNC_REQUEST_SOURCE, read.id, 0,
            AdmissionPredicate(SessionOccupancyCondition(actuals[0].id,
                tuple(ExistingAccess(read.id, index) for index in range(len(read.accesses)))),
                OperationResult(read.id, 0), branch.id), _source_gap())
        graph = replace(graph,
            operations=(*(terminal if op.id == terminal.id else op for op in graph.operations), read, branch),
            values=(*graph.values, boolean, *formals),
            regions=tuple(replace(owner, blocks=blocks) if owner.id == region.id else owner for owner in graph.regions),
            control_scopes=tuple(replace(owner, blocks=tuple(item.id for item in blocks))
                if owner.region == region.id else owner for owner in graph.control_scopes),
            protocol_operations=(*graph.protocol_operations, row))
        probe = {'row': row, 'predicate': read, 'branch': branch,
                 'body_use': OwnedUse(read.id, actuals[0].id)}
    return graph, terminal, probe


def _competition_driver(body, *, reject_raised, known_association):
    graph, initial = _pipeline_stage(body, 1)
    interface, = graph.driver_interfaces
    association, = graph.request_association_reads
    operation = _operation(graph, association.operation)
    association_row = next(row for row in graph.protocol_operations if row.operation == operation.id)
    initial_row = next(row for row in graph.protocol_operations if row.operation == initial['predicate'])
    initial_branch = _operation(graph, initial['branch'])
    # R2 is fresh. Its initial true arm selects L2, independently of R1.
    initial_branch = replace(initial_branch, terminator=replace(initial_branch.terminator,
        successors=tuple(reversed(initial_branch.terminator.successors))))
    graph = replace(graph, operations=tuple(initial_branch if row.id == initial_branch.id else row for row in graph.operations))
    region = next(row for row in graph.regions if row.id == interface.region)
    body_value = next(row for row in graph.values if type(row.origin) is BlockArgument
        and row.origin.block == BlockID('driver.no_body.test') and row.origin.index == 0)
    initial_map = initial['admitted_map']
    owner = initial_map[body_value.id]
    session = initial_map[next(row for row in graph.values if type(row.origin) is BlockArgument
        and row.origin.block == BlockID('driver.no_body.test') and row.origin.index == 3).id]
    scheduler = initial_map[next(row for row in graph.values if type(row.origin) is BlockArgument
        and row.origin.block == BlockID('driver.no_body.test') and row.origin.index == 2).id]
    graph, guard = _guard(graph, block_id=initial['admitted'], owner=owner, raw=scheduler,
        label='competition.occupied', rule=BodyStatePredicateRule.IS_RUNNING,
        admit_true=False, request_kind=RawRequestKind.NEXT)
    predicate = _operation(graph, guard['predicate'])
    branch = _operation(graph, guard['branch'])
    # The READ's owner is B. The rejecting helper's receiver is R2: these are
    # separate Uses even though the old local split builder had one owner role.
    rejected = branch.terminator.successors[0]
    request_forward = rejected.arguments[0]
    branch = replace(branch, uses=tuple(replace(use, value=session) if use.id == request_forward else use
                                        for use in branch.uses))
    graph = replace(graph, operations=tuple(branch if row.id == branch.id else row for row in graph.operations))
    rows = []
    for row in graph.protocol_operations:
        if row.operation == predicate.id:
            subject = row.contract.subject
            row = replace(row, clause=ProtocolClause.ASYNC_REQUEST_SOURCE,
                contract=replace(row.contract, subject=SessionOccupancyCondition(subject.instance_use, subject.state_sites)))
        elif row.operation == guard['helper'] and type(row.contract) is RequestProcessing:
            helper = _operation(graph, row.operation)
            row = replace(row, clause=ProtocolClause.ASYNC_REQUEST_DELIVERY,
                contract=replace(row.contract, raw_use=helper.invoke_arguments[0], rule=RequestProcessingRule.CHECK_ASYNC_REQUEST))
        rows.append(row)
    graph = replace(graph, protocol_operations=tuple(rows))
    getter, refinements = None, ()
    if known_association:
        domains = tuple(row.id for row in graph.locations)
        getter = _getter(domains)
        operation = replace(operation, callee=getter['region'].id, effects_complete=True)
        values = {row.id: row for row in graph.values}
        refinements = tuple(StateAddressRefinement(InvokeStateOperand(operation.id, identity),
            (AddressEffect(DomainOwner(), (WholeOwner(),), AccessMode.READ),), AddressCoverage.DECLARED_COMPLETE)
            for identity in operation.invoke_arguments
            if type(values[next(use.value for use in operation.uses if use.id == identity)]) is StateVersionValue)
        graph = replace(graph, operations=tuple(operation if row.id == operation.id else row for row in graph.operations))
    component = _driver_component(graph, 'competition.r2.driver')
    rename = component['rename']
    selected = guard['terminals'][int(reject_raised)]
    choices = {association.operation: 0, initial_branch.id: 0, branch.id: 0, guard['helper']: int(reject_raised)}
    component['choices'] = {rename(key): value for key, value in choices.items()}
    component['host'] = rename(selected)
    component['path'] = _selected_path(replace(graph, regions=(component['region'],), operations=component['operations']),
                                     component['region'], component['choices'])
    component['state_address_refinements'] = rename(refinements)
    metadata = rename({'row': next(row for row in rows if row.operation == predicate.id),
        'predicate': predicate, 'branch': branch, 'body_use': OwnedUse(predicate.id, predicate.uses[0].id),
        'initial_row': initial_row, 'initial_predicate': _operation(graph, initial_row.operation),
        'initial_branch': initial_branch, 'association_row': association_row,
        'association': association, 'association_operation': operation, 'interface': interface,
        'reject': _operation(graph, guard['helper']),
        'reject_rows': tuple(row for row in rows if row.operation == guard['helper']),
        'terminals': tuple(_operation(graph, identity) for identity in guard['terminals']),
        'selected_terminal': _operation(graph, selected), 'scope': rows[0].scope})
    if getter is not None:
        getter = {**getter, 'region': replace(getter['region'], parent=component['region'].id)}
    return component, metadata, getter


def async_competition_program(*, reject_raised=False, known_association=True, local_probe=False):
    """One B, independent R1/L1 and R2/L2, rejection, then R1's real continuation."""
    if any(type(value) is not bool for value in (reject_raised, known_association, local_probe)):
        raise TypeError('fixture switches require exact bools')
    body = _three_cut_body()
    stages, local = [], None
    for number in range(3):
        graph, initial = _pipeline_stage(body, number)
        if number == 0 and (known_association or local_probe):
            graph, handoff, local = _ordinary_handoff(graph, local_probe=local_probe)
        component = _driver_component(graph, f'competition.r1.driver.{number}')
        rename = component['rename']
        component['predicate'] = rename(initial)
        drive, = component['body_drives']
        association, = component['request_association_reads']
        choices = {component['final_body_gates'][0].operation: 0, drive.operation: 0,
                   association.operation: 0, component['predicate']['branch']: 0 if number == 0 else 1}
        if number == 2:
            choices[rename(OperationID('adapter.construct'))] = 0
        if number == 0 and (known_association or local_probe):
            component['host'] = rename(handoff.id)
            if local is not None:
                local = rename(local)
                choices[local['branch'].id] = 0
        else:
            component['host'] = rename(OperationID('adapter.normal.exit' if number == 2 else 'driver.outcome.exit'))
        component['choices'] = choices
        component['path'] = _selected_path(replace(graph, regions=(component['region'],), operations=component['operations']),
                                         component['region'], choices)
        component['state_address_refinements'] = ()
        stages.append(component)
    competitor, occupancy, getter = _competition_driver(body,
        reject_raised=reject_raised, known_association=known_association)
    components = (stages[0], competitor, stages[1], stages[2])
    domains = tuple(row.id for row in graph.locations)
    states = tuple(ValueType(ValueKind.STATE, location=identity) for identity in domains)
    b = _Builder('competition.caller', domains)
    body_region, = body.regions
    values = {row.id: row for row in body.values}
    body_entry = next(row for row in body_region.blocks if row.id == body_region.entry)
    captures = tuple(identity for identity in body_entry.arguments if type(values[identity]) is ReferenceValue)
    entry_name = 'competition.entry'
    entry = b.formals(entry_name, (*(REFERENCE for _ in range(len(captures) + 2)), *states))
    actuals = tuple(_use(f'competition.create.input.{i}', value, state=i >= len(captures))
                   for i, value in enumerate((*entry[:len(captures)], *entry[-len(states):])))
    created = b.formals('competition.body.created', (REFERENCE, *states))
    failure = b.formals('competition.body.failed', (PACKET, *states))
    create = b.add('competition.body.create', actuals, effects_complete=True,
        invoke_arguments=tuple(use.id for use in actuals),
        completions=(CompletionSignature(ExitKind.NORMAL, (REFERENCE, *states)),
                     CompletionSignature(ExitKind.RAISE, (PACKET, *states))),
        terminator=Terminator(TerminatorKind.INVOKE, (
            CompletionSuccessor(ExitKind.NORMAL, BlockID('competition.body.created'), tuple(CompletionResult(i) for i in range(len(created)))),
            CompletionSuccessor(ExitKind.RAISE, BlockID('competition.body.failed'), tuple(CompletionResult(i) for i in range(len(failure)))))))
    capture_rows = tuple(CreationCapture(formal, use.id) for formal, use in zip(captures, actuals))
    function = next(row.reference for row in body.environments[0].members if row.role is EnvironmentRole.FUNCTION_OBJECT)
    creation = ResumableCreate(create.id, body_region.id,
        next(row.source for row in capture_rows if row.formal == function), capture_rows, NormalResultPort(0), driver_rule())
    b.block(entry_name, entry, (create,))
    b.block('competition.body.failed', failure, (_terminal(b, 'competition.body.failure', failure, raised=True),))
    none = ReferenceValue(ValueID('competition.none'), b.region, LiteralOrigin(None), targets=())
    b.values.append(none)
    current_name, current, current_body, current_states = 'competition.body.created', created, created[0], created[1:]
    requests, raws, request_values, choices = [], [], [], {create.id: 0}
    for number in range(2):
        label = f'competition.request.{number}'
        operands = () if number == 0 else (_use(label + '.latent.operand', none.id),)
        latent = b.value(label + '.latent', OperationResult(OperationID(label + '.pack'), 0), RAW)
        pack = b.add(label + '.pack', operands, results=(latent,), effects_complete=True)
        raw = RawRequestDefinition(latent, RawRequestKind.NEXT if number == 0 else RawRequestKind.SEND,
                                   tuple(use.id for use in operands))
        raws.append(raw)
        actuals = (_use(label + '.body', current_body), _use(label + '.latent.use', latent, protocol=True),
                  *(_use(f'{label}.state.{i}', state, state=True) for i, state in enumerate(current_states)))
        retained = (*request_values, current_body)
        forwards = tuple(_use(f'{label}.forward.{i}', value) for i, value in enumerate(retained))
        normal_name, failure_name = label + '.normal', label + '.failure'
        normal = b.formals(normal_name, (REFERENCE, *(REFERENCE for _ in retained), *states))
        failure = b.formals(failure_name, (PACKET, *states))
        call = b.add(label + '.create', (*actuals, *forwards), effects_complete=True,
            invoke_arguments=tuple(use.id for use in actuals),
            completions=(CompletionSignature(ExitKind.NORMAL, (REFERENCE, *states)),
                         CompletionSignature(ExitKind.RAISE, (PACKET, *states))),
            terminator=Terminator(TerminatorKind.INVOKE, (
                CompletionSuccessor(ExitKind.NORMAL, BlockID(normal_name),
                    (CompletionResult(0), *(use.id for use in forwards), *(CompletionResult(i + 1) for i in range(len(states))))),
                CompletionSuccessor(ExitKind.RAISE, BlockID(failure_name), tuple(CompletionResult(i) for i in range(len(failure)))))))
        row = AsyncRequestCreate(call.id, actuals[0].id, actuals[1].id, NormalResultPort(0), driver_rule())
        requests.append({'create': row, 'operation': call, 'latent': raw,
            'result': InvokeLiteralResult(ControlEdgeReference(call.id, 0), CompletionResult(0))})
        b.block(current_name, current, (pack, call))
        b.block(failure_name, failure, (_terminal(b, failure_name + '.exit', failure, raised=True),))
        choices[call.id] = 0
        current_name, current = normal_name, normal
        request_values = [normal[0], *normal[1:1 + len(request_values)]]
        current_body = normal[1 + number]
        current_states = normal[2 + number:]
    # Request2's own result precedes the independently forwarded Request1.
    r2, r1 = request_values
    drivers, catch, final = [], None, None
    for ordinal, component in enumerate(components):
        label = f'competition.call.{ordinal}'
        receiver = r2 if ordinal == 1 else r1
        sent = none.id if ordinal == 1 else entry[len(captures) + (0 if ordinal < 3 else 1)]
        operands = () if ordinal in (0, 1) else (_use(label + '.scheduler.operand', sent),)
        scheduler = b.value(label + '.scheduler', OperationResult(OperationID(label + '.pack'), 0), RAW)
        pack = b.add(label + '.pack', operands, results=(scheduler,), effects_complete=True)
        raws.append(RawRequestDefinition(scheduler, RawRequestKind.NEXT if not operands else RawRequestKind.SEND,
                                        tuple(use.id for use in operands)))
        interface, = component['driver_interfaces']
        region = component['region']
        driver_values = {row.id: row for row in component['values']}
        driver_entry = next(row for row in region.blocks if row.id == region.entry)
        supplied = []
        for identity in driver_entry.arguments:
            definition = driver_values[identity]
            actual = (receiver if identity == interface.receiver else
                scheduler if identity == interface.scheduler_request else
                current_states[domains.index(definition.location)] if type(definition) is StateVersionValue else
                current_body if identity == driver_entry.arguments[0] else sent)
            supplied.append(actual)
        actuals = _roles({row.id: row for row in b.values}, label + '.actual', supplied)
        keep = (r1, r2, current_body)
        edges, forwards, continuation = [], [], None
        selected_kind = ExitKind.NORMAL if ordinal in (0, 2) else ExitKind.RAISE
        for port in region.exits:
            name = label + '.' + port.kind.value.lower()
            continuing = ordinal in (0, 2) and port.kind is ExitKind.NORMAL or ordinal == 1 and port.kind is ExitKind.RAISE
            args = b.formals(name, (*(REFERENCE for _ in keep), *port.types) if continuing else port.types)
            forwarded = tuple(_use(f'{name}.forward.{i}', value) for i, value in enumerate(keep)) if continuing else ()
            forwards.extend(forwarded)
            edges.append(CompletionSuccessor(port.kind, BlockID(name),
                (*(use.id for use in forwarded), *(CompletionResult(i) for i in range(len(port.types))))))
            if continuing:
                continuation = name, args
            else:
                terminal = _terminal(b, name + '.exit', args, raised=port.kind is ExitKind.RAISE)
                b.block(name, args, (terminal,))
                if ordinal == 3 and port.kind is selected_kind:
                    final = terminal
        call = b.add(label, (*actuals, *forwards), effects_complete=True, callee=region.id,
            invoke_arguments=tuple(use.id for use in actuals),
            completions=tuple(CompletionSignature(port.kind, port.types) for port in region.exits),
            terminator=Terminator(TerminatorKind.INVOKE, tuple(edges)))
        prefix = (catch,) if ordinal == 2 else ()
        b.block(current_name, current, (*prefix, pack, call))
        choices[call.id] = next(i for i, edge in enumerate(edges) if edge.kind is selected_kind)
        drive, = component['body_drives']
        commit, = component['body_outcome_commits']
        update, = component['session_updates']
        host = next(row for row in component['operations'] if row.id == component['host'])
        drivers.append({'caller_call': call, 'driver_region': region, 'interface': interface,
            'drive': drive, 'outcome': commit.outcome, 'outcome_source': commit.outcome.source,
            'update': update, 'publication': commit, 'host_terminal': host,
            'association_read': component['request_association_reads'][0],
            'path': component['path'], 'path_choices': component['choices'], 'selected_call_edge': choices[call.id]})
        if continuation is not None:
            current_name, current = continuation
            r1, r2, current_body = current[:3]
            current_states = current[4:]
            if ordinal == 1:
                context = body.resumable_bodies[0].context_location
                position = domains.index(context)
                before = _use('competition.catch.before', current_states[position], state=True)
                packet_use = _use('competition.catch.packet', current[3], protocol=True)
                after = b.value('competition.catch.after', OperationResult(OperationID('competition.catch'), 0), states[position])
                catch = Operation(OperationID('competition.catch'), 'install this exact rejected packet',
                    (Behavior.CONTROL, Behavior.READ, Behavior.WRITE), (packet_use, before), (after,),
                    (StateAccess(context, before.id, AccessMode.READ_WRITE, after),), effects_complete=True)
                b.operations.append(catch)
                current_states = tuple(after if i == position else value for i, value in enumerate(current_states))
    caller = b.finish(entry_name)
    helper_regions = () if getter is None else (getter['region'],)
    helper_operations = () if getter is None else getter['operations']
    helper_values = () if getter is None else getter['values']
    helper_scopes = () if getter is None else (getter['control_scope'],)
    graph = replace(body, root=caller.id,
        regions=(caller, replace(body_region, parent=caller.id),
            *(replace(component['region'], parent=caller.id) for component in components), *helper_regions),
        operations=(*body.operations, *(row for component in components for row in component['operations']),
                    *helper_operations, *b.operations),
        values=(*body.values, *(row for component in components for row in component['values']), *helper_values, *b.values),
        locations=graph.locations,
        control_scopes=(*body.control_scopes, *(row for component in components for row in component['control_scopes']),
            *helper_scopes, ControlScope(ControlScopeID('competition.caller.function'), caller.id, None, ControlScopeKind.FUNCTION,
                _source('two supplied request Creates and exact selected calls'), tuple(block.id for block in caller.blocks))),
        resumable_creates=(creation,), async_request_creates=tuple(item['create'] for item in requests),
        raw_driver_requests=tuple(raws),
        state_address_refinements=tuple(row for component in components for row in component['state_address_refinements']),
        **{name: tuple(row for component in components for row in component[name])
            + ((getter['packet'],) if name == 'exception_packets' and getter is not None else ()) for name in _TABLES})
    hold = drivers[0]['update']
    return graph, {'caller': caller, 'caller_path': _selected_path(graph, caller, choices),
        'body': body_region, 'primary_creation': creation, 'body_create': create,
        'drivers': tuple(drivers), 'requests': tuple(requests), 'occupancy': occupancy,
        'first_hold': {'update': hold, 'operation': _operation(graph, hold.operation), 'driver': drivers[0]},
        'getter': getter, 'local_probe': local, 'catch': catch, 'final': final,
        'known_association': known_association, 'reject_raised': reject_raised,
        'domains': domains, 'caller_choices': choices}
