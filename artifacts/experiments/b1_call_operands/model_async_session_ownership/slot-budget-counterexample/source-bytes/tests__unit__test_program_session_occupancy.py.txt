"""The original checking request supplies B; occupancy is not frame phase."""
from dataclasses import replace

import pytest

from scar.ir.execution.validity_budget import ValidityBudget
from scar.ir.program.ids import ValueID
from scar.ir.program.model import DataValue, LiteralOrigin, StateVersionValue
from scar.ir.program.protocol_connection_flow import ConnectionContext
from scar.ir.program.protocol_connections import RequestAssociation, SessionOccupancyCondition
from scar.ir.program.protocols import AdmissionPredicate
from scar.ir.program.queries import ProgramIndex
from scar.ir.program.session_occupancy import occupancy_errors, occupancy_view
from scar.ir.program.validation import require_valid, validate
from tests.fixtures.program_async_competition_graphs import async_competition_program


def _core(graph):
    return replace(graph, protocol_scopes=(), protocol_operations=(),
                   protocol_call_inputs=(), protocol_result_consumers=(), protocol_bindings=())


def _context(graph, *, budget=32768):
    return ConnectionContext(graph, index=ProgramIndex(graph), budget=ValidityBudget(budget))


def _flags(view):
    for key in ('source_runtime_verified', 'runtime_identity_verified',
                'value_identity_verified', 'transformation_authorized',
                'native_phase_verified', 'native_admission_verified'):
        assert view[key] is False


@pytest.mark.parametrize('reject_raised', (False, True), ids=('reject-own-normal', 'reject-own-raise'))
def test_occupancy_uses_this_requests_association_body_without_authorizing_a_gate(reject_raised):
    graph, details = async_competition_program(reject_raised=reject_raised)
    require_valid(_core(graph))
    require_valid(graph)
    supplied = details['occupancy']
    row = supplied['row']
    ctx = _context(graph)
    assert occupancy_errors(ctx, row) == ()
    view = occupancy_view(ctx, row)
    assert type(view['subject']) is SessionOccupancyCondition
    assert view['actual'] == supplied['body_use']
    assert view['bool_must_copy'] is True
    assert view['body_binding']['match'] is True
    assert view['binding']['match'] is True
    assert view['expected_bool'] is view['target_compatible_at_snapshot'] is None
    assert view['association_inventory'] == graph.request_association_reads
    expected_predicates = tuple(item for item in graph.protocol_operations
        if type(item.contract) is AdmissionPredicate
        and type(item.contract.subject) is SessionOccupancyCondition)
    assert view['predicate_inventory'] == expected_predicates
    candidate, = tuple(item for item in view['associations']
                       if item['declaration'] == supplied['association'])
    assert candidate['body'] == supplied['association_row'].contract.body
    assert candidate['receiver']['match'] is True
    assert candidate['body_binding']['match'] is True
    assert candidate['projections']
    assert candidate['binding']['match'] is True
    assert len(view['state_support']) == len(supplied['predicate'].accesses)
    assert all(item['binding']['match'] is True for item in view['state_support'])
    assert view['control_routes']['frontiers']
    assert all(item['target_compatible_at_snapshot'] is None
               for item in view['control_routes']['frontiers'])
    assert view['inventory']['known_inventory_complete'] is True
    public = ProgramIndex(graph).protocol_obligations(budget=131072)
    public_rows = tuple(item for scope in public['scopes'] for obligation in scope['obligations']
                        for item in obligation['rows']
                        if item['record'] == row)
    assert len(public_rows) == 1
    assert public_rows[0]['predicate']['body_binding']['match'] is True
    _flags(view)


def test_same_typed_request_reference_cannot_replace_the_association_body():
    graph, details = async_competition_program()
    supplied = details['occupancy']
    association = supplied['association_operation']
    request = next(use for use in association.uses if use.id == supplied['association'].request)
    predicate = supplied['predicate']
    altered = replace(predicate, uses=tuple(replace(use, value=request.value)
        if use.id == supplied['body_use'].use else use for use in predicate.uses))
    graph = replace(graph, operations=tuple(altered if operation.id == altered.id else operation
                                           for operation in graph.operations))
    require_valid(_core(graph))
    issues = validate(graph)
    assert 'async_session_occupancy_body' in {issue.code for issue in issues}
    view = occupancy_view(_context(graph), supplied['row'])
    assert view['body_binding']['match'] is False
    assert view['status'] == 'CONTRADICTED'
    assert view['expected_bool'] is None
    _flags(view)


def test_another_bool_cannot_supply_the_occupancy_branch():
    graph, details = async_competition_program()
    supplied = details['occupancy']
    branch = supplied['branch']
    predicate = supplied['predicate']
    region = next(value.region for value in graph.values if value.id == predicate.results[0])
    other = DataValue(ValueID('occupancy.independent.bool'), region, LiteralOrigin(False), 'builtins.bool')
    altered = replace(branch, uses=tuple(replace(use, value=other.id)
        if use.id == branch.terminator.condition else use for use in branch.uses))
    graph = replace(graph, values=(*graph.values, other), operations=tuple(
        altered if operation.id == altered.id else operation for operation in graph.operations))
    require_valid(_core(graph))
    assert 'admission_bool_source' in {issue.code for issue in validate(graph)}
    view = occupancy_view(_context(graph), supplied['row'])
    assert view['bool_must_copy'] is False
    assert view['status'] == 'CONTRADICTED'
    _flags(view)


def test_removing_both_association_declarations_keeps_the_original_occupancy_read_unresolved():
    graph, details = async_competition_program()
    supplied = details['occupancy']
    original = supplied['association']
    graph = replace(graph,
        request_association_reads=tuple(item for item in graph.request_association_reads if item != original),
        protocol_operations=tuple(item for item in graph.protocol_operations
            if not (item.operation == original.operation and type(item.contract) is RequestAssociation)))
    require_valid(_core(graph))
    view = occupancy_view(_context(graph), supplied['row'])
    assert view['predicate'] == supplied['row']
    assert view['predicate_inventory']
    assert supplied['row'] in view['predicate_inventory']
    assert view['operation'] == supplied['predicate']
    assert view['branch'] == supplied['branch']
    assert view['body_binding']['match'] is None
    assert view['status'] == 'PARTIAL'
    assert 'async_session_occupancy_association_missing' in view['gaps']
    assert view['expected_bool'] is None
    # State and bool consumers remain actual original records, even without a
    # body projection. A future Drive must not reconstruct a missing source.
    values = {value.id: value for value in graph.values}
    assert len(view['state_support']) == len(supplied['predicate'].accesses)
    assert all(isinstance(values[ctx_use.value], StateVersionValue)
               for item in view['state_support']
               for ctx_use in (next(use for use in supplied['predicate'].uses
                                   if use.id == item['actual'].use),))
    _flags(view)


def test_budget_one_keeps_all_original_association_and_predicate_inventory():
    graph, details = async_competition_program()
    require_valid(graph)
    supplied = details['occupancy']
    view = occupancy_view(_context(graph, budget=1), supplied['row'])
    assert view['predicate'] == supplied['row']
    assert view['association_inventory'] == graph.request_association_reads
    assert supplied['row'] in view['predicate_inventory']
    assert view['association_declarations']
    assert view['inventory']['associations_total'] == len(view['association_declarations'])
    assert view['inventory']['known_inventory_complete'] is False
    assert view['body_binding']['match'] is view['binding']['match'] is None
    assert view['expected_bool'] is None
    assert view['budget']['truncated'] is True and view['cursor']
    assert 'protocol_budget_exhausted' in view['gaps']
    _flags(view)
