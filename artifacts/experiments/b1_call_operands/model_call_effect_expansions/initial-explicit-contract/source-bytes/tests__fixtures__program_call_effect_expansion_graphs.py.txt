"""Two calls to one effectful implementation, with a suffix/prefix interval.

Both whole-call summaries retain their true RW effects. Boundary work is in
the original implementation CFG; the independent expansion rows say that this
CFG exhausts the call, without claiming purity or native correspondence.
"""
from dataclasses import replace

from scar.ir.program.call_effect_expansion import CallEffectExpansion
from scar.ir.program.ids import BlockID, OperationID
from scar.ir.program.model import (
    AccessMode, Behavior, Block, CompletionResult, CompletionSignature,
    CompletionSuccessor, ExitKind, Footprint, FootprintKind, Operation,
    OperationResult, StateAccess, StateVersionValue, Successor, Terminator,
    TerminatorKind, ValueKind, ValueType,
)
from scar.ir.program.state_addresses import InvokeStateOperand
from tests.fixtures.program_driver_graphs import PACKET, REFERENCE, _Builder
from tests.fixtures.program_graphs import shared_body_calls_program
from tests.unit.test_execution_state_preservation import (
    _formals, _preservation_program, _terminal, _uses,
)


def call_effect_expansion_program(*, outer_complete=True, expanded=True,
                                  interval_write=False, opaque=False):
    """Original N/R fullslots, two state domains, and one shared getter body."""
    graph, original = _preservation_program(outer_complete=outer_complete,
        write_location='same', write_mode=AccessMode.READ_WRITE)
    parent_region = next(row for row in graph.regions if row.id == graph.root)
    child_region = next(row for row in graph.regions if row.id == original['call'].callee)
    domains = original['locations']
    states = tuple(ValueType(ValueKind.STATE, location=location) for location in domains)
    boolean = ValueType(ValueKind.DATA, data_type='builtins.bool')
    values = {row.id: row for row in graph.values}
    child_entry = next(block for block in child_region.blocks if block.id == child_region.entry)
    child = _Builder(child_region.id.value, domains)
    child.values = [row for row in graph.values if row.region == child_region.id]
    prefix_uses = _uses(child, 'expansion.prefix.read.use', child_entry.arguments[2:])
    prefix = Operation(OperationID('expansion.prefix.read'), 'explicit implementation entry READ',
        (Behavior.READ,), prefix_uses,
        accesses=tuple(StateAccess(location, use.id, AccessMode.READ)
                       for location, use in zip(domains, prefix_uses)), effects_complete=True)
    child_region = replace(child_region, blocks=tuple(replace(block,
        operations=(prefix.id, *block.operations)) if block.id == child_region.entry else block
        for block in child_region.blocks))

    parent = _Builder(parent_region.id.value, domains)
    parent.values = [row for row in graph.values if row.region == parent_region.id]
    parent.operations = [row for row in graph.operations
                         if any(row.id in block.operations for block in parent_region.blocks)]
    parent_entry = next(block for block in parent_region.blocks if block.id == parent_region.entry)
    joined = _formals(parent, 'expansion.between', (REFERENCE, boolean, *states))
    blocks, joins = [], {}
    for block in parent_region.blocks:
        terminal = next((op for op in original['terminals'].values() if op.id in block.operations), None)
        if terminal is None:
            blocks.append(block)
            continue
        current = {values[value].location: value for value in block.arguments
                   if type(values[value]) is StateVersionValue}
        forwarded = _uses(parent, terminal.id.value + '.continue',
            (block.arguments[0], parent_entry.arguments[1], *(current[location] for location in domains)))
        # Keep an exact consumer for the original first-call payload as well
        # as every full state slot; only the continuation tuple is forwarded.
        old_payloads = tuple(use for use in terminal.uses
                             if type(values[use.value]) is not StateVersionValue)
        jump = replace(terminal, uses=(*old_payloads, *forwarded),
            terminator=Terminator(TerminatorKind.JUMP, (
                Successor(BlockID('expansion.between'), tuple(use.id for use in forwarded)),)))
        parent.operations = [jump if op.id == terminal.id else op for op in parent.operations]
        blocks.append(block)
        joins[block.id] = jump

    current = list(joined[2:])
    middle, writer, primitive = [], None, None
    if interval_write:
        identity = OperationID('expansion.independent.write')
        before, = _uses(parent, identity.value + '.before', (current[1],))
        after = parent.value(identity.value + '.after', OperationResult(identity, 0), states[1])
        writer = Operation(identity, 'independent context write, no source Reference operand',
            (Behavior.WRITE,), (before,), (after,),
            (StateAccess(domains[1], before.id, AccessMode.WRITE, after,
                         Footprint(FootprintKind.WHOLE)),), effects_complete=True)
        parent.operations.append(writer)
        middle.append(writer)
        current[1] = after

    second_name, second_arguments = 'expansion.between', (joined[0], joined[1], *current)
    if opaque:
        actuals = _uses(parent, 'expansion.primitive.argument', (joined[0], *current))
        forward_bool, = _uses(parent, 'expansion.primitive.forward.bool', (joined[1],))
        normal = _formals(parent, 'expansion.primitive.normal', (REFERENCE, boolean, *states))
        failure = _formals(parent, 'expansion.primitive.failure', (PACKET, *states))
        primitive = parent.add('expansion.primitive', (*actuals, forward_bool),
            effects_complete=False, invoke_arguments=tuple(use.id for use in actuals),
            completions=(CompletionSignature(ExitKind.NORMAL, (REFERENCE, *states)),
                         CompletionSignature(ExitKind.RAISE, (PACKET, *states))),
            terminator=Terminator(TerminatorKind.INVOKE, (
                CompletionSuccessor(ExitKind.NORMAL, BlockID('expansion.primitive.normal'),
                    (CompletionResult(0), forward_bool.id, CompletionResult(1), CompletionResult(2))),
                CompletionSuccessor(ExitKind.RAISE, BlockID('expansion.primitive.failure'),
                    tuple(CompletionResult(slot) for slot in range(3))))))
        middle.append(primitive)
        parent.block('expansion.between', joined, middle)
        failed = _terminal(parent, 'expansion.primitive.raise', ExitKind.RAISE,
                           (failure[2], failure[0], failure[1]))
        parent.block('expansion.primitive.failure', failure, (failed,))
        second_name, second_arguments, middle = 'expansion.primitive.normal', normal, []

    normal = _formals(parent, 'expansion.second.normal', (REFERENCE, states[1], REFERENCE, states[0]))
    raised = _formals(parent, 'expansion.second.raise', (REFERENCE, states[0], PACKET, states[1]))
    actuals = _uses(parent, 'expansion.second.argument', second_arguments)
    forward_normal, = _uses(parent, 'expansion.second.normal.forward', (second_arguments[0],))
    forward_raise, = _uses(parent, 'expansion.second.raise.forward', (second_arguments[0],))
    first = original['call']
    second = parent.add('expansion.second.call', (*actuals, forward_normal, forward_raise),
        effects_complete=outer_complete, callee=first.callee,
        invoke_arguments=tuple(use.id for use in actuals), completions=first.completions,
        terminator=Terminator(TerminatorKind.INVOKE, (
            CompletionSuccessor(ExitKind.RAISE, BlockID('expansion.second.raise'),
                (forward_raise.id, CompletionResult(2), CompletionResult(1), CompletionResult(0))),
            CompletionSuccessor(ExitKind.NORMAL, BlockID('expansion.second.normal'),
                (forward_normal.id, CompletionResult(2), CompletionResult(0), CompletionResult(1))))))
    parent.block(second_name, joined if not opaque else second_arguments, (*middle, second))
    final_normal = _terminal(parent, 'expansion.second.return', ExitKind.NORMAL,
                             (normal[2], normal[3], normal[1]))
    final_raise = _terminal(parent, 'expansion.second.raise.exit', ExitKind.RAISE,
                            (raised[3], raised[2], raised[1]))
    parent.block('expansion.second.normal', normal, (final_normal,))
    parent.block('expansion.second.raise', raised, (final_raise,))
    parent_region = replace(parent_region, blocks=(*blocks, *parent.blocks))
    refinements = (*graph.state_address_refinements,
        *(replace(row, site=InvokeStateOperand(second.id, actuals[ordinal + 2].id))
          for ordinal, row in enumerate(graph.state_address_refinements)))
    expansions = tuple(CallEffectExpansion(call.id, call.callee) for call in (first, second))
    graph = replace(graph, regions=(parent_region, child_region),
        operations=(*parent.operations,
            *(op for op in graph.operations if op.id not in {row.id for row in parent.operations}), prefix),
        values=(*parent.values, *(row for row in graph.values if row.region == child_region.id)),
        control_scopes=tuple(replace(scope, blocks=tuple(block.id for block in (
            parent_region.blocks if scope.region == parent_region.id else child_region.blocks)))
            for scope in graph.control_scopes),
        state_address_refinements=refinements,
        call_effect_expansions=expansions if expanded else ())
    paths = {}
    for kind, suffix in ((ExitKind.NORMAL, 'normal'), (ExitKind.RAISE, 'raise')):
        selected = 1 if kind is ExitKind.NORMAL else 0
        paths[kind] = ((parent_region.entry, selected),
            (BlockID('preserve.parent.' + suffix), 0),
            (BlockID('expansion.between'), 0 if opaque else selected),
            *((((BlockID('expansion.primitive.normal'), selected),)) if opaque else ()),
            (BlockID('expansion.second.' + suffix), None))
    return graph, {'locations': domains, 'world': original['world'], 'context': original['context'],
        'parent': parent_region, 'child': child_region, 'calls': (first, second),
        'expansions': expansions, 'source': original['writer'], 'target': original['getter'],
        'prefix': prefix, 'branch': original['branch'], 'pack': original['pack'],
        'child_terminals': original['child_terminals'], 'caller_paths': paths,
        'joins': joins, 'writer': writer, 'primitive': primitive,
        'second_terminals': {ExitKind.NORMAL: final_normal, ExitKind.RAISE: final_raise}}


def ordinary_call_effect_expansion_program():
    """Keep two original ordinary CALLs and their data/state result ordinals."""
    graph = shared_body_calls_program()
    old = next(op for op in graph.operations if op.id == OperationID('callee.done'))
    terminal = replace(old, id=OperationID('expansion.ordinary.return'),
        label='explicit ordinary CONTROL return', effects_complete=True)
    calls = tuple(op for op in graph.operations if op.callee is not None)
    graph = replace(graph,
        operations=tuple(terminal if op.id == old.id else op for op in graph.operations),
        regions=tuple(replace(region, blocks=tuple(replace(block,
            operations=tuple(terminal.id if op == old.id else op for op in block.operations))
            for block in region.blocks)) if region.id == calls[0].callee
            else region for region in graph.regions),
        call_effect_expansions=tuple(CallEffectExpansion(call.id, call.callee) for call in calls))
    source = next(op for op in graph.operations if op.id == OperationID('count.call'))
    compute = next(op for op in graph.operations if op.id == OperationID('twice'))
    return graph, {'calls': calls, 'source': source, 'target': source,
        'prefix': compute, 'terminal': terminal,
        'locations': tuple(location.id for location in graph.locations)}


__all__ = ['call_effect_expansion_program', 'ordinary_call_effect_expansion_program']
