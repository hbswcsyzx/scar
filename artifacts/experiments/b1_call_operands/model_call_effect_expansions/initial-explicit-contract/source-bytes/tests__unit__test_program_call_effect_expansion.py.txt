"""A full-effect expansion is an original call relation, not a purity flag."""
from dataclasses import FrozenInstanceError, fields, replace

import pytest

from scar.ir.program.call_effect_expansion import (
    CALL_EFFECT_EXPANSION_RECORD_TAGS, CallEffectExpansion, call_effect_expansion_errors,
)
from scar.ir.program.ids import OperationID, RegionID
from scar.ir.program.model import AccessMode, ProgramGraph, TerminatorKind
from scar.ir.program.queries import ProgramIndex
from scar.ir.program.validation import require_valid, validate
from tests.fixtures.program_call_return_graphs import call_return_program
from tests.unit.test_execution_state_preservation import _preservation_program


def _program(kind='invoke'):
    if kind == 'invoke':
        graph, details = _preservation_program(outer_complete=False)
        operation = details['call']
    else:
        graph, details = call_return_program()
        operation = next(item for item in graph.operations if item.id == details['root_calls'][0])
        operation = replace(operation, effects_complete=False)
        graph = replace(graph, operations=tuple(operation if item.id == operation.id else item
                                               for item in graph.operations))
    require_valid(graph)
    return graph, operation


def _attach(graph, operation):
    row = CallEffectExpansion(operation.id, operation.callee)
    return replace(graph, call_effect_expansions=(row,)), row


def test_record_is_exact_frozen_and_has_no_purity_or_authority_fields():
    operation, callee = OperationID('expansion.call'), RegionID('expansion.body')
    row = CallEffectExpansion(operation, callee)
    assert tuple(field.name for field in fields(row)) == ('operation', 'callee')
    assert row.operation is operation and row.callee is callee
    assert not hasattr(row, '__dict__')
    with pytest.raises(FrozenInstanceError):
        row.callee = RegionID('another.body')
    for bad_operation in (callee, operation.wire):
        with pytest.raises(TypeError, match='exact OperationID'):
            CallEffectExpansion(bad_operation, callee)
    for bad_callee in (operation, callee.wire):
        with pytest.raises(TypeError, match='exact RegionID'):
            CallEffectExpansion(operation, bad_callee)

    class OtherOperationID(OperationID):
        pass

    with pytest.raises(TypeError, match='exact OperationID'):
        CallEffectExpansion(OtherOperationID(operation.value), callee)
    assert dict(CALL_EFFECT_EXPANSION_RECORD_TAGS) == {
        CallEffectExpansion: ('kind', 'CALL_EFFECT_EXPANSION')}
    with pytest.raises(TypeError):
        CALL_EFFECT_EXPANSION_RECORD_TAGS[CallEffectExpansion] = ('kind', 'OTHER')


@pytest.mark.parametrize('kind', ('call', 'invoke'))
def test_explicit_expansion_is_read_as_the_original_row_without_rewriting_the_parent_summary(kind):
    graph, operation = _program(kind)
    original_operations = graph.operations
    graph, row = _attach(graph, operation)
    require_valid(graph)
    assert call_effect_expansion_errors(graph) == ()
    assert graph.operations == original_operations
    current = next(item for item in graph.operations if item.id == operation.id)
    assert current is operation and current.effects_complete is False
    assert (current.terminator is not None) is (kind == 'invoke')
    if kind == 'invoke':
        assert current.terminator.kind is TerminatorKind.INVOKE
    assert ProgramIndex(graph).call_effect_expansion(operation.id) is row


@pytest.mark.parametrize('fault,code', (
    ('missing-operation', 'call_effect_expansion_operation'),
    ('missing-region', 'call_effect_expansion_region'),
    ('different-callee', 'call_effect_expansion_callee'),
    ('primitive', 'call_effect_expansion_callee'),
))
def test_new_relation_rejects_wrong_original_endpoints_on_an_otherwise_valid_graph(fault, code):
    graph, operation = _program()
    original_callee = operation.callee
    if fault == 'primitive':
        operation = replace(operation, callee=None)
        graph = replace(graph, operations=tuple(operation if item.id == operation.id else item
                                               for item in graph.operations))
    require_valid(graph)
    row = CallEffectExpansion(
        OperationID('expansion.missing.operation') if fault == 'missing-operation' else operation.id,
        RegionID('expansion.missing.region') if fault == 'missing-region' else
        graph.root if fault == 'different-callee' else original_callee)
    graph = replace(graph, call_effect_expansions=(row,))
    issues = call_effect_expansion_errors(graph)
    assert code in {issue[0] for issue in issues}
    assert all(path == str(row.operation) and message for _, path, message in issues)
    assert code in {issue.code for issue in validate(graph)}
    with pytest.raises(ValueError, match=code):
        ProgramIndex(graph)


def test_graph_constructor_requires_exact_records_tuple_and_unique_operation_key():
    graph, operation = _program()
    row = CallEffectExpansion(operation.id, operation.callee)
    with pytest.raises(TypeError, match='tuple'):
        replace(graph, call_effect_expansions=[row])
    with pytest.raises(TypeError, match='exact CallEffectExpansion'):
        replace(graph, call_effect_expansions=(operation,))
    with pytest.raises(ValueError, match='call effect expansion operations'):
        replace(graph, call_effect_expansions=(row, row))
    with pytest.raises(ValueError, match='call effect expansion operations'):
        replace(graph, call_effect_expansions=(row, replace(row, callee=graph.root)))


@pytest.mark.parametrize('inside', ('write', 'opaque'))
def test_complete_expansion_can_expose_real_writes_and_unknown_implementation_effects(inside):
    graph, details = _preservation_program(outer_complete=False,
        write_location='same' if inside == 'write' else None,
        child_complete=inside != 'opaque')
    require_valid(graph)
    operation = details['call']
    graph, row = _attach(graph, operation)
    require_valid(graph)
    assert call_effect_expansion_errors(graph) == ()
    assert ProgramIndex(graph).call_effect_expansion(operation.id) is row
    assert operation.effects_complete is False
    if inside == 'write':
        writer = details['writer']
        assert writer in graph.operations
        assert writer.accesses and writer.accesses[0].mode in (AccessMode.WRITE, AccessMode.READ_WRITE)
    else:
        assert details['getter'] in graph.operations
        assert details['getter'].effects_complete is False
    assert tuple(field.name for field in fields(row)) == ('operation', 'callee')


def test_known_callee_and_identical_source_data_do_not_create_an_implicit_effect_relation():
    graph, operation = _program()
    index = ProgramIndex(graph)
    assert operation.callee is not None
    assert graph.call_effect_expansions == ()
    assert index.call_effect_expansion(operation.id) is None
    graph, row = _attach(graph, operation)
    assert ProgramIndex(graph).call_effect_expansion(operation.id) is row
    removed = replace(graph, call_effect_expansions=())
    assert removed.operations == graph.operations
    assert removed.values == graph.values
    assert ProgramIndex(removed).call_effect_expansion(operation.id) is None
    with pytest.raises(KeyError):
        ProgramIndex(removed).call_effect_expansion(OperationID('expansion.unknown.call'))


def test_valid_program_graph_subclass_retains_public_expansion_validation_and_read():
    graph, operation = _program()

    class Derived(ProgramGraph):
        pass

    derived = Derived(**{field.name: getattr(graph, field.name) for field in fields(graph)})
    # Establish the original subclass/core contract before adding the new row.
    assert validate(derived) == ()
    require_valid(derived)
    assert ProgramIndex(derived).call_effect_expansion(operation.id) is None
    expanded, row = _attach(derived, operation)
    assert type(expanded) is Derived
    assert validate(expanded) == ()
    require_valid(expanded)
    assert ProgramIndex(expanded).call_effect_expansion(operation.id) is row
    assert expanded.operations == derived.operations
