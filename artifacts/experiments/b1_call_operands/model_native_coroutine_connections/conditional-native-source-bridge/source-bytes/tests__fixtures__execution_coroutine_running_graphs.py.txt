"""One supplied native coroutine reenters itself during its second real Drive.

The body handle travels through the caller's SEND operand, an expanded known
normalizer, its own NORMAL result, and the original SENT_VALUE ingress.  No
holder read, wrapper, second Create, or runtime observation is manufactured.
"""

def _operation(graph, identity):
    return next(row for row in graph.operations if row.id == identity)


def _final_region(graph, details):
    """Graft only the real E/F region, preserving the single-Create caller."""
    template, implementation = await_final_region_program(zero_cut=False, depth=1)
    parent, original = details['extract'], implementation['original_extract']
    assert (parent.uses, parent.completions, parent.terminator) == (
        original.uses, original.completions, original.terminator)
    regions = {row.id for row in implementation['stage_regions']}
    operations = {identity for row in implementation['stage_regions']
                  for block in row.blocks for identity in block.operations}
    parent = replace(parent, callee=implementation['parent_call'].callee)
    rows = {(row.scope, row.operation, row.slot): row for row in graph.protocol_operations}
    receipts = tuple(replace(receipt, site=replace(receipt.site,
        clause=rows[receipt.site.scope, receipt.site.operation, receipt.site.slot].clause))
        for receipt in implementation['conversion_receipts'])
    graph = replace(graph,
        regions=(*graph.regions, *(row for row in template.regions if row.id in regions)),
        operations=tuple(parent if row.id == parent.id else row for row in graph.operations)
            + tuple(row for row in template.operations if row.id in operations),
        values=(*graph.values, *(row for row in template.values if row.region in regions)),
        control_scopes=(*graph.control_scopes,
            *(row for row in template.control_scopes if row.region in regions)),
        protocol_result_consumers=(*graph.protocol_result_consumers, *receipts))
    return graph, {**details, 'extract': parent, 'parent_call': parent,
        'final_rows': details['adapter_rows'],
        'stage_records': implementation['stage_records'],
        'stage_regions': implementation['stage_regions'],
        'primitive': implementation['primitive']}


def _expanded_send(graph, details):
    """Expand the actual CHECK_SEND boundary's ordered operand and N/R exits."""
    driver = details['drivers'][1]
    region = driver['driver_region']
    entry = next(row for row in region.blocks if row.id == region.entry)
    values = {row.id: row for row in graph.values}
    scope_ids = {row.id for row in graph.protocol_scopes if type(row.subject) is DriverSubject
                 and row.subject.interface_region == region.id}
    declarations = tuple(row for row in graph.protocol_operations if row.scope in scope_ids
        and type(row.contract) is RequestProcessing
        and row.contract.rule is RequestProcessingRule.CHECK_SEND)
    assert len({row.operation for row in declarations}) == 1
    processing = _operation(graph, declarations[0].operation)
    interface = next(row for row in graph.driver_interfaces if row.region == region.id)
    operand_value, = tuple(identity for identity in entry.arguments
        if identity not in (interface.receiver, interface.scheduler_request)
        and value_type(values[identity]) == REFERENCE)
    operand = _use('native.running.send.operand', operand_value)
    processing = replace(processing, uses=(*processing.uses, operand),
        invoke_arguments=(*processing.invoke_arguments, operand.id))
    actuals = {use.id: use for use in processing.uses}
    input_types = tuple(value_type(values[actuals[identity].value])
                        for identity in processing.invoke_arguments)
    domains = tuple(type_.location for type_ in input_types
                    if isinstance(type_, ValueType) and type_.kind is ValueKind.STATE)
    state_types = tuple(ValueType(ValueKind.STATE, location=location) for location in domains)
    builder = _Builder('native.running.send.implementation', domains)
    formals = builder.formals('native.running.send.entry', input_types)
    raw = formals[processing.invoke_arguments.index(declarations[0].contract.raw_use)]
    forwarded_operand = formals[-1]
    states = tuple(formal for formal, type_ in zip(formals, input_types)
                   if isinstance(type_, ValueType) and type_.kind is ValueKind.STATE)
    predicate_id = OperationID('native.running.send.check')
    boolean = DataValue(ValueID('native.running.send.accepted'), builder.region,
                        OperationResult(predicate_id, 0), 'builtins.bool')
    builder.values.append(boolean)
    state_uses = tuple(_use(f'native.running.send.check.state.{slot}', state, state=True)
                       for slot, state in enumerate(states))
    check = Operation(predicate_id, 'declared request check implementation',
        (Behavior.COMPUTE, Behavior.READ),
        (_use('native.running.send.check.raw', raw, protocol=True),
         _use('native.running.send.check.operand', forwarded_operand), *state_uses),
        (boolean.id,), tuple(StateAccess(location, use.id, AccessMode.READ)
            for location, use in zip(domains, state_uses)), effects_complete=True)
    builder.operations.append(check)
    branch_uses = [Use(UseID('native.running.send.condition'), boolean.id, UseRole.DATA)]
    edges, terminals = [], {}
    for number, kind in enumerate((ExitKind.NORMAL, ExitKind.RAISE)):
        name = 'native.running.send.' + kind.value.lower()
        landing = builder.formals(name, (REFERENCE, *state_types))
        forward = tuple(_use(f'{name}.forward.{slot}', value, state=slot > 0)
                        for slot, value in enumerate((forwarded_operand, *states)))
        branch_uses.extend(forward)
        edges.append(Successor(BlockID(name), tuple(use.id for use in forward)))
        prefix = []
        payload = landing[0]
        if kind is ExitKind.RAISE:
            exception = builder.value(name + '.exception',
                OperationResult(OperationID(name + '.construct'), 0), REFERENCE)
            construct = builder.add(name + '.construct', results=(exception,), effects_complete=False)
            packet = builder.value(name + '.packet', OperationResult(OperationID(name + '.pack'), 0), PACKET)
            components = (_use(name + '.exception.use', exception),
                          _use(name + '.traceback.use', landing[0]))
            pack = builder.add(name + '.pack', components, results=(packet,), effects_complete=True)
            prefix.extend((construct, pack))
            packet_definition = ExceptionPacketDefinition(packet, components[0].id, components[1].id)
            payload = packet
        terminal = builder.terminal(name + '.exit', (payload, *landing[1:]), fail=kind is ExitKind.RAISE)
        terminal = replace(terminal, effects_complete=True)
        builder.operations[-1] = terminal
        builder.block(name, landing, (*prefix, terminal))
        terminals[kind] = terminal
    branch = builder.add('native.running.send.branch', branch_uses, effects_complete=True,
        terminator=Terminator(TerminatorKind.BRANCH, tuple(edges), condition=branch_uses[0].id))
    builder.block('native.running.send.entry', formals, (check, branch))
    child = builder.finish('native.running.send.entry', parent=region.id)
    processing = replace(processing, callee=child.id)
    changed = {row: replace(row, contract=replace(row.contract,
        operand_bindings=(RequestOperandBinding(0, operand.id),))) for row in declarations}
    graph = replace(graph, regions=(*graph.regions, child),
        operations=tuple(processing if row.id == processing.id else row for row in graph.operations)
            + tuple(builder.operations), values=(*graph.values, *builder.values),
        control_scopes=(*graph.control_scopes, _scope(child.id, child.blocks)),
        exception_packets=(*graph.exception_packets, packet_definition),
        protocol_operations=tuple(changed.get(row, row) for row in graph.protocol_operations))
    # Both the real raw operand and the separately supplied boundary actual
    # consume the current independently forwarded Create handle.
    caller = driver['caller_call']
    own = {use.id: use for use in caller.uses}
    owner = own[caller.invoke_arguments[0]].value
    raw_value = own[caller.invoke_arguments[1]].value
    definition = next(row for row in graph.raw_driver_requests if row.value == raw_value)
    raw_operation = next(value.origin.operation for value in graph.values if value.id == raw_value)
    raw_pack = _operation(graph, raw_operation)
    assert definition.kind is RawRequestKind.SEND and len(definition.operands) == 1
    raw_pack = replace(raw_pack, uses=tuple(replace(use, value=owner)
        if use.id == definition.operands[0] else use for use in raw_pack.uses))
    caller = replace(caller, uses=tuple(replace(use, value=owner)
        if use.id == caller.invoke_arguments[2] else use for use in caller.uses))
    graph = replace(graph, operations=tuple(raw_pack if row.id == raw_pack.id else
        caller if row.id == caller.id else row for row in graph.operations))
    drivers = tuple({**item, 'caller_call': caller} if number == 1 else item
                    for number, item in enumerate(details['drivers']))
    calls = tuple(caller if row.id == caller.id else row for row in details['calls'])
    normal = next(row.contract.result for row in changed.values()
        if processing.terminator.successors[row.slot].kind is ExitKind.NORMAL)
    metadata = {'send_call': caller, 'raw': definition,
        'raw_operand': OwnedUse(raw_pack.id, definition.operands[0]),
        'normalizer': processing, 'normalizer_operand': OwnedUse(processing.id, operand.id),
        'normalizer_result': normal, 'normalizer_region': child,
        'normalizer_formals': formals, 'normalizer_branch': branch,
        'normalizer_terminal': terminals[ExitKind.NORMAL],
        'normalizer_path': ((child.entry, 0), (BlockID('native.running.send.normal'), None))}
    return graph, {**details, 'drivers': drivers, 'calls': calls}, metadata


def _inner_driver():
    """Keep the original admitted CFG, but supply the real true/reject path."""
    from scar.ir.program.resumption import ResumableKind, SuspendTerminator, SuspensionFlavor
    body = _body(zero_cut=False, kind=ResumableKind.COROUTINE)
    body = replace(body, operations=tuple(replace(row,
        terminator=replace(row.terminator, flavor=SuspensionFlavor.AWAIT))
        if type(row.terminator) is SuspendTerminator else row for row in body.operations))
    graph = driver_program(target='start', body_graph=body)
    interface, = graph.driver_interfaces
    identity = ProtocolScopeID('native.running.inner.scope')
    graph = replace(graph, protocol_scopes=(ProtocolScope(identity,
        ProtocolProfile.COROUTINE_BODY_V1, DriverSubject(interface.region), _source_gap()),))
    from scar.ir.program.protocols import BodyStatePredicateRule
    driver = next(row for row in graph.regions if row.id == interface.region)
    graph, guard = _guard(graph, block_id=driver.entry,
        owner=interface.receiver, raw=interface.scheduler_request, label='native.running.inner',
        rule=BodyStatePredicateRule.IS_RUNNING, admit_true=False, request_kind=RawRequestKind.NEXT)
    graph = with_publications(graph)
    part, maps = _driver_part(graph, 'native.running.inner.driver')
    predicate = next(row for row in part['protocol_operations']
                     if row.operation == maps[OperationID][guard['predicate']])
    branch = next(row for row in part['operations'] if row.id == maps[OperationID][guard['branch']])
    reject = next(row for row in part['operations'] if row.id == maps[OperationID][guard['helper']])
    terminal = next(row for row in part['operations'] if row.id == maps[OperationID][guard['terminals'][0]])
    choices = {branch.id: 0, reject.id: 0}
    return part, {'predicate': predicate, 'branch': branch, 'reject': reject,
        'host_terminal': terminal, 'path': _selected_path(
            replace(graph, regions=(part['region'],), operations=part['operations']), part['region'], choices)}


def _insert_reentry(graph, details, inner, guard):
    # The old factory's metadata keeps the pre-composition Region object.
    # Graft the actual current Region, including its existing lexical parent.
    body = next(row for row in graph.regions if row.id == details['body'].id)
    values = {row.id: row for row in graph.values}
    first = _operation(graph, details['drivers'][0]['outcome_source'])
    target = details['drivers'][1]['drive'].target
    ingress = first.terminator.successors[target.successor]
    slot, = tuple(number for number, item in enumerate(ingress.inputs)
                  if item.kind is ResumeSlotKind.SENT_VALUE)
    from scar.ir.program.resumption import ResumeInput
    block = next(row for row in body.blocks if row.id == ingress.block)
    position, = tuple(number for number, actual in enumerate(ingress.arguments)
                     if type(actual) is ResumeInput and actual.resume_slot == slot)
    sent = values[block.arguments[position]]
    attach = next(row for row in graph.frame_context_attaches
                  if row.operation in block.operations)
    attached = _operation(graph, attach.operation)
    context = next(values[value] for value in attached.results if type(values[value]) is StateVersionValue)
    write = _operation(graph, OperationID('resume.value.write'))
    world = values[write.results[0]]
    domains = tuple(location.id for location in graph.locations)
    state_types = tuple(ValueType(ValueKind.STATE, location=location) for location in domains)
    builder = _Builder(body.id.value, domains)
    raw = builder.value('native.running.inner.raw.value',
        OperationResult(OperationID('native.running.inner.raw'), 0), RAW)
    pack = builder.add('native.running.inner.raw', results=(raw,), effects_complete=True)
    actuals = (_use('native.running.inner.call.receiver', sent.id),
        _use('native.running.inner.call.raw', raw, protocol=True),
        _use('native.running.inner.call.world', world.id, state=True),
        _use('native.running.inner.call.context', context.id, state=True))
    forwards, edges, handlers, installs = [], [], [], []
    join_name = 'native.running.body.continue'
    joined = builder.formals(join_name,
        (ProtocolValueType(P.DRIVE_CALLER_CONTEXT), *state_types, REFERENCE, REFERENCE))
    for kind in (ExitKind.NORMAL, ExitKind.RAISE):
        port = next(row for row in inner['region'].exits if row.kind is kind)
        name = 'native.running.body.' + kind.value.lower()
        landing = builder.formals(name,
            (*port.types, ProtocolValueType(P.DRIVE_CALLER_CONTEXT), REFERENCE))
        copied = (_use(name + '.caller', attach.caller, protocol=True),
                  _use(name + '.saved', ValueID('resume.value.x')))
        forwards.extend(copied)
        edges.append(CompletionSuccessor(kind, BlockID(name),
            (*(CompletionResult(index) for index in range(len(port.types))),
             *(use.id for use in copied))))
        prefix, current_context = [], landing[2]
        incoming = _use(name + '.result', landing[0], protocol=kind is ExitKind.RAISE)
        if kind is ExitKind.RAISE:
            install_id = OperationID(name + '.context')
            current_context = builder.value(name + '.context.after', OperationResult(install_id, 0), state_types[1])
            uses = (_use(name + '.caught.packet', landing[0], protocol=True),
                    _use(name + '.context.before', landing[2], state=True))
            install = Operation(install_id, 'original caught-context update',
                (Behavior.CONTROL, Behavior.READ, Behavior.WRITE), uses, (current_context,),
                (StateAccess(domains[1], uses[1].id, AccessMode.READ_WRITE, current_context),),
                effects_complete=True)
            builder.operations.append(install)
            prefix.append(install)
            installs.append(install)
        continued = builder.value(name + '.continued', OperationResult(OperationID(name + '.handle'), 0), REFERENCE)
        handler = builder.add(name + '.handle', (incoming,), results=(continued,), effects_complete=True)
        outgoing = tuple(_use(name + f'.join.{number}', value,
            state=number in (1, 2), protocol=number == 0)
            for number, value in enumerate((landing[-2], landing[1], current_context, landing[-1], continued)))
        jump = builder.add(name + '.jump', outgoing, effects_complete=True,
            terminator=Terminator(TerminatorKind.JUMP,
                (Successor(BlockID(join_name), tuple(use.id for use in outgoing)),)))
        builder.block(name, landing, (*prefix, handler, jump))
        handlers.append(handler)
    call = builder.add('native.running.body.call', (*actuals, *forwards),
        callee=inner['region'].id, effects_complete=False,
        invoke_arguments=tuple(use.id for use in actuals),
        completions=tuple(CompletionSignature(port.kind, port.types) for port in inner['region'].exits),
        terminator=Terminator(TerminatorKind.INVOKE, tuple(edges)))
    detach = next(row for row in graph.frame_context_detaches if row.operation in block.operations)
    detach_op = _operation(graph, detach.operation)
    second = _operation(graph, details['drivers'][1]['outcome_source'])
    substitutions = {attach.caller: joined[0], world.id: joined[1], context.id: joined[2],
                     ValueID('resume.value.x'): joined[3]}
    changed = {row.id: replace(row, uses=tuple(replace(use,
        value=joined[4] if row.id == second.id and use.id == second.terminator.exported
              else substitutions.get(use.value, use.value)) for use in row.uses))
        for row in (detach_op, second)}
    kept = tuple(identity for identity in block.operations if identity not in changed)
    blocks = tuple(replace(row, operations=(*kept, pack.id, call.id)) if row.id == block.id else row
                   for row in body.blocks)
    join = Block(BlockID(join_name), joined, (detach_op.id, second.id))
    body = replace(body, blocks=(*blocks, *builder.blocks, join))
    tables = ('driver_interfaces', 'final_body_gates', 'body_drives', 'driver_lifecycles',
              'body_outcome_commits', 'terminal_frame_releases', 'protocol_scopes',
              'protocol_operations', 'literal_evaluations', 'exception_packets')
    graph = replace(graph,
        regions=tuple(body if row.id == body.id else row for row in graph.regions)
            + (replace(inner['region'], parent=body.id),),
        operations=tuple(changed.get(row.id, row) for row in graph.operations)
            + tuple(builder.operations) + inner['operations'],
        values=(*graph.values, *builder.values, *inner['values']),
        control_scopes=tuple(replace(row, blocks=tuple(item.id for item in body.blocks))
            if row.region == body.id else row for row in graph.control_scopes) + inner['control_scopes'],
        raw_driver_requests=(*graph.raw_driver_requests, RawRequestDefinition(raw, RawRequestKind.NEXT, ())),
        **{name: (*getattr(graph, name), *inner[name]) for name in tables})
    raise_index = next(number for number, edge in enumerate(call.terminator.successors)
                       if edge.kind is ExitKind.RAISE)
    body_path = ((block.id, raise_index),
                 (call.terminator.successors[raise_index].block, 0), (join.id, None))
    running = {**guard, 'inner_driver': next(row for row in graph.regions if row.id == inner['region'].id),
        'inner_call': call, 'inner_receiver': OwnedUse(call.id, actuals[0].id),
        'resume_formal': sent, 'resume_slot': slot, 'active_drive': details['drivers'][1]['drive'],
        'handler': handlers[1], 'context_access': installs[0], 'join': join,
        'second_suspend': changed[second.id], 'body_path': body_path}
    return graph, {**details, 'body': body}, running


def coroutine_running_program():
    graph, details = native_resume_program(zero_cut=False, extra_creation=False)
    graph, details = _final_region(graph, details)
    graph, details, send = _expanded_send(graph, details)
    inner, guard = _inner_driver()
    graph, details, running = _insert_reentry(graph, details, inner, guard)
    return graph, {**details, 'running': {**send, **running}}


def coroutine_running_execution(*, instances=1, missing_inner_call=None, missing_sent_use=None):
    graph, details = coroutine_running_program()
    running = details['running']
    stage, = details['stage_records']
    calls = (
        {'label': 'send', 'parent': ('driver', 1), 'caller': running['normalizer'].id,
         'region': running['normalizer_region'].id, 'path': running['normalizer_path'],
         'completion': running['normalizer_terminal'].id},
        {'label': 'inner', 'parent': ('body', 1), 'caller': running['inner_call'].id,
         'region': running['inner_driver'].id, 'path': running['path'],
         'completion': running['host_terminal'].id},
        {'label': 'final', 'parent': 'caller', 'caller': details['extract'].id,
         'region': stage['region'].id, 'path': stage['paths'][ExitKind.NORMAL],
         'completion': stage['terminals'][ExitKind.NORMAL].id},
    )
    model, capture = capture_program(graph, details, instances=instances,
        label='native.running', body_paths={1: running['body_path']}, extra_calls=calls)
    if missing_inner_call is not None:
        link = capture['instances'][missing_inner_call]['extra_call_activations']['inner']
        model = replace(model, call_activations=tuple(row for row in model.call_activations if row.id != link.id))
    if missing_sent_use is not None:
        example = capture['instances'][missing_sent_use]
        drive = example['driver_scopes'][1][2][running['active_drive'].operation]
        resume = next(row for row in running['active_drive'].resume_inputs if row.slot == running['resume_slot'])
        model = replace(model, uses=tuple(row for row in model.uses
            if (row.operation, row.use) != (drive, resume.source)))
    return graph, model, capture
