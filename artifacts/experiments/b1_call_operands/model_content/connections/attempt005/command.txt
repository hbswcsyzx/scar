from scar.ir.execution.validation import validate
from tests.fixtures.execution_content_graphs import content_connection_fixture, native_content_fixture

cases = [
    ("host-data", dict()),
    ("host-reference", dict(kind="reference")),
    ("work-two-tickets", dict(consumer="work", duplicate_use=True)),
    ("observation", dict(observation=True)),
]
cases += [(f"native-{completion}", (native_content_fixture, {"completion": completion}))
          for completion in ("return", "raise")]
for name, options in cases:
    if isinstance(options, tuple):
        builder, kwargs = options
        program, model, values, evidence, details = builder(**kwargs)
    else:
        program, model, values, evidence, details = content_connection_fixture(**options)
    report = validate(model, program, value_graphs=values, evidence_graphs=evidence,
                      semantic_graphs=details.get("semantic_graphs", {}))
    print(name, "valid=", report.valid,
          "issues=", [(item.code, item.path) for item in report.issues],
          "gaps=", sorted({item.code for item in report.gaps}))
    if not report.valid:
        raise SystemExit(1)
PY
